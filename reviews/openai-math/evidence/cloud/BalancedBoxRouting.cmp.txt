Chal.BoxTransport.Routing.balanced_box_routing : ∀ (n : ℕ) (a b h k : Fin n → Chal.BoxTransport.Routing.RationalSpace),
  (∀ (i : Fin n) (j : Fin 3), 0 < h i j) →
    (∀ (i : Fin n) (j : Fin 3), 0 < k i j) →
      (Pairwise fun i j =>
          Disjoint (Chal.BoxTransport.Routing.solidBox (a i) (h i)) (Chal.BoxTransport.Routing.solidBox (a j) (h j))) →
        (Pairwise fun i j =>
            Disjoint (Chal.BoxTransport.Routing.solidBox (b i) (k i))
              (Chal.BoxTransport.Routing.solidBox (b j) (k j))) →
          (∀ (i : Fin n), ∏ j, k i j / h i j = 1) →
            ∃ U Φ,
              Chal.BoxTransport.Routing.AdmissibleField U ∧
                Chal.BoxTransport.Routing.IsGlobalFlow U Φ ∧ Chal.BoxTransport.Routing.Delivers a b h k Φ
