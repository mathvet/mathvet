def OAI.RelativeGeneration.MainTarget.{level} : Prop :=
∀ (Hilbert : Type level) [inst : NormedAddCommGroup Hilbert] [inst_1 : InnerProductSpace ℂ Hilbert]
  [inst_2 : CompleteSpace Hilbert] (small large : StarSubalgebra ℂ (Hilbert →L[ℂ] Hilbert)),
  OAI.RelativeGeneration.IsII1Factor small →
    OAI.RelativeGeneration.IsII1Factor large →
      ∀ (inclusion : small ≤ large),
        (∀ (element : ↥large),
            (∀ (member : ↥small),
                (StarSubalgebra.inclusion inclusion) member * element =
                  element * (StarSubalgebra.inclusion inclusion) member) →
              ∃ scalar, element = (algebraMap ℂ ↥large) scalar) →
          OAI.RelativeGeneration.HasSeparablePredual large →
            ∃ trace,
              OAI.RelativeGeneration.IsNormalizedTrace large trace ∧
                IsGδ (OAI.RelativeGeneration.relativeGeneratorLocus small large) ∧
                  Dense (OAI.RelativeGeneration.relativeGeneratorLocus small large)
