Chal.BinaryMatching.deterministic_approximate_counting : ∃ A machine,
  (∀ (k : machine.tm.K), Finite (machine.tm.Γ k)) ∧
    (∃ P,
        ∀ (G : Chal.BinaryMatching.Input),
          (A G).bits.length ≤ Polynomial.eval (Chal.BinaryMatching.encodeInput G).length P) ∧
      ∀ (G : Chal.BinaryMatching.Input),
        A G ≤ Chal.BinaryMatching.count G ∧
          Chal.BinaryMatching.count G ≤ 2 ^ (9 * G.n) * A G ∧ (A G = 0 ↔ Chal.BinaryMatching.count G = 0)

  {E : Type u_2} → Chal.MatchingEntropy.LooplessGraph V E → V → E → Prop :=
fun {V} {E} G v e => G.left e = v ∨ G.right e = v

  {E : Type u_2} → Chal.MatchingEntropy.LooplessGraph V E → Finset E → Prop :=
fun {V} {E} G M => ∀ (v : V), ∃! e, e ∈ M ∧ G.Incident v e

@[reducible] def Chal.BinaryMatching.Pair : ℕ → Type :=
fun n => { ij // ij.1 < ij.2 }
def Chal.BinaryMatching.completeGraph : (n : ℕ) →
  Chal.MatchingEntropy.LooplessGraph (Fin n) (Chal.BinaryMatching.Pair n) :=
fun n => { left := fun e => (↑e).1, right := fun e => (↑e).2, loopless := ⋯ }

def Chal.BinaryMatching.multiplicity : (G : Chal.BinaryMatching.Input) → Chal.BinaryMatching.Pair G.n → ℕ :=
fun G e =>
  match List.find? (fun r => decide (r.left = ↑(↑e).1) && decide (r.right = ↑(↑e).2)) G.records with
  | none => 0
  | some r => r.multiplicity
def Chal.BinaryMatching.count : Chal.BinaryMatching.Input → ℕ :=
fun G => ∑ M, ∏ e ∈ ↑M, Chal.BinaryMatching.multiplicity G e
def Chal.BinaryMatching.encodeNat : ℕ → List Bool :=
fun n => List.replicate n.bits.length false ++ true :: n.bits
def Chal.BinaryMatching.encodeRecord : Chal.BinaryMatching.Record → List Bool :=
fun r =>
  Chal.BinaryMatching.encodeNat r.left ++ Chal.BinaryMatching.encodeNat r.right ++
    Chal.BinaryMatching.encodeNat r.multiplicity
def Chal.BinaryMatching.encodeInput : Chal.BinaryMatching.Input → List Bool :=
fun G =>
  Chal.BinaryMatching.encodeNat G.n ++ Chal.BinaryMatching.encodeNat G.records.length ++
    List.flatMap Chal.BinaryMatching.encodeRecord G.records
