def Chal.HarmonicCounterexample.Complete : {n : ℕ} → Chal.HarmonicCounterexample.SmoothMetric n → Prop :=
fun {n} g =>

def Chal.HarmonicCounterexample.EuclideanNearOrigin : {n : ℕ} → Chal.HarmonicCounterexample.SmoothMetric n → Prop :=
fun {n} g => ∃ ε, 0 < ε ∧ ∀ (x : Chal.HarmonicCounterexample.Space n), ∑ i, x i ^ 2 < ε ^ 2 → g.coeff x = 1
def Chal.HarmonicCounterexample.ballVolume : {n : ℕ} → Chal.HarmonicCounterexample.SmoothMetric n → ℝ → ENNReal :=

def Chal.HarmonicCounterexample.omega : ℕ → ENNReal :=

def Chal.HarmonicCounterexample.PositiveSubunitAVR : {n : ℕ} → Chal.HarmonicCounterexample.SmoothMetric n → Prop :=
fun {n} g =>
  ∃ a,
    0 < a ∧
      a < 1 ∧
        Filter.Tendsto
          (fun R =>
            Chal.HarmonicCounterexample.ballVolume g R / (Chal.HarmonicCounterexample.omega n * ENNReal.ofReal (R ^ n)))
          Filter.atTop (nhds (ENNReal.ofReal a))
def Chal.HarmonicCounterexample.HarmonicGrowth : {n : ℕ} →
  Chal.HarmonicCounterexample.SmoothMetric n → ℕ → (Chal.HarmonicCounterexample.Space n → ℝ) → Prop :=
fun {n} g k u =>
  ContDiff ℝ (↑⊤) u ∧
    (∀ (x : Chal.HarmonicCounterexample.Space n), Chal.HarmonicCounterexample.laplaceBeltrami g u x = 0) ∧
      ∃ C,
        0 ≤ C ∧
          ∀ (x : Chal.HarmonicCounterexample.Space n), |u x| ≤ C * (1 + Chal.HarmonicCounterexample.distance g 0 x) ^ k
def Chal.HarmonicCounterexample.euclideanDimension : ℕ → ℕ → ℕ :=
fun n k => (n + k - 1).choose k + (n + k - 2).choose (k - 1)
def Chal.HarmonicCounterexample.MainClaim : Prop :=
∃ n k g,
  Even n ∧
    8 ≤ n ∧
      2 ≤ k ∧
        Chal.HarmonicCounterexample.Complete g ∧
          Chal.HarmonicCounterexample.RicciNonnegative g ∧
            Chal.HarmonicCounterexample.EuclideanNearOrigin g ∧
              Chal.HarmonicCounterexample.PositiveSubunitAVR g ∧
                ∃ u,
                  LinearIndependent ℝ u ∧
                    ∀ (i : Fin (Chal.HarmonicCounterexample.euclideanDimension n k + 1)),
                      Chal.HarmonicCounterexample.HarmonicGrowth g k (u i)
