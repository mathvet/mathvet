def Chal.GammaFPP.HasC1CurveChart : Set Chal.PlanarFPP.Plane → Chal.PlanarFPP.Plane → Prop :=
fun S v =>
  ∃ U,
    IsOpen U ∧
      v ∈ U ∧
        v ∈ S ∧
          ∃ γ θ,
            ContDiff ℝ 1 γ ∧
              ContinuousOn θ U ∧
                (∀ (t : ℝ), γ t ∈ S ∩ U) ∧
                  (∀ (t : ℝ), θ (γ t) = t) ∧ (∀ z ∈ S ∩ U, γ (θ z) = z) ∧ ∀ (t : ℝ), ∃ d, HasDerivAt γ d t ∧ d ≠ 0
