OAI.Brennan.Sharp.sharp_endpoints : OAI.Brennan.Sharp.SharpEndpointStatement
'OAI.Brennan.Sharp.sharp_endpoints' depends on axioms: [propext, Classical.choice, Quot.sound]
def OAI.Brennan.disk : Set ℂ :=
Metric.ball 0 1
def OAI.Brennan.UnivalentOn : (ℂ → ℂ) → Set ℂ → Prop :=
fun f U => DifferentiableOn ℂ f U ∧ Set.InjOn f U
def OAI.Brennan.Schlicht : (ℂ → ℂ) → Prop :=
fun f => OAI.Brennan.UnivalentOn f OAI.Brennan.disk ∧ f 0 = 0 ∧ deriv f 0 = 1
def OAI.Brennan.koebeMap : ℂ → ℂ :=
fun z => z / (1 - z) ^ 2
def OAI.Brennan.Sharp.koebeDomain : Set ℂ :=
OAI.Brennan.koebeMap '' OAI.Brennan.disk
def OAI.Brennan.Sharp.koebeInverse : ℂ → ℂ :=
Function.invFunOn OAI.Brennan.koebeMap OAI.Brennan.disk
def OAI.Brennan.Sharp.areaMoment : (ℂ → ℂ) → Set ℂ → ℝ → ENNReal :=
fun f U t => ∫⁻ (z : ℂ) in U, ENNReal.ofReal (‖deriv f z‖ ^ t)
def OAI.Brennan.Sharp.SharpEndpointStatement : Prop :=
OAI.Brennan.Schlicht OAI.Brennan.koebeMap ∧
  OAI.Brennan.Sharp.areaMoment OAI.Brennan.koebeMap OAI.Brennan.disk (-2) = ⊤ ∧
    OAI.Brennan.Sharp.areaMoment OAI.Brennan.koebeMap OAI.Brennan.disk (2 / 3) = ⊤ ∧
      ¬MeasureTheory.IntegrableOn (fun z => ‖deriv OAI.Brennan.Sharp.koebeInverse z‖ ^ 4) OAI.Brennan.Sharp.koebeDomain
            MeasureTheory.volume ∧
        ¬MeasureTheory.IntegrableOn (fun z => ‖deriv OAI.Brennan.Sharp.koebeInverse z‖ ^ (4 / 3))
            OAI.Brennan.Sharp.koebeDomain MeasureTheory.volume
