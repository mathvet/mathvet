fun n => List (Fin n × Bool)
def OAI.FiniteAlgebraicEnvelopes.evalWord.{u} : {G : Type u} →
  [Group G] → {n : ℕ} → (Fin n → G) → OAI.FiniteAlgebraicEnvelopes.Word n → G :=
fun {G} [Group G] {n} s w => (List.map (fun a => if a.2 = true then s a.1 else (s a.1)⁻¹) w).prod

fun G [Group G] =>

def OAI.FiniteAlgebraicEnvelopes.EmbedsInFinitelyPresentedSimpleGroup.{u} : (G : Type u) → [Group G] → Prop :=
fun G [Group G] => ∃ H x, Group.IsFinitelyPresented H ∧ IsSimpleGroup H ∧ ∃ f, Function.Injective ⇑f
