@Chal.SymmetricMahler.symmetric_mahler_equality : ∀ {n : ℕ},
  1 ≤ n →
    ∀ {K : Set (Fin n → ℝ)},
      IsCompact K →
        Convex ℝ K →
          (∀ x ∈ K, -x ∈ K) →
            (interior K).Nonempty →
              ((MeasureTheory.volume K).toReal *
                    (MeasureTheory.volume (Chal.SymmetricMahler.coordinatePolar K)).toReal =
                  4 ^ n / ↑n.factorial ↔
                Chal.SymmetricMahler.IsLinearHanner K)
def Chal.SymmetricMahler.coordinatePolar.{u_1} : {I : Type u_1} → [Fintype I] → Set (I → ℝ) → Set (I → ℝ) :=
fun {I} [Fintype I] K => {p | ∀ v ∈ K, ∑ i, p i * v i ≤ 1}
def Chal.SymmetricMahler.hannerProduct : {k l : ℕ} → Set (Fin k → ℝ) → Set (Fin l → ℝ) → Set (Fin (k + l) → ℝ) :=
fun {k l} A B => {z | ∃ x ∈ A, ∃ y ∈ B, z = Fin.append x y}
def Chal.SymmetricMahler.hannerJoin : {k l : ℕ} → Set (Fin k → ℝ) → Set (Fin l → ℝ) → Set (Fin (k + l) → ℝ) :=
fun {k l} A B => (convexHull ℝ) ((fun x => Fin.append x 0) '' A ∪ (fun y => Fin.append 0 y) '' B)
