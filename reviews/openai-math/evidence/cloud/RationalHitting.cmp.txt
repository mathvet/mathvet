def Chal.RationalHitting.OutputsWithin : {m n : ℕ} →
  Chal.RationalHitting.Machine m → ℕ → ℕ → Chal.RationalHitting.Output n → Prop :=
fun {m n} G s bound H =>
  ∃ t ≤ bound,

      Chal.RationalHitting.run G t (Turing.TM0.init (Chal.RationalHitting.unaryInput n s)) = some a ∧
        Turing.TM0.step G a = none ∧

            Turing.ListBlank.mk (List.map Chal.RationalHitting.tapeBit (Chal.RationalHitting.encodeOutput H))
def Chal.RationalHitting.Hits : {n : ℕ} → ℕ → Chal.RationalHitting.Output n → Prop :=
fun {n} s H =>
  ∀ (f : Chal.RationalHitting.Formula n),
    f.size ≤ s →
      Chal.RationalHitting.Admissible f →
        Chal.RationalHitting.Nonzero f → ∃ X ∈ H.tuples, ∃ v, Chal.RationalHitting.Evaluates f X v ∧ IsUnit v
def Chal.RationalHitting.MainStatement : Prop :=
∃ m G C k,
  0 < C ∧
    0 < k ∧
      ∀ (n s : ℕ),
        1 ≤ n →
          1 ≤ s →
            ∃ H,
              H.dimension ≤ C * (n + s + 1) ^ k ∧
                (Chal.RationalHitting.encodeOutput H).length ≤ C * (n + s + 1) ^ k ∧
                  Chal.RationalHitting.OutputsWithin G s (C * (n + s + 1) ^ k) H ∧ Chal.RationalHitting.Hits s H
