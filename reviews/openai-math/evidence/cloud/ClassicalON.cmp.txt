def Chal.ClassicalON.ExponentialDecay : Prop :=
∀ (n : ℕ),
  3 ≤ n →
    ∀ (β : ℝ),
      0 < β →
        ∃ A m,
          0 < m ∧
            ∀ (G : Chal.ClassicalON.LatticeGraph) (b : ↥G.edges → ℝ),
              (∀ (e : ↥G.edges), 0 ≤ b e ∧ b e ≤ β) →
                ∀ (x y : ↥G.vertices),
                  0 ≤ Chal.ClassicalON.correlation n G b x y ∧
                    Chal.ClassicalON.correlation n G b x y ≤ A * Real.exp (-m * Chal.ClassicalON.siteDistance ↑x ↑y)
