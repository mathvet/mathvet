def Chal.QuarticPowerFree.PowerFree : ℕ → ℤ → Prop :=
fun k a => ∀ (p : ℕ), Nat.Prime p → ¬↑p ^ k ∣ a
def Chal.QuarticPowerFree.rho : Polynomial ℤ → ℕ → ℕ :=
fun f q => {a ∈ Finset.range q | ↑q ∣ Polynomial.eval (↑a) f}.card
def Chal.QuarticPowerFree.count : Polynomial ℤ → ℕ → ℝ → ℕ :=
fun f k X => {n ∈ Finset.Icc 1 ⌊X⌋₊ | Chal.QuarticPowerFree.PowerFree k (Polynomial.eval (↑n) f)}.card
def Chal.QuarticPowerFree.LocallyAdmissible : Polynomial ℤ → ℕ → Prop :=
fun f k => ∀ (p : ℕ), Nat.Prime p → Chal.QuarticPowerFree.rho f (p ^ k) < p ^ k
def Chal.QuarticPowerFree.localFactor : Polynomial ℤ → ℕ → Nat.Primes → ℝ :=
fun f k p => 1 - ↑(Chal.QuarticPowerFree.rho f (↑p ^ k)) / ↑↑p ^ k
def Chal.QuarticPowerFree.densityConstant : Polynomial ℤ → ℕ → ℝ :=
fun f k => ∏' (p : Nat.Primes), Chal.QuarticPowerFree.localFactor f k p
def Chal.QuarticPowerFree.DensityStatement : Polynomial ℤ → ℕ → Prop :=
fun f k =>
  Multipliable (Chal.QuarticPowerFree.localFactor f k) ∧
