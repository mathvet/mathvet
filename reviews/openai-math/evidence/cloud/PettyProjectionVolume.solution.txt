OAI.PettyProjection.petty_projection_volume : ∀ (n : ℕ),

    ∀ (K : Set (OAI.PettyProjection.Space n)),
      OAI.PettyProjection.IsConvexBody K →

@[reducible] def OAI.PettyProjection.Space : ℕ → Type :=
fun n => EuclideanSpace ℝ (Fin n)
def OAI.PettyProjection.IsConvexBody : {n : ℕ} → Set (OAI.PettyProjection.Space n) → Prop :=
fun {n} K => IsCompact K ∧ Convex ℝ K ∧ (interior K).Nonempty
def OAI.PettyProjection.unitBall : (n : ℕ) → Set (OAI.PettyProjection.Space n) :=
fun n => Metric.closedBall 0 1
def OAI.PettyProjection.kappa : ℕ → ℝ :=
fun n => MeasureTheory.volume.real (OAI.PettyProjection.unitBall n)
def OAI.PettyProjection.perpendicular : {n : ℕ} →

def OAI.PettyProjection.shadowVolume : {n : ℕ} → Set (OAI.PettyProjection.Space n) → OAI.PettyProjection.Space n → ℝ :=
fun {n} K u => MeasureTheory.volume.real (⇑(OAI.PettyProjection.perpendicular u).orthogonalProjectionOnto '' K)
def OAI.PettyProjection.projectionBody : {n : ℕ} →
  Set (OAI.PettyProjection.Space n) → Set (OAI.PettyProjection.Space n) :=
fun {n} K => {x | ∀ (u : OAI.PettyProjection.Space n), ‖u‖ = 1 → inner ℝ u x ≤ OAI.PettyProjection.shadowVolume K u}
def OAI.PettyProjection.projectionRatio : {n : ℕ} → Set (OAI.PettyProjection.Space n) → ℝ :=
fun {n} K => MeasureTheory.volume.real (OAI.PettyProjection.projectionBody K) / MeasureTheory.volume.real K ^ (n - 1)
def OAI.PettyProjection.pettyConstant : ℕ → ℝ :=
fun n => OAI.PettyProjection.kappa (n - 1) ^ n * OAI.PettyProjection.kappa n ^ (2 - ↑n)
def OAI.PettyProjection.IsEllipsoid : {n : ℕ} → Set (OAI.PettyProjection.Space n) → Prop :=
fun {n} K => ∃ a T, K = (fun x => a + T x) '' OAI.PettyProjection.unitBall n
