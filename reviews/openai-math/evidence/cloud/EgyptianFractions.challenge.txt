def OAI.Problem337.IsEgyptianExpansion : {k : ℕ} → ℚ → (Fin k → ℕ) → Prop :=
fun {k} x n => (∀ (i : Fin k), 2 ≤ n i) ∧ StrictMono n ∧ ∑ i, 1 / ↑(n i) = x
def OAI.Problem337.egyptianLengths : ℕ → ℕ → Set ℕ :=
fun a b => {k | ∃ n, OAI.Problem337.IsEgyptianExpansion (↑a / ↑b) n}
def OAI.Problem337.egyptianLength : ℕ → ℕ → ℕ :=
fun a b => sInf (OAI.Problem337.egyptianLengths a b)
def OAI.Problem337.maxEgyptianLength : ℕ → ℕ :=
fun b => (Finset.range b).sup fun a => if 1 ≤ a then OAI.Problem337.egyptianLength a b else 0
def OAI.Problem337.IsOneExpansion : {k : ℕ} → (Fin k → ℕ) → Prop :=
fun {k} n => (∀ (i : Fin k), 1 ≤ n i) ∧ StrictMono n ∧ ∑ i, 1 / ↑(n i) = 1
def OAI.Problem337.OneExpansions : (k : ℕ) → Set (Fin k → ℕ) :=
fun k => {n | OAI.Problem337.IsOneExpansion n}
def OAI.Problem337.F : ℕ → ℕ :=
fun k => (OAI.Problem337.OneExpansions k).ncard
def OAI.Problem337.D : ℕ → Set ℕ :=
fun k => {m | 2 ≤ m ∧ ∃ n, OAI.Problem337.IsOneExpansion n ∧ ∃ i, n i = m}
def OAI.Problem337.missingDenominators : ℕ → Set ℕ :=
fun k => {m | 2 ≤ m} \ OAI.Problem337.D k
def OAI.Problem337.v : ℕ → ℕ :=
fun k => sInf (OAI.Problem337.missingDenominators k)
def OAI.Problem337.prescribedSlope : ℕ → ℝ :=
fun k => Real.log (Real.log ↑(OAI.Problem337.v k)) / ↑k
