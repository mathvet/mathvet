fun g C R =>
  (↑(OAI.TriangularUniversal.diskCount C R))⁻¹ *
    ∑ x ∈ OAI.TriangularUniversal.diskPoints C R,
      ∑ y ∈ (OAI.TriangularUniversal.diskPoints C R).erase x, ENNReal.ofReal (g (‖x - y‖ ^ 2))

fun g C => Filter.liminf (OAI.TriangularUniversal.diskEnergy g C) Filter.atTop
