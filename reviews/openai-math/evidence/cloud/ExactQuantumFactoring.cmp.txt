def Chal.ExactQuantumFactoring.Primitive.matrix : Chal.ExactQuantumFactoring.Primitive → List Bool → List Bool → ℂ :=
fun x x_1 x_2 =>
  match x, x_1, x_2 with
  | Chal.ExactQuantumFactoring.Primitive.not, [a], [b] => if a = !b then 1 else 0
  | Chal.ExactQuantumFactoring.Primitive.cnot, [a, b], [c, d] => if a = c ∧ b = (c ^^ d) then 1 else 0
  | Chal.ExactQuantumFactoring.Primitive.toffoli, [a, b, c], [d, e, f] =>
    if a = d ∧ b = e ∧ c = (f ^^ d && e) then 1 else 0
  | Chal.ExactQuantumFactoring.Primitive.hadamard, [a], [b] => (if (a && b) = true then -1 else 1) / ↑√2
  | Chal.ExactQuantumFactoring.Primitive.phase, [a], [b] => if a = b then if b = true then Complex.I else 1 else 0
  | x, x_3, x_4 => 0
def Chal.ExactQuantumFactoring.Gate.baseMatrix : Chal.ExactQuantumFactoring.Gate → List Bool → List Bool → ℂ :=
fun g a b => if g.inverse = true then star (g.primitive.matrix b a) else g.primitive.matrix a b
def Chal.ExactQuantumFactoring.Gate.matrix : Chal.ExactQuantumFactoring.Gate → List Bool → List Bool → ℂ :=
fun g a b =>
  if g.controlled = true then
    match a, b with
    | false :: a', false :: b' => if a' = b' then 1 else 0
    | true :: a', true :: b' => g.baseMatrix a' b'
    | x, x_1 => 0
  else g.baseMatrix a b

def Chal.ExactQuantumFactoring.outputList : {q : ℕ} → ℕ → Chal.ExactQuantumFactoring.Basis q → List ℕ :=
fun {q} ℓ x =>
  have n := Chal.ExactQuantumFactoring.paddedLength ℓ;
  List.ofFn fun a =>
    ∑ j ∈ Finset.range n, if Chal.ExactQuantumFactoring.readBit x (ℓ + ↑a * n + j) = true then 2 ^ j else 0
def Chal.ExactQuantumFactoring.CorrectEncoding : ℕ → ℕ → List ℕ → Prop :=
fun N n xs =>
  ∃ ps,
    (∀ p ∈ ps, Nat.Prime p) ∧
      List.Pairwise (fun x1 x2 => x1 ≤ x2) ps ∧
        ps.prod = N ∧ ps.length ≤ n ∧ xs = ps ++ List.replicate (n - ps.length) 0
def Chal.ExactQuantumFactoring.Circuit.correctProbability : Chal.ExactQuantumFactoring.Circuit → ℕ → ℕ → ℝ :=
fun c ℓ N =>
  ∑ x,
    if
        Chal.ExactQuantumFactoring.CorrectEncoding N (Chal.ExactQuantumFactoring.paddedLength ℓ)
          (Chal.ExactQuantumFactoring.outputList ℓ x) then
      Complex.normSq (c.outputState ℓ N x)
    else 0
def Chal.ExactQuantumFactoring.natWord : ℕ → List Bool :=
fun k => List.replicate k true ++ [false]
def Chal.ExactQuantumFactoring.Instruction.encode : {q : ℕ} → Chal.ExactQuantumFactoring.Instruction q → List Bool :=
fun {q} o =>
  Chal.ExactQuantumFactoring.natWord o.gate.primitive.code ++ [o.gate.inverse, o.gate.controlled] ++
    List.flatMap Chal.ExactQuantumFactoring.natWord (List.ofFn fun i => ↑(o.wire i))
def Chal.ExactQuantumFactoring.Circuit.encode : Chal.ExactQuantumFactoring.Circuit → List Bool :=
fun c =>
  Chal.ExactQuantumFactoring.natWord c.qubits ++ Chal.ExactQuantumFactoring.natWord c.instructions.length ++
    List.flatMap Chal.ExactQuantumFactoring.Instruction.encode c.instructions
def Chal.ExactQuantumFactoring.Uniform : (ℕ → Chal.ExactQuantumFactoring.Circuit) → Prop :=
fun family => ∃ generator, ∀ (k : generator.tm.K), Finite (generator.tm.Γ k)
def Chal.ExactQuantumFactoring.PolynomialResources : (ℕ → Chal.ExactQuantumFactoring.Circuit) → Prop :=
fun family =>
  ∃ bound,
    ∀ (ℓ : ℕ), (family ℓ).qubits ≤ Polynomial.eval ℓ bound ∧ (family ℓ).instructions.length ≤ Polynomial.eval ℓ bound
def Chal.ExactQuantumFactoring.MainTheorem : Prop :=
∃ family,
  Chal.ExactQuantumFactoring.Uniform family ∧
    Chal.ExactQuantumFactoring.PolynomialResources family ∧
      ∀ (ℓ N : ℕ),
        2 ≤ N →
          N.size = ℓ →
            ℓ + Chal.ExactQuantumFactoring.paddedLength ℓ ^ 2 ≤ (family ℓ).qubits ∧
              (family ℓ).correctProbability ℓ N = 1
