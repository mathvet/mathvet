Chal.RadialTransition.densityInterval : ∃ p φ B C η βc f,
  0 < p ∧
    Chal.RadialTransition.IsPackingDensity p ∧
      Chal.RadialTransition.IsBoundedPotential φ ∧
        Continuous φ ∧
          0 < B ∧
            0 < C ∧
              Chal.RadialTransition.StableWith φ B ∧
                Chal.RadialTransition.AlgebraicDecayWith φ C ∧
                  0 < η ∧
                    Set.Ioo (5 * p / 3 - η) (5 * p / 3 + η) ⊆ Set.Ioi 0 ∧
                      βc ∈ Set.Icc (7 / 8) (9 / 8) ∧
                        ∀ ρ ∈ Set.Ioo (5 * p / 3 - η) (5 * p / 3 + η),
                          (Chal.RadialTransition.IsCanonicalFreeEnergy φ ρ fun β => f β ρ) ∧
                            Chal.RadialTransition.StrictTemperatureCorner (fun β => f β ρ) βc
@[reducible] def Chal.RadialTransition.Position : Type :=
EuclideanSpace ℝ (Fin 3)
@[reducible] def Chal.RadialTransition.Potential : Type :=
NNReal → ℝ
def Chal.RadialTransition.cube : ℝ → Set Chal.RadialTransition.Position :=
fun L => {x | ∀ (k : Fin 3), 0 ≤ x.ofLp k ∧ x.ofLp k ≤ L}
def Chal.RadialTransition.configurationCube : ℝ → (N : ℕ) → Set (Fin N → Chal.RadialTransition.Position) :=
fun L N => {x | ∀ (i : Fin N), x i ∈ Chal.RadialTransition.cube L}
def Chal.RadialTransition.energy : Chal.RadialTransition.Potential →
  {N : ℕ} → (Fin N → Chal.RadialTransition.Position) → ℝ :=
fun φ {N} x => ∑ i, ∑ j with i < j, φ (nndist (x i) (x j))
def Chal.RadialTransition.partition : Chal.RadialTransition.Potential → ℝ → ℕ → ℝ → ℝ :=
fun φ L N β =>
  (↑N.factorial)⁻¹ *
    ∫ (x : Fin N → Chal.RadialTransition.Position) in Chal.RadialTransition.configurationCube L N,
      Real.exp (-β * Chal.RadialTransition.energy φ x)
def Chal.RadialTransition.packingNumber : ℝ → ℕ :=
fun L =>
  sSup {n | ∃ X, X.card = n ∧ (∀ x ∈ X, x ∈ Chal.RadialTransition.cube L) ∧ ∀ x ∈ X, ∀ y ∈ X, x ≠ y → 1 ≤ dist x y}
def Chal.RadialTransition.IsPackingDensity : ℝ → Prop :=
fun p => Filter.Tendsto (fun L => ↑(Chal.RadialTransition.packingNumber L) / L ^ 3) Filter.atTop (nhds p)
def Chal.RadialTransition.IsBoundedPotential : Chal.RadialTransition.Potential → Prop :=
fun φ => ∃ M, 0 ≤ M ∧ ∀ (r : NNReal), |φ r| ≤ M
def Chal.RadialTransition.StableWith : Chal.RadialTransition.Potential → ℝ → Prop :=
fun φ B => ∀ (N : ℕ) (x : Fin N → Chal.RadialTransition.Position), -(B * ↑N) ≤ Chal.RadialTransition.energy φ x
def Chal.RadialTransition.AlgebraicDecayWith : Chal.RadialTransition.Potential → ℝ → Prop :=
fun φ C => ∀ (r : NNReal), 1 ≤ r → |φ r| ≤ C * ↑r ^ (-3 - 1 / 32)
def Chal.RadialTransition.IsCanonicalFreeEnergy : Chal.RadialTransition.Potential → ℝ → (ℝ → ℝ) → Prop :=
fun φ ρ f =>
  ∀ (β : ℝ),
    0 < β →
      Filter.Tendsto (fun L => -(Real.log (Chal.RadialTransition.partition φ L ⌊ρ * L ^ 3⌋₊ β) / (β * L ^ 3)))
        Filter.atTop (nhds (f β))
def Chal.RadialTransition.StrictTemperatureCorner : (ℝ → ℝ) → ℝ → Prop :=
fun f βc =>
  ∃ dLeft dRight,
    HasDerivWithinAt f dLeft (Set.Iic βc) βc ∧
      HasDerivWithinAt f dRight (Set.Ici βc) βc ∧ dRight < dLeft ∧ ¬DifferentiableAt ℝ f βc
