def Chal.SplitTangent.Integrable.{u} : (X : Chal.SplitTangent.ComplexManifold) →
  ((x : X.Carrier) → Submodule ℂ (Chal.SplitTangent.Tangent X x)) → Prop :=
fun X D =>
  ∀ (U : Set X.Carrier),
    IsOpen U →
      ∀ (V W : (x : X.Carrier) → Chal.SplitTangent.Tangent X x),
        Chal.SplitTangent.HolomorphicSectionOn X V U →
          Chal.SplitTangent.HolomorphicSectionOn X W U →
            (∀ x ∈ U, V x ∈ D x) →
              (∀ x ∈ U, W x ∈ D x) →
                ∀ x ∈ U, VectorField.mlieBracketWithin (Chal.SplitTangent.ModelWithCorners X) V W U x ∈ D x
def Chal.SplitTangent.IntegrabilityStatement.{u} : Prop :=
∀ (X : Chal.SplitTangent.ComplexManifold),
  ConnectedSpace X.Carrier →
    CompactSpace X.Carrier →
      2 ≤ X.dim →
        ∀ (e : Chal.SplitTangent.ProjectiveEmbedding X),
          Chal.SplitTangent.RationallyConnected e →
            ∀ (S : Chal.SplitTangent.TangentSplitting X),
              Chal.SplitTangent.Integrable X S.first ∧ Chal.SplitTangent.Integrable X S.second
