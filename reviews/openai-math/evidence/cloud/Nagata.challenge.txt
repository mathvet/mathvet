def OAI.Nagata.FullNagata : Prop :=
∀ (count : ℕ),
  10 ≤ count →
    ∃ exceptional,
      (∀ (index : ℕ),
          OAI.Nagata.ProjectiveGeometry.IsConfigurationZariskiClosed (exceptional index) ∧
            exceptional index ≠ Set.univ) ∧
        (∃ points, ∀ (index : ℕ), points ∉ exceptional index) ∧
          ∀ (points : OAI.Nagata.ProjectiveGeometry.OrderedDistinctPoints count),
            (∀ (index : ℕ), points ∉ exceptional index) →
              ∀ (degree : ℕ),
                1 ≤ degree →
                  ∀ (polynomial : MvPolynomial (Fin 3) ℂ),
                    polynomial ≠ 0 →
                      polynomial.IsHomogeneous degree →
                        ∀ (multiplicities : Fin count → ℕ),
                          (∀ (index : Fin count),
                              OAI.Nagata.ProjectiveGeometry.multiplicityAtLeast polynomial (↑points index)
                                (multiplicities index)) →
                            ∑ index, ↑(multiplicities index) < ↑degree * √↑count
