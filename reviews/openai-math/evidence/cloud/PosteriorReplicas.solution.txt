def OAI.PosteriorReplicas.terminalConditionalEntropy : {d V W : ℕ} →

  OAI.PosteriorReplicas.finiteEntropy (MeasureTheory.Measure.map (fun z => (z.1.2, z.2)) q) -
    OAI.PosteriorReplicas.finiteEntropy (MeasureTheory.Measure.map (fun z => z.1.2) q)
def OAI.PosteriorReplicas.HaarReplicaBlock : Prop :=
∃ C,
  0 < C ∧
    ∀ (d : ℕ),
      64 ≤ d →
        ∀ (V W : ℕ) [NeZero V] [inst : NeZero W] (q : MeasureTheory.Measure (OAI.PosteriorReplicas.Vec d × Fin V))
          [inst_1 : MeasureTheory.IsProbabilityMeasure q],
          q.fst = OAI.PosteriorReplicas.sphereLaw d →
            ∀ (κ : ProbabilityTheory.Kernel (Fin V × OAI.PosteriorReplicas.Data d (d / 16)) (Fin W))
              [inst_2 : ProbabilityTheory.IsMarkovKernel κ],
              OAI.PosteriorReplicas.cmi
                  (OAI.PosteriorReplicas.stateHaarSideLaw (d / 4) (OAI.PosteriorReplicas.stateMessageLaw q κ)) ≤
                ENNReal.ofReal
                  (OAI.PosteriorReplicas.terminalConditionalEntropy (OAI.PosteriorReplicas.stateMessageLaw q κ) /
                          ↑(d / 16 + 1) +
                        C * ↑d +
                      Real.log (1 + C * ↑d + OAI.PosteriorReplicas.finiteEntropy q.snd) +
                    C * (OAI.PosteriorReplicas.finiteEntropy q.snd + Real.log ↑d) / ↑d ^ 2)

                      MeasureTheory.IsProbabilityMeasure ρ →
                        ∀ (E : OAI.PosteriorRegression.JointKernelExperiment d (M d) T Ω ρ),
                          2 / 3 ≤ E.uniformSuccess ε → c * ↑d * Real.log (1 / ε) ≤ ↑T
