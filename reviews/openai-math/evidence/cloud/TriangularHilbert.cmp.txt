        (∀ᵐ (z : Chal.TriangularHilbert.Plane), Chal.TriangularHilbert.GoodPoint F G z) ∧
          AEMeasurable (Chal.TriangularHilbert.maximal F G) MeasureTheory.volume ∧
            Chal.TriangularHilbert.maximalNorm F G ≤
              ↑C * MeasureTheory.eLpNorm F 3 MeasureTheory.volume * MeasureTheory.eLpNorm G 3 MeasureTheory.volume
