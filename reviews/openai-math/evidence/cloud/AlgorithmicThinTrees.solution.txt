def OAI.StrongThinTree.MultiGraph.cut : {n m : ℕ} →
  OAI.StrongThinTree.MultiGraph n m → Finset (Fin m) → Finset (Fin n) → Finset (Fin m) :=
fun {n m} G T S => {e ∈ T | G.left e ∈ S ∧ G.right e ∉ S ∨ G.right e ∈ S ∧ G.left e ∉ S}
def OAI.StrongThinTree.MultiGraph.Connected : {n m : ℕ} → OAI.StrongThinTree.MultiGraph n m → Finset (Fin m) → Prop :=
fun {n m} G T => ∀ (S : Finset (Fin n)), S.Nonempty → S ≠ Finset.univ → (G.cut T S).Nonempty
def OAI.StrongThinTree.MultiGraph.SpanningTree : {n m : ℕ} →
  OAI.StrongThinTree.MultiGraph n m → Finset (Fin m) → Prop :=
fun {n m} G T => G.Connected T ∧ ∀ e ∈ T, ¬G.Connected (T.erase e)
def OAI.StrongThinTree.MultiGraph.EdgeConnected : {n m : ℕ} → OAI.StrongThinTree.MultiGraph n m → ℕ → Prop :=
fun {n m} G k => ∀ (S : Finset (Fin n)), S.Nonempty → S ≠ Finset.univ → k ≤ (G.cut Finset.univ S).card

def OAI.CurrentKS.Machine.step : (M : OAI.CurrentKS.Machine) →
  OAI.CurrentKS.Config M.stackCount M.states → OAI.CurrentKS.Config M.stackCount M.states :=
fun M c =>
  match c.state with
  | none => c
  | some q =>
    have instruction :=
      M.table.getD (3 ^ M.stackCount * ↑q + OAI.CurrentKS.headsCode M.stackCount c.stacks)
        (OAI.CurrentKS.Instruction.halt M.stackCount M.states);
    { state := instruction.next, «stacks» := fun j => (instruction.actions j).apply (c.stacks j) }

fun M input => { state := some M.initial, «stacks» := fun j => if j = M.inputStack then input else [] }

  List Bool → ℕ → OAI.CurrentKS.Config M.stackCount M.states :=
fun M input fuel => M.step^[fuel] (M.init input)

def OAI.AlgorithmicThinTrees.ThinTree : {n m : ℕ} → ℝ → ℕ → OAI.StrongThinTree.MultiGraph n m → Finset (Fin m) → Prop :=
fun {n m} C k G T =>
  G.SpanningTree T ∧
    ∀ (S : Finset (Fin n)), S.Nonempty → S ≠ Finset.univ → ↑(G.cut T S).card ≤ C / ↑k * ↑(G.cut Finset.univ S).card

def OAI.AlgorithmicThinTrees.TreeOutput : ℝ → OAI.AlgorithmicThinTrees.Input → List Bool → Prop :=
fun C x x_1 =>
  match x, x_1 with
  | OAI.AlgorithmicThinTrees.Input.explicit x, bits =>
    ∃ edges, edges.Nodup ∧ bits = x.encodeEdges edges ∧ OAI.AlgorithmicThinTrees.ThinTree C x.k x.graph edges.toFinset
  | OAI.AlgorithmicThinTrees.Input.binary x, bits =>
    ∃ edges,
      edges.Nodup ∧ bits = x.encodeEdges edges ∧ OAI.AlgorithmicThinTrees.ThinTree C x.k x.graph (x.selected edges)
def OAI.AlgorithmicThinTrees.GoodOutput : ℝ → OAI.AlgorithmicThinTrees.Input → List Bool → Prop :=
fun C x bits => OAI.AlgorithmicThinTrees.TreeOutput C x bits ∧ (x.n = 1 → bits = [])
def OAI.AlgorithmicThinTrees.AlgorithmicStrongThinTrees : Prop :=
∃ C,
  0 < C ∧
    ∃ M a degree,
      0 < a ∧
        ∀ (x : OAI.AlgorithmicThinTrees.Input),
          OAI.AlgorithmicThinTrees.Valid x →
            have result := M.run x.encode (a * (x.length + 1) ^ degree);
            result.state = none ∧ OAI.AlgorithmicThinTrees.GoodOutput C x (result.stacks M.outputStack)
