structure OAI.ThreeMachine.Instance (n : ℕ) : Type
number of parameters: 1
fields:
  OAI.ThreeMachine.Instance.edges : List (Fin n × Fin n)
constructor:
  OAI.ThreeMachine.Instance.mk {n : ℕ} (edges : List (Fin n × Fin n)) : OAI.ThreeMachine.Instance n
def OAI.ThreeMachine.Instance.edge : {n : ℕ} → OAI.ThreeMachine.Instance n → Fin n → Fin n → Prop :=
fun {n} G u v => (u, v) ∈ G.edges
def OAI.ThreeMachine.Instance.acyclic : {n : ℕ} → OAI.ThreeMachine.Instance n → Prop :=
fun {n} G => ∀ (v : Fin n), ¬Relation.TransGen G.edge v v
def OAI.ThreeMachine.Feasible : {n : ℕ} → OAI.ThreeMachine.Instance n → ℕ → (Fin n → ℕ) → Prop :=
fun {n} G T τ =>
  (∀ (v : Fin n), 1 ≤ τ v ∧ τ v ≤ T) ∧ (∀ (t : ℕ), {v | τ v = t}.card ≤ 3) ∧ ∀ (u v : Fin n), G.edge u v → τ u < τ v
def OAI.ThreeMachine.encodeNat : ℕ → List Bool :=
fun n =>
  have b := Computability.encodeNat n;
  List.replicate b.length true ++ false :: b
def OAI.ThreeMachine.encodeList : {α : Type} → (α → List Bool) → List α → List Bool :=
fun {α} f xs => OAI.ThreeMachine.encodeNat xs.length ++ List.flatMap f xs
def OAI.ThreeMachine.encodeInput : {n : ℕ} → OAI.ThreeMachine.Instance n → Option ℕ → List Bool :=
fun {n} G deadline =>
  (match deadline with
      | none => [false]
      | some T => true :: OAI.ThreeMachine.encodeNat T) ++
      OAI.ThreeMachine.encodeList OAI.ThreeMachine.encodeNat (List.map (fun x => x + 1) (List.range n)) ++
    OAI.ThreeMachine.encodeList
      (fun e => OAI.ThreeMachine.encodeNat (↑e.1 + 1) ++ OAI.ThreeMachine.encodeNat (↑e.2 + 1)) G.edges
def OAI.ThreeMachine.encodeAnswer : Option (List ℕ) → List Bool :=
fun x =>
  match x with
  | none => [false]
  | some slots => true :: OAI.ThreeMachine.encodeList OAI.ThreeMachine.encodeNat slots
def OAI.ThreeMachine.slots : {n : ℕ} → (Fin n → ℕ) → List ℕ :=
fun {n} τ => List.ofFn τ
def OAI.ThreeMachine.CorrectOutput : {n : ℕ} → OAI.ThreeMachine.Instance n → Option ℕ → List Bool → Prop :=
fun {n} G deadline out =>
  match deadline with
  | none =>
    ∃ T τ,
      OAI.ThreeMachine.Feasible G T τ ∧
        (∀ (T' : ℕ) (τ' : Fin n → ℕ), OAI.ThreeMachine.Feasible G T' τ' → T ≤ T') ∧
          out = OAI.ThreeMachine.encodeAnswer (some (OAI.ThreeMachine.slots τ))
  | some T =>
    (out = OAI.ThreeMachine.encodeAnswer none ∧ ¬∃ τ, OAI.ThreeMachine.Feasible G T τ) ∨
      ∃ τ, OAI.ThreeMachine.Feasible G T τ ∧ out = OAI.ThreeMachine.encodeAnswer (some (OAI.ThreeMachine.slots τ))

def OAI.ThreeMachine.bitSymbol : {g : ℕ} → Bool → Fin (g + 3) :=
fun {g} b => if b = true then ⟨2, ⋯⟩ else ⟨1, ⋯⟩
def OAI.ThreeMachine.Machine.step : {k q g : ℕ} →
  OAI.ThreeMachine.Machine k q g →
    OAI.ThreeMachine.Configuration k q g → Option (OAI.ThreeMachine.Configuration k q g) :=
fun {k q g} M c => do
  let __x ← M.transition c.state fun i => (c.tapes i).head
  match __x with
    | (state, action) =>
      pure
        { state := state,
          tapes := fun i =>
            have tape := Turing.Tape.write (action i).1 (c.tapes i);
            match (action i).2 with
            | none => tape
            | some d => Turing.Tape.move d tape }
def OAI.ThreeMachine.Machine.initial : {k q g : ℕ} →
  OAI.ThreeMachine.Machine k q g → List Bool → OAI.ThreeMachine.Configuration k q g :=
fun {k q g} _M input =>
  { state := 0, tapes := fun i => Turing.Tape.mk₁ (if i = 0 then List.map OAI.ThreeMachine.bitSymbol input else []) }
def OAI.ThreeMachine.Machine.run : {k q g : ℕ} →
  OAI.ThreeMachine.Machine k q g → List Bool → ℕ → OAI.ThreeMachine.Configuration k q g :=
fun {k q g} M input t => (fun c => (M.step c).getD c)^[t] (M.initial input)
def OAI.ThreeMachine.Machine.Produces : {k q g : ℕ} →
  OAI.ThreeMachine.Machine k q g → List Bool → List Bool → ℕ → Prop :=
fun {k q g} M input out timeBound =>
  ∃ t ≤ timeBound,
    M.step (M.run input t) = none ∧
      ((M.run input t).tapes 0).right₀ = Turing.ListBlank.mk (List.map OAI.ThreeMachine.bitSymbol out)
def OAI.ThreeMachine.ReleaseTheorem : Prop :=
∃ k q g M C,
  0 < C ∧
    ∀ (n : ℕ) (G : OAI.ThreeMachine.Instance n),
      1 ≤ n →
        G.acyclic →
          ∀ (deadline : Option ℕ),
            (∀ T ∈ deadline, 1 ≤ T ∧ T ≤ n) →
              ∃ out,
                OAI.ThreeMachine.CorrectOutput G deadline out ∧
                  M.Produces (OAI.ThreeMachine.encodeInput G deadline) out
                    (C * ((OAI.ThreeMachine.encodeInput G deadline).length + 2) ^ 150020)
