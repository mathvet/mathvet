def Chal.TwoPointCorrelations.correlationSum : (ℕ → ℂ) → (ℕ → ℂ) → ℕ → ℕ → ℕ → ℂ :=
fun f g h₁ h₂ N => ∑ n ∈ Finset.Icc 1 N, f (n + h₁) * g (n + h₂)
def Chal.TwoPointCorrelations.affineSum : (ℕ → ℂ) → (ℕ → ℂ) → ℕ → ℕ → ℕ → ℕ → ℕ → ℂ :=
fun f g a₁ a₂ b₁ b₂ N => ∑ n ∈ Finset.Icc 1 N, f (a₁ * n + b₁) * g (a₂ * n + b₂)
def Chal.TwoPointCorrelations.Multiplicative : (ℕ → ℂ) → Prop :=
fun f => ∀ (m n : ℕ), 0 < m → 0 < n → m.Coprime n → f (m * n) = f m * f n
def Chal.TwoPointCorrelations.OneBounded : (ℕ → ℂ) → Prop :=
fun f => ∀ (n : ℕ), 0 < n → ‖f n‖ ≤ 1
def Chal.TwoPointCorrelations.primesUpTo : ℕ → Finset ℕ :=
fun N => Finset.filter Nat.Prime (Finset.range (N + 1))
def Chal.TwoPointCorrelations.squaredDistance : (ℕ → ℂ) → (ℕ → ℂ) → ℕ → ℝ :=
fun f g N => ∑ p ∈ Chal.TwoPointCorrelations.primesUpTo N, (1 - (f p * (starRingEnd ℂ) (g p)).re) / ↑p
