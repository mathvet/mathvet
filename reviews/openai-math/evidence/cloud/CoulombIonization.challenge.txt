def OAI.CoulombAtom.formEnergy : {N : ℕ} → ℝ → OAI.CoulombAtom.FormVector N → ℝ :=

  ((1 / 2 * ∑ s, ∑ i, ∑ a, ∫ (x : OAI.CoulombAtom.Configuration N), ‖ψ.gradient s i a x‖ ^ 2) -
      Z * ∑ s, ∑ i, ∫ (x : OAI.CoulombAtom.Configuration N), ‖ψ.value s x‖ ^ 2 / ‖x i‖) +

def OAI.CoulombAtom.energy : ℝ → ℕ → ℝ :=
fun Z N => if N = 0 then 0 else sInf {e | ∃ ψ, OAI.CoulombAtom.FormAdmissible ψ ∧ OAI.CoulombAtom.formEnergy Z ψ = e}
def OAI.CoulombAtom.ionization : ℕ → ℕ → ℝ :=
fun m Z => OAI.CoulombAtom.energy (↑Z) (Z - m) - OAI.CoulombAtom.energy (↑Z) Z
def OAI.CoulombAtom.tfKinetic : ℝ :=
3 / 10 * (3 * Real.pi ^ 2) ^ (2 / 3)
def OAI.CoulombAtom.directIntegrand : (OAI.CoulombAtom.Space → ℝ) → OAI.CoulombAtom.Space × OAI.CoulombAtom.Space → ℝ :=
fun ρ p => ρ p.1 * ρ p.2 / ‖p.1 - p.2‖

fun M ρ =>

      ∫ (x : OAI.CoulombAtom.Space), ρ x = M ∧

          MeasureTheory.Integrable (fun x => ρ x / ‖x‖) MeasureTheory.volume ∧
            MeasureTheory.Integrable (OAI.CoulombAtom.directIntegrand ρ) MeasureTheory.volume
def OAI.CoulombAtom.tfFunctional : ℝ → (OAI.CoulombAtom.Space → ℝ) → ℝ :=
fun Z ρ =>
  ((OAI.CoulombAtom.tfKinetic * ∫ (x : OAI.CoulombAtom.Space), ρ x ^ (5 / 3)) -
      Z * ∫ (x : OAI.CoulombAtom.Space), ρ x / ‖x‖) +
    1 / 2 * ∫ (p : OAI.CoulombAtom.Space × OAI.CoulombAtom.Space), OAI.CoulombAtom.directIntegrand ρ p

def OAI.CoulombAtom.tfIonization : ℝ → ℝ → ℝ :=
fun m Z => OAI.CoulombAtom.tfEnergy Z (Z - m) - OAI.CoulombAtom.tfEnergy Z Z

def OAI.CoulombAtom.JointLimit : ℝ → Prop :=
fun a =>
  ∀ (m Z : ℕ → ℕ),
    (∀ (j : ℕ), 1 ≤ m j ∧ m j < Z j) →
      Filter.Tendsto m Filter.atTop Filter.atTop →
        Filter.Tendsto (fun j => ↑(Z j) / ↑(m j)) Filter.atTop Filter.atTop →
          Filter.Tendsto (fun j => OAI.CoulombAtom.ionization (m j) (Z j) / ↑(m j) ^ (7 / 3)) Filter.atTop (nhds a)
def OAI.CoulombAtom.IteratedLimits : ℝ → Prop :=
fun a =>
  Filter.Tendsto (fun m => Filter.limsup (OAI.CoulombAtom.ionization m) Filter.atTop / ↑m ^ (7 / 3)) Filter.atTop
      (nhds a) ∧
    Filter.Tendsto (fun m => Filter.liminf (OAI.CoulombAtom.ionization m) Filter.atTop / ↑m ^ (7 / 3)) Filter.atTop
      (nhds a)
def OAI.CoulombAtom.MainStatement : Prop :=
∃ a, 0 < a ∧ OAI.CoulombAtom.TFCharacterization a ∧ OAI.CoulombAtom.JointLimit a ∧ OAI.CoulombAtom.IteratedLimits a
