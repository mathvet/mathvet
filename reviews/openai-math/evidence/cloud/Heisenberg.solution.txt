def OAI.Heisenberg.IsKMS : {d ℓ : ℕ} → ℝ → OAI.Heisenberg.State d ℓ → Prop :=
fun {d ℓ} β ω =>
  0 < β ∧
    ∀ (A B : ↥(OAI.Heisenberg.QuasiLocal d ℓ)),
      ∃ F,
        ContinuousOn F {z | 0 ≤ z.im ∧ z.im ≤ β} ∧
          DifferentiableOn ℂ F {z | 0 < z.im ∧ z.im < β} ∧
            (∃ C, ∀ (z : ℂ), 0 ≤ z.im → z.im ≤ β → ‖F z‖ ≤ C) ∧
              (∀ (t : ℝ), F ↑t = ω.functional (A * OAI.Heisenberg.dynamics d ℓ t B)) ∧
