def Chal.BorsukNine.projector : Chal.BorsukNine.Vector4 → Chal.BorsukNine.Matrix4 :=
fun u => WithLp.toLp 2 fun ij => u.ofLp ij.1 * u.ofLp ij.2

def Chal.BorsukNine.traceOneSymmetric : Set Chal.BorsukNine.Matrix4 :=
{A | (∀ (i j : Fin 4), A.ofLp (i, j) = A.ofLp (j, i)) ∧ ∑ i, A.ofLp (i, i) = 1}

    Chal.BorsukNine.projectorSet ⊆ ⋃ i, C i ∧ ∀ (i : Fin 10), Metric.diam (C i) < √2
