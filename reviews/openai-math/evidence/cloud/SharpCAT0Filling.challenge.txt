def OAI.CAT0Fillings.IsCAT0.{u} : (X : Type u) → [MetricSpace X] → Prop :=
fun X [MetricSpace X] =>
  ∃ segment,
    (∀ (x y : X), segment x y 0 = x ∧ segment x y 1 = y) ∧
      (∀ (x y : X) (s t : ℝ),
          s ∈ Set.Icc 0 1 → t ∈ Set.Icc 0 1 → dist (segment x y s) (segment x y t) = |s - t| * dist x y) ∧
        ∀ (o x y : X) (s t : ℝ),
          s ∈ Set.Icc 0 1 →
            t ∈ Set.Icc 0 1 →
              dist (segment o x s) (segment o y t) ^ 2 ≤
                (s * dist o x - t * dist o y) ^ 2 + s * t * (dist x y ^ 2 - (dist o x - dist o y) ^ 2)

def OAI.CAT0Fillings.sphereArea : ℕ → ℝ :=

fun n => 1 / ((↑n + 1) * OAI.CAT0Fillings.sphereArea n ^ (1 / ↑n))

fun n => (↑n + 1) / ↑n
