def Chal.Brennan.disk : Set ℂ :=
Metric.ball 0 1
def Chal.Brennan.UnivalentOn : (ℂ → ℂ) → Set ℂ → Prop :=
fun f U => DifferentiableOn ℂ f U ∧ Set.InjOn f U
def Chal.Brennan.Schlicht : (ℂ → ℂ) → Prop :=
fun f => Chal.Brennan.UnivalentOn f Chal.Brennan.disk ∧ f 0 = 0 ∧ deriv f 0 = 1
def Chal.Brennan.integralMean : (ℂ → ℂ) → ℝ → ℝ → ℝ :=
fun f t r => (2 * Real.pi)⁻¹ * ∫ (θ : ℝ) in -Real.pi..Real.pi, ‖deriv f (↑r * Complex.exp (↑θ * Complex.I))‖ ^ t
def Chal.Brennan.beta : (ℂ → ℂ) → ℝ → EReal :=
fun f t =>
  Filter.limsup (fun r => ↑(Real.log (Chal.Brennan.integralMean f t r) / Real.log (1 / (1 - r))))
    (nhdsWithin 1 (Set.Iio 1))
def Chal.Brennan.spectrum : ℝ → EReal :=
fun t => sSup {b | ∃ f, Chal.Brennan.Schlicht f ∧ Chal.Brennan.beta f t = b}
def Chal.Brennan.SphericalBoundaryNontrivial : Set ℂ → Prop :=
fun W => (frontier (OnePoint.some '' W)).Nontrivial
def Chal.Brennan.MainStatement : Prop :=
(∀ (ε : ℝ),
    0 < ε →
      ∃ C,
        ∀ (f : ℂ → ℂ),
          Chal.Brennan.Schlicht f →

  Chal.Brennan.spectrum (-2) = 1 ∧
    (∀ (W : Set ℂ),
        IsOpen W →
