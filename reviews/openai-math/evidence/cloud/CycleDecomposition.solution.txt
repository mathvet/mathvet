def OAI.ErdosGallai.CycleOrSingleEdge : {V : Type} → SimpleGraph V → Set (Sym2 V) → Prop :=
fun {V} G s => (∃ v p, p.IsCycle ∧ s = p.edgeSet) ∨ ∃ e ∈ G.edgeSet, s = {e}

fun {V} G k =>
  ∃ parts,
    (∀ (i : Fin k), OAI.ErdosGallai.CycleOrSingleEdge G (parts i)) ∧
      (Pairwise fun i j => Disjoint (parts i) (parts j)) ∧ ⋃ i, parts i = G.edgeSet
