@[reducible] def Chal.EuclideanRamsey.Space : ℕ → Type :=
fun d => EuclideanSpace ℝ (Fin d)
def Chal.EuclideanRamsey.Congruent : {s d D : ℕ} →
  (Fin s → Chal.EuclideanRamsey.Space d) → (Fin s → Chal.EuclideanRamsey.Space D) → Prop :=
fun {s d D} a b => ∀ (i j : Fin s), dist (b i) (b j) = dist (a i) (a j)
def Chal.EuclideanRamsey.Ramsey : {s d : ℕ} → (Fin s → Chal.EuclideanRamsey.Space d) → Prop :=
fun {s d} a =>
  ∀ (r : ℕ),
    2 ≤ r →
      ∃ D,
        1 ≤ D ∧
          ∀ (c : Chal.EuclideanRamsey.Space D → Fin r),
            ∃ b, Chal.EuclideanRamsey.Congruent a b ∧ ∃ k, ∀ (i : Fin s), c (b i) = k
def Chal.EuclideanRamsey.coordinateField : {s d : ℕ} → (Fin s → Chal.EuclideanRamsey.Space d) → IntermediateField ℚ ℝ :=
fun {s d} a => IntermediateField.adjoin ℚ (Set.range fun ij => (a ij.1).ofLp ij.2)
@[reducible] def Chal.EuclideanRamsey.Coeff : {s d : ℕ} → (Fin s → Chal.EuclideanRamsey.Space d) → Type :=
fun {s d} a => ↥(Chal.EuclideanRamsey.coordinateField a)
def Chal.EuclideanRamsey.coordinate : {s d : ℕ} →
  (a : Fin s → Chal.EuclideanRamsey.Space d) → Fin s → Fin d → Chal.EuclideanRamsey.Coeff a :=
fun {s d} a i j => ⟨(a i).ofLp j, ⋯⟩
