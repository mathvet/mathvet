  4 ≤ n →
    (∀ (v : Fin n), d v ≤ n - 1) →
      Chal.Problem315.Graphical n d →
        (∃ (h : ∃ t, Chal.Problem315.MixedAt n d t), Chal.Problem315.mixingTime n d h ≤ 2 * n ^ 8) ∧
          (Chal.Problem315.stateCount n d = 1 →
              Chal.Problem315.MixedAt n d 0 ∧

def Chal.Problem315.graphDegree : {n : ℕ} → SimpleGraph (Fin n) → Fin n → ℕ :=
fun {n} G v => {w | G.Adj v w}.card
def Chal.Problem315.Realizes : (n : ℕ) → (Fin n → ℕ) → SimpleGraph (Fin n) → Prop :=
fun n d G => ∀ (v : Fin n), Chal.Problem315.graphDegree G v = d v

def Chal.Problem315.Graphical : (n : ℕ) → (Fin n → ℕ) → Prop :=

@[reducible] def Chal.Problem315.FourSet : ℕ → Type :=

def Chal.Problem315.IsPerfectMatchingOn : {n : ℕ} → Finset (Fin n) → SimpleGraph (Fin n) → Prop :=
fun {n} S M => ∀ (v : Fin n), (v ∈ S → ∃! w, M.Adj v w) ∧ (v ∉ S → ∀ (w : Fin n), ¬M.Adj v w)
@[reducible] def Chal.Problem315.PerfectMatching : {n : ℕ} → Chal.Problem315.FourSet n → Type :=

def Chal.Problem315.addedMatching : {n : ℕ} → Chal.Problem315.SwitchProposal n → SimpleGraph (Fin n) :=

def Chal.Problem315.validSwitch : {n : ℕ} → SimpleGraph (Fin n) → Chal.Problem315.SwitchProposal n → Prop :=
fun {n} G p =>
  (∀ ⦃u v : Fin n⦄, (Chal.Problem315.removedMatching p).Adj u v → G.Adj u v) ∧
    ∀ ⦃u v : Fin n⦄, (Chal.Problem315.addedMatching p).Adj u v → ¬G.Adj u v
def Chal.Problem315.replaceEdges : {n : ℕ} →

fun {n} G M N => { Adj := fun u v => G.Adj u v ∧ ¬M.Adj u v ∨ N.Adj u v, symm := ⋯, loopless := ⋯ }
def Chal.Problem315.proposalResult : {n : ℕ} →

fun {n} G p =>
  if Chal.Problem315.validSwitch G p then
