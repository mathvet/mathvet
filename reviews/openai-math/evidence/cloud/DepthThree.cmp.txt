Chal.DepthThreeLowerBound.exists_polynomial_time_language_depth_three_lower_bound : ∃ L M C a,
  0 < C ∧
    0 < a ∧
      (∀ (w : List Bool), Chal.DepthThreeLowerBound.MultiTapeHaltsIn M w (L w) (C * (w.length + 1) ^ a)) ∧
        ∀ (A : ℝ),
          0 < A →
            ∃ N,
              ∀ (n : ℕ),
                N ≤ n →
                  ∀ (D : Chal.DepthThreeLowerBound.Circuit3 (Fin n)),
                    (D.Computes fun x => L (List.ofFn x)) → 2 ^ (A * √↑n) < ↑D.gateCount
