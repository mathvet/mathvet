def Chal.QAC.Gate.support : {N : ℕ} → Chal.QAC.Gate N → Finset (Fin N) :=
fun {N} x =>
  match x with
  | Chal.QAC.Gate.local q u => {q}
  | Chal.QAC.Gate.toffoli c t hne => insert t c
def Chal.QAC.Gate.toffoliWord : {N : ℕ} → Finset (Fin N) → Fin N → Chal.QAC.Word N → Chal.QAC.Word N :=
fun {N} c t x => if ∀ i ∈ c, x i = 1 then Function.update x t (1 - x t) else x
def Chal.QAC.Gate.matrix : {N : ℕ} → Chal.QAC.Gate N → Chal.QAC.Operator N :=
fun {N} x =>
  match x with
  | Chal.QAC.Gate.local q u => Chal.QAC.tensorMatrix fun i => if i = q then ↑u else 1
  | Chal.QAC.Gate.toffoli c t hne => fun x y => if x = Chal.QAC.Gate.toffoliWord c t y then 1 else 0

def Chal.QAC.physicalLayerMatrix : {N : ℕ} → Chal.QAC.PhysicalLayer N → Chal.QAC.Operator N :=
fun {N} L => (List.map Chal.QAC.Gate.matrix L.gates).prod
def Chal.QAC.physicalCircuitMatrix : {N : ℕ} → List (Chal.QAC.PhysicalLayer N) → Chal.QAC.Operator N :=
fun {N} layers => (List.map Chal.QAC.physicalLayerMatrix layers).reverse.prod
def Chal.QAC.inputWord : {n N : ℕ} → Chal.QAC.Word n → Chal.QAC.Word N :=
fun {n N} x i => if h : ↑i < n then x ⟨↑i, h⟩ else 0
def Chal.QAC.parity : {n : ℕ} → Chal.QAC.Word n → Chal.QAC.Bit :=
fun {n} x => ∑ i, x i
def Chal.QAC.successProbability : {n N : ℕ} → Chal.QAC.Operator N → Fin N → Chal.QAC.Word n → ℝ :=
fun {n N} W out x => ∑ y, if y out = Chal.QAC.parity x then Complex.normSq (W y (Chal.QAC.inputWord x)) else 0
def Chal.QAC.ParityStatement : Prop :=
∀ (d : ℕ) (c ε : ℝ),
  1 ≤ c →
    0 < ε →
      ε ≤ 1 / 2 →
        ∃ n₀,
          ∀ n ≥ n₀,
            ∀ (N : ℕ),
              n ≤ N →
                ↑N ≤ ↑n ^ c →
                  ∀ (layers : List (Chal.QAC.PhysicalLayer N)),
                    layers.length ≤ d →
                      ∀ (out : Fin N),
                        ¬∀ (x : Chal.QAC.Word n),
                            1 / 2 + ε ≤ Chal.QAC.successProbability (Chal.QAC.physicalCircuitMatrix layers) out x
