Chal.PettyProjection.petty_projection_volume : ∀ (n : ℕ),
  4 ≤ n →
    ∀ (K : Set (Chal.PettyProjection.Space n)),
      Chal.PettyProjection.IsConvexBody K →
        Chal.PettyProjection.pettyConstant n ≤ Chal.PettyProjection.projectionRatio K ∧
          (Chal.PettyProjection.projectionRatio K = Chal.PettyProjection.pettyConstant n ↔
            Chal.PettyProjection.IsEllipsoid K)
@[reducible] def Chal.PettyProjection.Space : ℕ → Type :=
fun n => EuclideanSpace ℝ (Fin n)
def Chal.PettyProjection.IsConvexBody : {n : ℕ} → Set (Chal.PettyProjection.Space n) → Prop :=
fun {n} K => IsCompact K ∧ Convex ℝ K ∧ (interior K).Nonempty
def Chal.PettyProjection.unitBall : (n : ℕ) → Set (Chal.PettyProjection.Space n) :=
fun n => Metric.closedBall 0 1
def Chal.PettyProjection.kappa : ℕ → ℝ :=
fun n => MeasureTheory.volume.real (Chal.PettyProjection.unitBall n)
def Chal.PettyProjection.perpendicular : {n : ℕ} →
  Chal.PettyProjection.Space n → Submodule ℝ (Chal.PettyProjection.Space n) :=
fun {n} u => (ℝ ∙ u)ᗮ
def Chal.PettyProjection.shadowVolume : {n : ℕ} →
  Set (Chal.PettyProjection.Space n) → Chal.PettyProjection.Space n → ℝ :=
fun {n} K u => MeasureTheory.volume.real (⇑(Chal.PettyProjection.perpendicular u).orthogonalProjectionOnto '' K)
def Chal.PettyProjection.projectionBody : {n : ℕ} →
  Set (Chal.PettyProjection.Space n) → Set (Chal.PettyProjection.Space n) :=
fun {n} K => {x | ∀ (u : Chal.PettyProjection.Space n), ‖u‖ = 1 → inner ℝ u x ≤ Chal.PettyProjection.shadowVolume K u}
def Chal.PettyProjection.projectionRatio : {n : ℕ} → Set (Chal.PettyProjection.Space n) → ℝ :=
fun {n} K => MeasureTheory.volume.real (Chal.PettyProjection.projectionBody K) / MeasureTheory.volume.real K ^ (n - 1)
def Chal.PettyProjection.pettyConstant : ℕ → ℝ :=
fun n => Chal.PettyProjection.kappa (n - 1) ^ n * Chal.PettyProjection.kappa n ^ (2 - ↑n)
def Chal.PettyProjection.IsEllipsoid : {n : ℕ} → Set (Chal.PettyProjection.Space n) → Prop :=
fun {n} K => ∃ a T, K = (fun x => a + T x) '' Chal.PettyProjection.unitBall n
