OAI.StableCoordinate.principal_package : OAI.StableCoordinate.PrincipalPackage
'OAI.StableCoordinate.principal_package' depends on axioms: [propext, Classical.choice, Quot.sound]
@[reducible] def OAI.StableCoordinate.R : ℕ → Type :=
fun n => MvPolynomial (Fin n) ℂ
def OAI.StableCoordinate.Q : OAI.StableCoordinate.R 4 :=
MvPolynomial.X 1 ^ 2 - MvPolynomial.X 3 ^ 2 + MvPolynomial.X 0 * MvPolynomial.X 2
def OAI.StableCoordinate.f : OAI.StableCoordinate.R 4 :=
MvPolynomial.X 0 -
  2 * OAI.StableCoordinate.Q *
    (OAI.StableCoordinate.Q * (MvPolynomial.X 1 + MvPolynomial.X 3) + MvPolynomial.X 0 * MvPolynomial.X 3)
def OAI.StableCoordinate.IsCoordinate : {n : ℕ} → 0 < n → OAI.StableCoordinate.R n → Prop :=
fun {n} hn g => ∃ φ, φ g = MvPolynomial.X ⟨0, hn⟩
def OAI.StableCoordinate.stabilize : OAI.StableCoordinate.R 4 →ₐ[ℂ] OAI.StableCoordinate.R 5 :=
MvPolynomial.rename Fin.castSucc
def OAI.StableCoordinate.IsOneStableCoordinate : OAI.StableCoordinate.R 4 → Prop :=
fun g => ∃ φ, φ (OAI.StableCoordinate.stabilize g) = MvPolynomial.X 0
def OAI.StableCoordinate.fiberIdeal : ℂ → Ideal (OAI.StableCoordinate.R 4) :=
fun t => Ideal.span {OAI.StableCoordinate.f - MvPolynomial.C t}
@[reducible] def OAI.StableCoordinate.FiberRing : ℂ → Type :=
fun t => OAI.StableCoordinate.R 4 ⧸ OAI.StableCoordinate.fiberIdeal t
def OAI.StableCoordinate.IsAffineThreeSpaceFiber : ℂ → Prop :=
fun t => Nonempty (OAI.StableCoordinate.FiberRing t ≃ₐ[ℂ] OAI.StableCoordinate.R 3)
def OAI.StableCoordinate.IsRectifiableFiber : ℂ → Prop :=
fun t => ∃ φ, Ideal.map φ.toRingEquiv.toRingHom (OAI.StableCoordinate.fiberIdeal t) = Ideal.span {MvPolynomial.X 0}
def OAI.StableCoordinate.PolynomialPackage : Prop :=
MvPolynomial.totalDegree OAI.StableCoordinate.f = 5 ∧
  OAI.StableCoordinate.IsOneStableCoordinate OAI.StableCoordinate.f ∧
    ¬OAI.StableCoordinate.IsCoordinate OAI.StableCoordinate.PolynomialPackage._proof_1 OAI.StableCoordinate.f
def OAI.StableCoordinate.FiberPackage : Prop :=
∀ (t : ℂ), OAI.StableCoordinate.IsAffineThreeSpaceFiber t ∧ ¬OAI.StableCoordinate.IsRectifiableFiber t
def OAI.StableCoordinate.PrincipalPackage : Prop :=
OAI.StableCoordinate.PolynomialPackage ∧ OAI.StableCoordinate.FiberPackage
