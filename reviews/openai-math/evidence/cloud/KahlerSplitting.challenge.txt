def OAI.UniversalCoverSplitting.CompatibleProduct.{u, v} : {n r₁ r₂ : ℕ} →
  {X : OAI.UniversalCoverSplitting.ComplexManifold n} →
    {Z : OAI.UniversalCoverSplitting.ComplexManifold n} →
      OAI.UniversalCoverSplitting.HolomorphicSplitting X r₁ r₂ →
        OAI.UniversalCoverSplitting.OrdinaryUniversalCover X Z →
          OAI.UniversalCoverSplitting.ComplexManifold r₁ → OAI.UniversalCoverSplitting.ComplexManifold r₂ → Prop :=
fun {n r₁ r₂} {X} {Z} S π Y₁ Y₂ =>
  ∃ Φ,
    MDiff ⇑Φ ∧
      MDiff ⇑Φ.symm ∧
        (∀ (z : Z.Carrier),
            Submodule.map (↑(mfderiv% ⇑Φ z))
                (Submodule.comap (↑(mfderiv% π.projection z)) (↑(S.projection (π.projection z))).range) =
              (↑(ContinuousLinearMap.snd ℂ (OAI.UniversalCoverSplitting.Model r₁)
                    (OAI.UniversalCoverSplitting.Model r₂))).ker) ∧
          ∀ (z : Z.Carrier),
            Submodule.map (↑(mfderiv% ⇑Φ z))
                (Submodule.comap (↑(mfderiv% π.projection z)) (↑(S.projection (π.projection z))).ker) =
              (↑(ContinuousLinearMap.fst ℂ (OAI.UniversalCoverSplitting.Model r₁)
                    (OAI.UniversalCoverSplitting.Model r₂))).ker
