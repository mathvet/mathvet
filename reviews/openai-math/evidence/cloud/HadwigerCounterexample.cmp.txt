def Chal.HadwigerMatching.IsTouchingMatching.{u} : {V : Type u} → SimpleGraph V → Finset (Finset V) → Prop :=
fun {V} G M =>
  (∀ e ∈ M, e.card = 2 ∧ G.IsClique ↑e) ∧
    (∀ e ∈ M, ∀ f ∈ M, e ≠ f → Disjoint e f) ∧ ∀ e ∈ M, ∀ f ∈ M, e ≠ f → ∃ v ∈ e, ∃ w ∈ f, G.Adj v w
def Chal.HadwigerMatching.connectedMatchingNumber.{u} : {V : Type u} → SimpleGraph V → ℕ :=
fun {V} G => sSup {n | ∃ M, Chal.HadwigerMatching.IsTouchingMatching G M ∧ M.card = n}
def Chal.LinearListHadwiger.HasCliqueMinor : {V : Type} → SimpleGraph V → ℕ → Prop :=
fun {V} G t =>
  ∃ B,
    (∀ (i : Fin t), (SimpleGraph.induce (B i) G).Connected) ∧
      (∀ (i j : Fin t), i ≠ j → Disjoint (B i) (B j)) ∧ ∀ (i j : Fin t), i ≠ j → ∃ v ∈ B i, ∃ w ∈ B j, G.Adj v w
def Chal.LinearListHadwiger.hadwigerNumber : {V : Type} → SimpleGraph V → ℕ :=
fun {V} G => sSup {t | Chal.LinearListHadwiger.HasCliqueMinor G t}
