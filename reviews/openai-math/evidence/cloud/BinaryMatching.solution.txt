def OAI.BinaryMatching.multiplicity : (G : OAI.BinaryMatching.Input) → OAI.BinaryMatching.Pair G.n → ℕ :=
fun G e =>
  match List.find? (fun r => decide (r.left = ↑(↑e).1) && decide (r.right = ↑(↑e).2)) G.records with
  | none => 0
  | some r => r.multiplicity
def OAI.BinaryMatching.count : OAI.BinaryMatching.Input → ℕ :=
fun G => ∑ M, ∏ e ∈ ↑M, OAI.BinaryMatching.multiplicity G e
def OAI.BinaryMatching.encodeNat : ℕ → List Bool :=
fun n => List.replicate n.bits.length false ++ true :: n.bits
def OAI.BinaryMatching.encodeRecord : OAI.BinaryMatching.Record → List Bool :=
fun r =>
  OAI.BinaryMatching.encodeNat r.left ++ OAI.BinaryMatching.encodeNat r.right ++
    OAI.BinaryMatching.encodeNat r.multiplicity
def OAI.BinaryMatching.encodeInput : OAI.BinaryMatching.Input → List Bool :=
fun G =>
  OAI.BinaryMatching.encodeNat G.n ++ OAI.BinaryMatching.encodeNat G.records.length ++
    List.flatMap OAI.BinaryMatching.encodeRecord G.records
