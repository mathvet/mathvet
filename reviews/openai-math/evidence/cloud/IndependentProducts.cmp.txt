def Chal.IndependentProducts.averagedMidpointModulus.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Seminorm ℝ X → ℝ → ℝ :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N t =>
  sInf
    {r |
      ∃ x,
        N x = 1 ∧
          r =
            sSup
              {a |
                ∃ F ∈ Chal.IndependentProducts.CofiniteSubspaces X,
                  a = sInf {b | ∃ y ∈ F, 1 ≤ N y ∧ b = (N (x + t • y) + N (x - t • y)) / 2 - 1}}}

def Chal.IndependentProducts.IsEquivalentNorm.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Seminorm ℝ X → Prop :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N => ∃ a b, 0 < a ∧ a ≤ b ∧ ∀ (x : X), a * ‖x‖ ≤ N x ∧ N x ≤ b * ‖x‖
def Chal.IndependentProducts.IsAUC.{uX} : (X : Type uX) →

fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N =>
  ∀ (t : ℝ), 0 < t → 0 < Chal.IndependentProducts.oneSidedModulus X N t
