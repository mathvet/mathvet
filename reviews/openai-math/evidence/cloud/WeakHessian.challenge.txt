fun K N t θ =>
  if t ≤ 0 then 0
  else
    if 1 ≤ t then 1
    else
      if θ = 0 then ENNReal.ofReal t
      else
        if 0 < K then
          if K * θ ^ 2 < N * Real.pi ^ 2 then ENNReal.ofReal (Real.sin (t * θ * √(K / N)) / Real.sin (θ * √(K / N)))
          else ⊤
        else
