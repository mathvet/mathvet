def Chal.TriangularBilliards.Triangle.side : Chal.TriangularBilliards.Triangle → Fin 3 → Set ℂ :=
fun Q i => openSegment ℝ (Q.vertex i) (Q.vertex (i + 1))
def Chal.TriangularBilliards.Triangle.tangent : Chal.TriangularBilliards.Triangle → Fin 3 → ℂ :=
fun Q i => Q.vertex (i + 1) - Q.vertex i
def Chal.TriangularBilliards.Triangle.angle : Chal.TriangularBilliards.Triangle → Fin 3 → ℝ :=
fun Q i => InnerProductGeometry.angle (Q.vertex (i + 1) - Q.vertex i) (Q.vertex (i + 2) - Q.vertex i)

fun Q => ∃ i, Irrational (Q.angle i / Real.pi)

def Chal.TriangularBilliards.FlightChain.at : {Q : Chal.TriangularBilliards.Triangle} →
  {z : Chal.TriangularBilliards.Phase} →
    Chal.TriangularBilliards.FlightChain Q z → ℝ → Chal.TriangularBilliards.Phase :=
fun {Q} {z} c t =>
  if h : ∃ n, c.time n ≤ t ∧ t < c.time (n + 1) then
    have n := Classical.choose h;
    (c.point n + (t - c.time n) • ↑(c.direction n), c.direction n)
  else z
def Chal.TriangularBilliards.billiardFlow : Chal.TriangularBilliards.Triangle →
  ℝ → Chal.TriangularBilliards.Phase → Chal.TriangularBilliards.Phase :=
fun Q t z => if h : Nonempty (Chal.TriangularBilliards.FlightChain Q z) then (Classical.choice h).at t else z
def Chal.TriangularBilliards.angularMeasure : MeasureTheory.Measure Circle :=

  ((ENNReal.ofReal (2 * Real.pi))⁻¹ • MeasureTheory.volume.restrict (Set.Ico 0 (2 * Real.pi)))

def Chal.TriangularBilliards.MainConclusion : Chal.TriangularBilliards.Triangle → Prop :=
fun Q =>
  (Chal.TriangularBilliards.phaseMeasure Q) Set.univ = 1 ∧
    (∀ᵐ (z : Chal.TriangularBilliards.Phase) ∂Chal.TriangularBilliards.phaseMeasure Q,
        Nonempty (Chal.TriangularBilliards.FlightChain Q z)) ∧
      (∀ (z : Chal.TriangularBilliards.Phase) (c d : Chal.TriangularBilliards.FlightChain Q z), c = d) ∧
        (∀ (t : ℝ),
            MeasureTheory.MeasurePreserving (Chal.TriangularBilliards.billiardFlow Q t)
              (Chal.TriangularBilliards.phaseMeasure Q) (Chal.TriangularBilliards.phaseMeasure Q)) ∧
          (∀ (s t : ℝ),
              Chal.TriangularBilliards.billiardFlow Q (s + t) =ᵐ[Chal.TriangularBilliards.phaseMeasure Q]
                Chal.TriangularBilliards.billiardFlow Q s ∘ Chal.TriangularBilliards.billiardFlow Q t) ∧
            ∀ (A : Set Chal.TriangularBilliards.Phase),
              MeasurableSet A →
                (∀ (t : ℝ),
                    (Chal.TriangularBilliards.phaseMeasure Q)
                        (symmDiff (Chal.TriangularBilliards.billiardFlow Q t ⁻¹' A) A) =
                      0) →
                  (Chal.TriangularBilliards.phaseMeasure Q) A = 0 ∨ (Chal.TriangularBilliards.phaseMeasure Q) A = 1
