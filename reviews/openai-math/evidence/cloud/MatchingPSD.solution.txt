OAI.PerfectMatchingPSD.main : OAI.PerfectMatchingPSD.MainClaim
'OAI.PerfectMatchingPSD.main' depends on axioms: [propext, Classical.choice, Quot.sound]
@[reducible] def OAI.PerfectMatchingPSD.Edge : ℕ → Type :=
fun n => { p // p.1 < p.2 }
def OAI.PerfectMatchingPSD.IsPerfectMatching : {n : ℕ} → Finset (OAI.PerfectMatchingPSD.Edge n) → Prop :=
fun {n} M => ∀ (v : Fin n), {e ∈ M | (↑e).1 = v ∨ (↑e).2 = v}.card = 1
@[reducible] def OAI.PerfectMatchingPSD.PerfectMatching : ℕ → Type :=
fun n => { M // OAI.PerfectMatchingPSD.IsPerfectMatching M }
@[reducible] def OAI.PerfectMatchingPSD.OddCut : ℕ → Type :=
fun n => { U // Odd U.card }
@[reducible] def OAI.PerfectMatchingPSD.Row : ℕ → Type :=
fun n => OAI.PerfectMatchingPSD.Edge n ⊕ OAI.PerfectMatchingPSD.OddCut n
def OAI.PerfectMatchingPSD.Crosses : {n : ℕ} → Finset (Fin n) → OAI.PerfectMatchingPSD.Edge n → Prop :=
fun {n} U e => (↑e).1 ∈ U ∧ (↑e).2 ∉ U ∨ (↑e).1 ∉ U ∧ (↑e).2 ∈ U
def OAI.PerfectMatchingPSD.crossingCount : {n : ℕ} → Finset (Fin n) → OAI.PerfectMatchingPSD.PerfectMatching n → ℕ :=
fun {n} U M => (Finset.filter (OAI.PerfectMatchingPSD.Crosses U) ↑M).card
def OAI.PerfectMatchingPSD.slack : {n : ℕ} →
  OAI.PerfectMatchingPSD.Row n → OAI.PerfectMatchingPSD.PerfectMatching n → ℝ :=
fun {n} i M =>
  match i with
  | Sum.inl e => if e ∈ ↑M then 1 else 0
  | Sum.inr U => ↑(OAI.PerfectMatchingPSD.crossingCount (↑U) M) - 1
def OAI.PerfectMatchingPSD.HasFactorization : ℕ → ℕ → Prop :=
fun n r =>
  ∃ A B,
    (∀ (i : OAI.PerfectMatchingPSD.Row n), (A i).PosSemidef) ∧
      (∀ (M : OAI.PerfectMatchingPSD.PerfectMatching n), (B M).PosSemidef) ∧
        ∀ (i : OAI.PerfectMatchingPSD.Row n) (M : OAI.PerfectMatchingPSD.PerfectMatching n),
          OAI.PerfectMatchingPSD.slack i M = (A i * B M).trace
def OAI.PerfectMatchingPSD.psdRank : ℕ → ℕ :=
fun n => sInf {r | 0 < r ∧ OAI.PerfectMatchingPSD.HasFactorization n r}
def OAI.PerfectMatchingPSD.MainClaim : Prop :=
∀ (C : ℝ), 0 < C → ∃ n₀, 4 ≤ n₀ ∧ ∀ (n : ℕ), n₀ ≤ n → Even n → ↑n ^ C < ↑(OAI.PerfectMatchingPSD.psdRank n)
