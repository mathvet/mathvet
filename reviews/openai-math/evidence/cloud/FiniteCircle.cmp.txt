def Chal.FiniteCircle.distortion.{u, v} : {α : Type u} →
  {X : Type v} → (X → List α) → (X → ↥Chal.FiniteCircle.RealL1) → ℝ :=
fun {α} {X} word f =>
  sSup {r | ∃ x y, x ≠ y ∧ r = ‖f x - f y‖ / ↑(Chal.FiniteCircle.editDistance (word x) (word y))} *
    sSup {r | ∃ x y, x ≠ y ∧ r = ↑(Chal.FiniteCircle.editDistance (word x) (word y)) / ‖f x - f y‖}

def Chal.FiniteCircle.OffsetLocal.{u} : {α : Type u} → {w : ℕ} → ℝ → (α → Fin w → Bool) → Prop :=
fun {α} {w} δ c =>
  ∀ (a b : α),
    a ≠ b →
      ∀ (s t : ℕ),
        s + w / 8 ≤ w →
          t + w / 8 ≤ w →
            ↑(Chal.FiniteCircle.lcs (Chal.FiniteCircle.interval c a s (w / 8))

              (1 - δ) * ↑(w / 8)
def Chal.FiniteCircle.indexSpan : {w : ℕ} → List (Fin w) → ℕ :=
fun {w} is =>
  match is with
  | [] => 0
  | i :: js => ↑(js.getLastD i) + 1 - ↑i
def Chal.FiniteCircle.AnchorLocal.{u} : {α : Type u} → {w : ℕ} → ℝ → (α → Fin w → Bool) → Prop :=
fun {α} {w} ε c =>
  ∀ (a b : α) (is js : List (Fin w)),
    List.Pairwise (fun i j => ↑i < ↑j) is →
      List.Pairwise (fun i j => ↑i < ↑j) js →
        is.length = js.length →
          ↑w / 4 ≤ ↑is.length →
            ↑(Chal.FiniteCircle.indexSpan is - is.length) ≤ ε * ↑w →
              ↑(Chal.FiniteCircle.indexSpan js - js.length) ≤ ε * ↑w →
                (∀ ij ∈ is.zip js, c a ij.1 = c b ij.2) → a = b ∧ ∃ i, (i, i) ∈ is.zip js
def Chal.FiniteCircle.SubstitutionBounds.{u} : {α : Type u} →
  [Fintype α] → {w : ℕ} → ℝ → ℝ → (α → Fin w → Bool) → Prop :=
fun {α} [Fintype α] {w} a C c =>
  0 < w ∧
    ↑w ≤ C * Real.log (2 * ↑(Fintype.card α)) ∧
      (∀ (L : ℕ) (x y : List α),
          x.length = L →
            y.length = L →
              a * ↑w * ↑(Chal.FiniteCircle.deficit x y) ≤
                  ↑(Chal.FiniteCircle.deficit (Chal.FiniteCircle.encode c x) (Chal.FiniteCircle.encode c y)) ∧
                Chal.FiniteCircle.deficit (Chal.FiniteCircle.encode c x) (Chal.FiniteCircle.encode c y) ≤
                  w * Chal.FiniteCircle.deficit x y) ∧
        ∀ (x y : List α),
          Chal.FiniteCircle.editDistance (Chal.FiniteCircle.encode c x) (Chal.FiniteCircle.encode c y) ≤
            w * Chal.FiniteCircle.editDistance x y
def Chal.FiniteCircle.TwoCodes : ℝ → ℝ → ℝ → ℝ → Prop :=

def Chal.FiniteCircle.period : ℕ → (ℕ → ℕ → ℕ) → ℕ → ℕ :=
fun h p l => ∏ k, p l ↑k
def Chal.FiniteCircle.circleSize : ℕ → (ℕ → ℕ → ℕ) → ℕ :=
fun h p => 2 * ∏ l, Chal.FiniteCircle.period h p (↑l + 1)
def Chal.FiniteCircle.MaskPrimes : ℕ → (ℕ → ℕ → ℕ) → Prop :=
fun h p =>
  (∀ (l k : Fin h), Nat.Prime (p (↑l + 1) ↑k)) ∧
    (Function.Injective fun lk => p (↑lk.1 + 1) ↑lk.2) ∧
      ∀ (l k : Fin h), h ^ 3 ≤ p (↑l + 1) ↑k ∧ p (↑l + 1) ↑k ≤ 2 * h ^ 3
def Chal.FiniteCircle.ExcludedPrimes : ℕ → (ℕ → ℕ → ℕ) → Prop :=
fun m p =>
  (∀ (l k : Fin m), Nat.Prime (p (↑l + 1) ↑k)) ∧
    (Function.Injective fun lk => p (↑lk.1 + 1) ↑lk.2) ∧ ∀ (l k : Fin m), m ^ 3 < p (↑l + 1) ↑k ∧ p (↑l + 1) ↑k ≤ m ^ 5

def Chal.FiniteCircle.MaskWitness : ℝ → ℝ → ℝ → ℝ → ℕ → Prop :=
fun a C δ lower d =>
  ∃ h,
    2 ≤ h ∧
      ∃ p,
        Chal.FiniteCircle.MaskPrimes h p ∧
          ∃ w c,
            Chal.FiniteCircle.SubstitutionBounds a C c ∧
              Chal.FiniteCircle.OffsetLocal δ c ∧
                ∃ n,
                  0 < n ∧
                    n ≤ d ∧
                      (∀ (v : Chal.FiniteCircle.MaskState h p (Chal.FiniteCircle.circleSize h p) h),
                          (Chal.FiniteCircle.encode c
                                (Chal.FiniteCircle.maskWord h p (h ^ (2 * h)) (Chal.FiniteCircle.circleSize h p) h
                                  v)).length =
                            n) ∧
                        Real.exp (lower * Chal.FiniteCircle.exponentScale d) ≤
                          Chal.FiniteCircle.setDistortion
                            (Set.range fun v =>
                              Chal.FiniteCircle.encode c
                                (Chal.FiniteCircle.maskWord h p (h ^ (2 * h)) (Chal.FiniteCircle.circleSize h p) h v))
def Chal.FiniteCircle.ExcludedWitness : ℝ → ℝ → ℝ → ℝ → ℕ → Prop :=
fun a C ε lower d =>
  ∃ m,
    2 ≤ m ∧
      ∃ p,
        Chal.FiniteCircle.ExcludedPrimes m p ∧
          ∃ w c,
            Chal.FiniteCircle.SubstitutionBounds a C c ∧
              Chal.FiniteCircle.AnchorLocal ε c ∧
                ∃ n,
                  0 < n ∧
                    n ≤ d ∧
                      (∀ (v : Chal.FiniteCircle.ExcludedState m (Chal.FiniteCircle.circleSize m p) m),
                          (Chal.FiniteCircle.encode c
                                (Chal.FiniteCircle.excludedWord m p (m ^ (2 * m)) (Chal.FiniteCircle.circleSize m p) m
                                  v)).length =
                            n) ∧
                        Real.exp (lower * Chal.FiniteCircle.exponentScale d) ≤
                          Chal.FiniteCircle.setDistortion
                            (Set.range fun v =>
                              Chal.FiniteCircle.encode c
                                (Chal.FiniteCircle.excludedWord m p (m ^ (2 * m)) (Chal.FiniteCircle.circleSize m p) m
                                  v))
def Chal.FiniteCircle.UniformUpper : ℝ → ℕ → Prop :=
fun C d₀ =>
  ∀ (α : Type) [inst : Fintype α],
    2 ≤ Fintype.card α →
      ∀ (d : ℕ),
        d₀ ≤ d →
          ∃ M F,
            ∀ (x y : Chal.FiniteCircle.Words α d),
              ↑(Chal.FiniteCircle.editDistance ↑x ↑y) ≤ ∑ i, |F x i - F y i| ∧
                ∑ i, |F x i - F y i| ≤
                  Real.exp (C * Chal.FiniteCircle.exponentScale d) * ↑(Chal.FiniteCircle.editDistance ↑x ↑y)
def Chal.FiniteCircle.FullMain : Prop :=
∃ c C a Ccode δ ε d₀,
  0 < c ∧
    0 < C ∧
      0 < a ∧
        0 < Ccode ∧
          0 < δ ∧
            0 < ε ∧
              Chal.FiniteCircle.TwoCodes a Ccode δ ε ∧
                Chal.FiniteCircle.UniformUpper C d₀ ∧
                  (∀ (α : Type) [inst : Fintype α],
                      2 ≤ Fintype.card α →
                        ∀ (d : ℕ),
                          d₀ ≤ d →
                            Real.exp (c * Chal.FiniteCircle.exponentScale d) ≤ Chal.FiniteCircle.E α d ∧
                              Chal.FiniteCircle.E α d ≤ Real.exp (C * Chal.FiniteCircle.exponentScale d)) ∧
                    ∀ (d : ℕ),
                      d₀ ≤ d →
                        Chal.FiniteCircle.MaskWitness a Ccode δ c d ∧
                          Chal.FiniteCircle.ExcludedWitness a Ccode ε c d ∧
                            Real.exp (c * Chal.FiniteCircle.exponentScale d) ≤ Chal.FiniteCircle.alphabetSupremum d ∧
                              Chal.FiniteCircle.alphabetSupremum d ≤ Real.exp (C * Chal.FiniteCircle.exponentScale d)
