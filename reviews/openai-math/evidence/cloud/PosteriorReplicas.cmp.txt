          let k := 2 * (d / 16);
          have t := k + 1;
          let ℓ := 4 * k;
          ∀ (p : MeasureTheory.Measure (Chal.PosteriorReplicas.Vec d)) [inst : MeasureTheory.IsProbabilityMeasure p]
            (f : Chal.PosteriorReplicas.Vec d → ℝ) (L : ℝ),

  if LinearIndependent ℝ A then
    match base with
    | Chal.PosteriorReplicas.BaseLaw.gaussian =>
      ENNReal.ofReal
        ((2 * Real.pi) ^ (-↑k / 2) * Real.exp (-‖Chal.PosteriorReplicas.liftLabel A y‖ ^ 2 / 2) /
          √(Chal.PosteriorReplicas.rowGram A).det)
    | Chal.PosteriorReplicas.BaseLaw.spherical =>
      if ‖Chal.PosteriorReplicas.liftLabel A y‖ < 1 then
        ENNReal.ofReal
          (Real.Gamma (↑d / 2) / (Real.pi ^ (↑k / 2) * Real.Gamma ((↑d - ↑k) / 2)) *
              (1 - ‖Chal.PosteriorReplicas.liftLabel A y‖ ^ 2) ^ ((↑d - ↑k - 2) / 2) /

  else 0
