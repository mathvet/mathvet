Chal.PermanentBorder.smooth_initial_form_lower_bound : ∀ (d r : ℕ) (g : Chal.PermanentBorder.Poly (Fin d))
  (a : Fin d → ℂ),
  2 ≤ d →
    2 ≤ r →
      (MvPolynomial.homogeneousComponent r)
            ((Chal.PermanentBorder.affineSubstitution a fun i j => if i = j then 1 else 0) g) ≠
          0 →
        (∀ j < r,
            (MvPolynomial.homogeneousComponent j)
                ((Chal.PermanentBorder.affineSubstitution a fun i j => if i = j then 1 else 0) g) =
              0) →
          (∀ (y : Fin d → ℂ),
              (MvPolynomial.eval y)
                    ((MvPolynomial.homogeneousComponent r)
                      ((Chal.PermanentBorder.affineSubstitution a fun i j => if i = j then 1 else 0) g)) =
                  0 →
                (∀ (i : Fin d),
                    (MvPolynomial.eval y)
                        ((MvPolynomial.pderiv i)
                          ((MvPolynomial.homogeneousComponent r)
                            ((Chal.PermanentBorder.affineSubstitution a fun i j => if i = j then 1 else 0) g))) =
                      0) →
                  y = 0) →
            ∀ (n : ℕ),
              (Chal.PermanentBorder.HasBorderDeterminant n g → ↑(r - 1) * ↑(d - 1) / (4 * Real.exp 1) ≤ ↑n) ∧
                (Chal.PermanentBorder.HasExactDeterminant n g → ↑(r - 1) * ↑(d - 1) / (4 * Real.exp 1) ≤ ↑n)

def Chal.PermanentBorder.affineMatrixPolynomial.{u_1} : {σ : Type u_1} →
  [Fintype σ] → (n : ℕ) → Matrix (Fin n) (Fin n) ℂ → (σ → Matrix (Fin n) (Fin n) ℂ) → Chal.PermanentBorder.Poly σ :=

  [Fintype σ] → ℕ → Chal.PermanentBorder.Poly σ → Prop :=
fun {σ} [Fintype σ] n p =>
  0 < n ∧
    ∃ A₀ A,

          (nhds (p.coeff d))
