def OAI.MeanPayoff.step : (M : OAI.MeanPayoff.Machine) →
  OAI.MeanPayoff.Configuration M → Bool → OAI.MeanPayoff.Configuration M :=
fun M c bit =>
  match c.control with
  | none => c
  | some s =>
    match M.transition s (fun k => c.tape k (c.head k)) bit with
    | none => { control := none, head := c.head, tape := c.tape }
    | some a =>
      { control := some a.next, head := fun k => c.head k + (a.shift k).offset,
        tape := fun k => Function.update (c.tape k) (c.head k) (a.write k) }
