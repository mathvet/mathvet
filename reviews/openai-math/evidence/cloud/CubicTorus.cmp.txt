Chal.CubicTorus.unit_cubic_isoperimetric : ∀ V ∈ Set.Ioo 0 1,
  (∃ E, Chal.CubicTorus.IsMinimizerAt V E) ∧
    (∀ (E : Set Chal.CubicTorus.Torus),
        Chal.CubicTorus.IsMinimizerAt V E →
          Chal.CubicTorus.perimeter E = ENNReal.ofReal (Chal.CubicTorus.candidateProfile V)) ∧
      ∀ (E : Set Chal.CubicTorus.Torus), Chal.CubicTorus.IsMinimizerAt V E ↔ Chal.CubicTorus.StandardRegionAt V E
@[reducible] def Chal.CubicTorus.Cover : Type :=
EuclideanSpace ℝ (Fin 3)
@[reducible] def Chal.CubicTorus.Torus : Type :=
PiLp 2 fun x => AddCircle 1
def Chal.CubicTorus.quotientMap : Chal.CubicTorus.Cover → Chal.CubicTorus.Torus :=
fun x => WithLp.toLp 2 fun i => ↑(x.ofLp i)
def Chal.CubicTorus.fundamentalCube : Set Chal.CubicTorus.Cover :=
{x | ∀ (i : Fin 3), 0 ≤ x.ofLp i ∧ x.ofLp i < 1}
def Chal.CubicTorus.torusVolume : MeasureTheory.Measure Chal.CubicTorus.Torus :=
MeasureTheory.Measure.map Chal.CubicTorus.quotientMap (MeasureTheory.volume.restrict Chal.CubicTorus.fundamentalCube)
def Chal.CubicTorus.latticeVector : (Fin 3 → ℤ) → Chal.CubicTorus.Cover :=
fun n => WithLp.toLp 2 fun i => ↑(n i)
def Chal.CubicTorus.AdmissibleField : (Chal.CubicTorus.Cover → Chal.CubicTorus.Cover) → Prop :=
fun F =>
  ContDiff ℝ (↑⊤) F ∧
    (∀ (n : Fin 3 → ℤ) (x : Chal.CubicTorus.Cover), F (x + Chal.CubicTorus.latticeVector n) = F x) ∧
      ∀ (x : Chal.CubicTorus.Cover), ‖F x‖ ≤ 1
def Chal.CubicTorus.divergence : (Chal.CubicTorus.Cover → Chal.CubicTorus.Cover) → Chal.CubicTorus.Cover → ℝ :=
fun F x => ∑ i, ((fderiv ℝ F x) (EuclideanSpace.single i 1)).ofLp i
def Chal.CubicTorus.perimeter : Set Chal.CubicTorus.Torus → ENNReal :=
fun E =>
  ⨆ F,
    ⨆ (_ : Chal.CubicTorus.AdmissibleField F),
      ENNReal.ofReal
        (∫ (x : Chal.CubicTorus.Cover) in Chal.CubicTorus.fundamentalCube ∩ Chal.CubicTorus.quotientMap ⁻¹' E,
          Chal.CubicTorus.divergence F x)
def Chal.CubicTorus.FinitePerimeter : Set Chal.CubicTorus.Torus → Prop :=
fun E => MeasureTheory.NullMeasurableSet E Chal.CubicTorus.torusVolume ∧ Chal.CubicTorus.perimeter E < ⊤
def Chal.CubicTorus.IsMinimizerAt : ℝ → Set Chal.CubicTorus.Torus → Prop :=
fun V E =>
  Chal.CubicTorus.FinitePerimeter E ∧
    Chal.CubicTorus.torusVolume E = ENNReal.ofReal V ∧
      ∀ (F : Set Chal.CubicTorus.Torus),
        Chal.CubicTorus.FinitePerimeter F →
          Chal.CubicTorus.torusVolume F = ENNReal.ofReal V → Chal.CubicTorus.perimeter E ≤ Chal.CubicTorus.perimeter F
def Chal.CubicTorus.candidateProfile : ℝ → ℝ :=
fun V =>
  have v := min V (1 - V);
  min ((36 * Real.pi) ^ (1 / 3) * v ^ (2 / 3)) (min (2 * √(Real.pi * v)) 2)
def Chal.CubicTorus.canonicalBall : ℝ → Set Chal.CubicTorus.Torus :=
fun V => Metric.ball 0 ((3 * V / (4 * Real.pi)) ^ (1 / 3))
def Chal.CubicTorus.canonicalTube : ℝ → Set Chal.CubicTorus.Torus :=
fun V => {x | ‖x.ofLp 1‖ ^ 2 + ‖x.ofLp 2‖ ^ 2 < √(V / Real.pi) ^ 2}
def Chal.CubicTorus.canonicalSlab : ℝ → Set Chal.CubicTorus.Torus :=
fun V => {x | ‖x.ofLp 0‖ < V / 2}
def Chal.CubicTorus.CongruentAE : Set Chal.CubicTorus.Torus → Set Chal.CubicTorus.Torus → Prop :=
fun E F => ∃ φ, E =ᵐ[Chal.CubicTorus.torusVolume] ⇑φ '' F
def Chal.CubicTorus.SmallStandardRegion : ℝ → Set Chal.CubicTorus.Torus → Prop :=
fun V E =>
  V ≤ 4 * Real.pi / 81 ∧ Chal.CubicTorus.CongruentAE E (Chal.CubicTorus.canonicalBall V) ∨
    4 * Real.pi / 81 ≤ V ∧ V ≤ 1 / Real.pi ∧ Chal.CubicTorus.CongruentAE E (Chal.CubicTorus.canonicalTube V) ∨
      1 / Real.pi ≤ V ∧ Chal.CubicTorus.CongruentAE E (Chal.CubicTorus.canonicalSlab V)
