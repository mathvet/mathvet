def Chal.RuledApproximation.NilpotentStatement : Prop :=
∀ (D : Set Chal.RuledApproximation.Plane) (G : Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane),
  IsCompact D →
    Chal.RuledApproximation.SmoothNear (↑⊤) D G →
      (∀ b ∈ D, fderiv ℝ G b ∘SL fderiv ℝ G b = 0) →
        ∀ (h eps : ℝ),
          0 < h →
            0 < eps →
              ∃ n T g,
                (∀ (i : Fin n), Chal.RuledApproximation.IsSquareTile (T i)) ∧
                  Chal.RuledApproximation.physicalRuled 1 D G 1 ⊆
                      ⋃ i, Chal.RuledApproximation.auxiliaryCylinder (T i) (g i) ∧
                    ∑ i, MeasureTheory.volume.real (T i) * Chal.RuledApproximation.physicalWeight h (g i) ≤
                        (∫ (b : Chal.RuledApproximation.Plane) in D, Chal.RuledApproximation.physicalWeight h (G b)) +
                          eps ∧
                      ∀ (a : ℝ),
                        ∃ C₁ C₂,
                          ∀ (i : Fin n),
                            (C₁ i).parallelogram ∧
                              (C₂ i).parallelogram ∧
                                (C₁ i).carrier =
                                    Chal.RuledApproximation.firstPhysicalMap h ''
                                      Chal.RuledApproximation.auxiliaryCylinder (T i) (g i) ∧
                                  (C₂ i).carrier =
                                      Chal.RuledApproximation.secondPhysicalMap h a ''
                                        Chal.RuledApproximation.auxiliaryCylinder (T i) (g i) ∧
                                    (C₁ i).area =
                                        h * MeasureTheory.volume.real (T i) *
                                          Chal.RuledApproximation.physicalWeight h (g i) ∧
                                      (C₂ i).area =
                                        h * MeasureTheory.volume.real (T i) *
                                          Chal.RuledApproximation.physicalWeight h (g i)
def Chal.RuledApproximation.tetrahedron : Set Chal.RuledApproximation.Space :=
{z | ∃ t, 0 ≤ t ∧ t ≤ 1 ∧ |z.ofLp 0| ≤ 1 - t ∧ |z.ofLp 1| ≤ t ∧ z.ofLp 2 = √2 * t}
def Chal.RuledApproximation.singularField : ℝ → Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane :=
fun p b =>
  Chal.RuledApproximation.vec2 (1 / 2 / 2 * (b.ofLp 1 / (1 / 2)) ^ (p - 1) * (p + (2 - p) * (b.ofLp 0 / b.ofLp 1) ^ 2))
    ((b.ofLp 1 / (1 / 2)) ^ (p - 2) * b.ofLp 0)
def Chal.RuledApproximation.ceilingPolynomial : ℝ → ℝ → ℝ :=
fun p b => b ^ 2 * (1 / 2 * p / 2 + (2 - p) / (2 * (1 / 2)) * b ^ 2)
def Chal.RuledApproximation.singularDomain : ℝ → ℝ → ℝ → Set Chal.RuledApproximation.Plane :=
fun p gamma c =>
  {b |
    c ^ gamma ≤ b.ofLp 1 ∧
      b.ofLp 1 ≤ 2 ∧
        |b.ofLp 0| ≤ b.ofLp 1 ∧
          b.ofLp 1 ≤ 1 / 2 + c ^ 2 * Chal.RuledApproximation.ceilingPolynomial p (b.ofLp 0) + c ^ (5 / 2)}
def Chal.RuledApproximation.singularLower : ℝ → ℝ → ℝ → Set Chal.RuledApproximation.Space :=
fun p gamma c =>
  Chal.RuledApproximation.physicalRuled (√2) (Chal.RuledApproximation.singularDomain p gamma c)
    (fun b => c • Chal.RuledApproximation.singularField p b) 1
def Chal.RuledApproximation.singularUpper : ℝ → ℝ → ℝ → Set Chal.RuledApproximation.Space :=
fun p gamma c =>
  {z |
    ∃ b ∈ Chal.RuledApproximation.singularDomain p gamma c,
      ∃ y,
        |y| ≤ 1 ∧
          z =
            Chal.RuledApproximation.vec3 (b.ofLp 0 - c * y * (Chal.RuledApproximation.singularField p b).ofLp 0) y
              (√2 * (1 - (b.ofLp 1 - c * y * (Chal.RuledApproximation.singularField p b).ofLp 1)))}
def Chal.RuledApproximation.lowerTip : ℝ → Set Chal.RuledApproximation.Space :=
fun d => {z | ∃ t, 0 ≤ t ∧ t ≤ 2 * d ∧ |z.ofLp 1| ≤ t ∧ z.ofLp 2 = √2 * t}
def Chal.RuledApproximation.upperTip : ℝ → Set Chal.RuledApproximation.Space :=
fun d => {z | ∃ s, 0 ≤ s ∧ s ≤ 2 * d ∧ |z.ofLp 0| ≤ s ∧ z.ofLp 2 = √2 * (1 - s)}
def Chal.RuledApproximation.singularCost : ℝ → ℝ → ℝ → ℝ :=
fun p gamma c =>
  √2 *
    ∫ (b : Chal.RuledApproximation.Plane) in Chal.RuledApproximation.singularDomain p gamma c,
      Chal.RuledApproximation.physicalWeight (√2) (c • Chal.RuledApproximation.singularField p b)
def Chal.RuledApproximation.singularCoefficient : ℝ → ℝ :=
fun p => (1 / 2) ^ 4 * (2 * p ^ 2 + 10 * p - 13) / (30 * p)
def Chal.RuledApproximation.SingularStatement : Prop :=
∀ (p gamma : ℝ),
  1 < p →
    p < (√51 - 5) / 2 →
      1 < gamma →
        gamma < 1 / (2 - p) →
          Chal.RuledApproximation.singularCoefficient p < 0 ∧
            ((fun c =>
                  2 * Chal.RuledApproximation.singularCost p gamma c + 8 * √2 * (c ^ gamma) ^ 2 -
                    (√2 / 2 +
                      2 * √2 * Chal.RuledApproximation.singularCoefficient p * c ^ 2)) =o[nhdsWithin 0 (Set.Ioi 0)]
                fun c => c ^ 2) ∧
              ∀ᶠ (c : ℝ) in nhdsWithin 0 (Set.Ioi 0),
                ∃ n m C D B₁ B₂,
                  (∀ (i : Fin n), (C i).parallelogram) ∧
                    (∀ (j : Fin m), (D j).parallelogram) ∧
                      B₁.triangle ∧
                        B₂.triangle ∧
                          Chal.RuledApproximation.singularLower p gamma c ⊆ Chal.RuledApproximation.familySet C ∧
                            Chal.RuledApproximation.singularUpper p gamma c ⊆ Chal.RuledApproximation.familySet D ∧
                              B₁.carrier = Chal.RuledApproximation.lowerTip (c ^ gamma) ∧
                                B₂.carrier = Chal.RuledApproximation.upperTip (c ^ gamma) ∧
                                  Chal.RuledApproximation.tetrahedron ⊆
                                      Chal.RuledApproximation.familySet C ∪ Chal.RuledApproximation.familySet D ∪
                                          B₁.carrier ∪
                                        B₂.carrier ∧
                                    Chal.RuledApproximation.familyArea C + Chal.RuledApproximation.familyArea D +
                                          B₁.area +
                                        B₂.area <
                                      √2 / 2
