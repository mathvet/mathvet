  (i : ℕ) → X.Cochain i → X.Cochain (i + 1) :=

def OAI.CoboundaryExpanders.Complex.coboundaries : (X : OAI.CoboundaryExpanders.Complex) →
  (i : ℕ) → Set (X.Cochain i) :=
fun X x =>
  match x with
  | 0 => {f | ∃ c, f = fun x => c}
  | i.succ => Set.range (X.coboundary i)

fun X s i f A => sInf ((fun a => X.norm s i (f - a)) '' A)
def OAI.CoboundaryExpanders.Complex.topDegree : (X : OAI.CoboundaryExpanders.Complex) → ℕ → Fin X.vertexCount → ℕ :=
fun X s v => {τ ∈ X.facesOf s | v ∈ τ}.card
def OAI.CoboundaryExpanders.MainStatement : Prop :=
∀ (d : ℕ),
  3 ≤ d →
    ∃ D ε,
      0 < ε ∧
        ∃ X,
          (∀ (m : ℕ), (X m).Pure d ∧ (X m).Connected) ∧
            Filter.Tendsto (fun m => ((X m).facesOf 0).card) Filter.atTop Filter.atTop ∧
              (∀ (m : ℕ) (v : Fin (X m).vertexCount), (X m).topDegree d v ≤ D) ∧
                ∀ (m i : ℕ),
                  i < d →
                    ∀ (f : (X m).Cochain i),
                      ε * (X m).distance d i f ((X m).coboundaries i) ≤ (X m).norm d (i + 1) ((X m).coboundary i f)
