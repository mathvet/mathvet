OAI.OrdinaryCorrelations.binary_corrected_elliott : ∀ (f₁ f₂ : ℕ → ℂ),
  OAI.OrdinaryCorrelations.OneBounded f₁ →
    OAI.OrdinaryCorrelations.OneBounded f₂ →
      OAI.OrdinaryCorrelations.Multiplicative f₁ →
        OAI.OrdinaryCorrelations.Multiplicative f₂ →
          OAI.OrdinaryCorrelations.UniformlyNonpretentious f₁ ∨ OAI.OrdinaryCorrelations.UniformlyNonpretentious f₂ →
            ∀ (h₁ h₂ : ℕ),
              h₁ ≠ h₂ → Filter.Tendsto (OAI.OrdinaryCorrelations.shiftAverage f₁ f₂ h₁ h₂) Filter.atTop (nhds 0)
'OAI.OrdinaryCorrelations.binary_corrected_elliott' depends on axioms: [propext, Classical.choice, Quot.sound]
def OAI.OrdinaryCorrelations.OneBounded : (ℕ → ℂ) → Prop :=
fun f => ∀ (n : ℕ), ‖f n‖ ≤ 1
def OAI.OrdinaryCorrelations.Multiplicative : (ℕ → ℂ) → Prop :=
fun f => ∀ (m n : ℕ), 0 < m → 0 < n → m.Coprime n → f (m * n) = f m * f n
def OAI.OrdinaryCorrelations.distanceSq : (ℕ → ℂ) → {q : ℕ} → DirichletCharacter ℂ q → ℝ → ℝ → ℝ :=
fun f {q} χ t X =>
  ∑ p ∈ Finset.Icc 2 ⌊X⌋₊ with Nat.Prime p,
    (1 - (f p * star (χ ↑p * Complex.exp (↑(t * Real.log ↑p) * Complex.I))).re) / ↑p
def OAI.OrdinaryCorrelations.distance : (ℕ → ℂ) → {q : ℕ} → DirichletCharacter ℂ q → ℝ → ℝ → ℝ :=
fun f {q} χ t X => √(OAI.OrdinaryCorrelations.distanceSq f χ t X)
def OAI.OrdinaryCorrelations.UniformlyNonpretentious : (ℕ → ℂ) → Prop :=
fun f =>
  ∀ (q : ℕ),
    0 < q →
      ∀ (χ : DirichletCharacter ℂ q),
        Filter.Tendsto (fun N => sInf ((fun t => OAI.OrdinaryCorrelations.distance f χ t ↑N) '' Set.Icc (-↑N) ↑N))
          Filter.atTop Filter.atTop
def OAI.OrdinaryCorrelations.shiftAverage : (ℕ → ℂ) → (ℕ → ℂ) → ℕ → ℕ → ℕ → ℂ :=
fun f₁ f₂ h₁ h₂ N => (↑N)⁻¹ * ∑ n ∈ Finset.Icc 1 N, f₁ (n + h₁) * f₂ (n + h₂)
