Chal.ThresholdParallelRepetition.threshold_parallel_repetition : ∃ κ₀,
  0 < κ₀ ∧
    ∀ (x y a b : ℕ) (G : Chal.ThresholdParallelRepetition.Game x y a b),
      Chal.ThresholdParallelRepetition.entangledValue G < 1 →
        ∀ (δ : ℝ),
          0 < δ →
            δ < 1 - Chal.ThresholdParallelRepetition.entangledValue G →
              ∀ (k : ℕ),
                1 ≤ k →
                  (∀ (S : Chal.ThresholdParallelRepetition.RepeatedStrategy x y a b k),
                      Chal.ThresholdParallelRepetition.thresholdProbability G δ S ≤
                        Real.exp (-κ₀ * δ ^ 13 / (1 + Real.log ↑((a + 1) * (b + 1))) * ↑k)) ∧
                    Chal.ThresholdParallelRepetition.thresholdValue G δ ≤
                      Real.exp (-κ₀ * δ ^ 13 / (1 + Real.log ↑((a + 1) * (b + 1))) * ↑k)

def Chal.ThresholdParallelRepetition.thresholdProbability : {x y a b k : ℕ} →
  Chal.ThresholdParallelRepetition.Game x y a b → ℝ → Chal.ThresholdParallelRepetition.RepeatedStrategy x y a b k → ℝ :=
fun {x y a b k} G δ S =>
  ∑ xs,
    ∑ ys,
      (∏ i, G.questionProb (xs i, ys i)) *
        ∑ as,
          ∑ bs,
            if
                ⌈(Chal.ThresholdParallelRepetition.entangledValue G + δ) * ↑k⌉₊ ≤
                  Chal.ThresholdParallelRepetition.wins G xs ys as bs then
              Chal.ThresholdParallelRepetition.Strategy.born S xs ys as bs
            else 0

fun {x y a b k} G δ => sSup (Set.range (Chal.ThresholdParallelRepetition.thresholdProbability G δ))
