@OAI.PlanarAnderson.anderson_spectrum_ae : ∀ {h : ℝ},
  0 < h →
    ∀ᵐ (v : OAI.PlanarAnderson.Configuration) ∂OAI.PlanarAnderson.disorderLaw h,
      ∃ H,
        OAI.PlanarAnderson.IsAndersonOperator v H ∧
          IsSelfAdjoint H ∧ spectrum ℂ H = OAI.PlanarAnderson.spectralInterval h
'OAI.PlanarAnderson.anderson_spectrum_ae' depends on axioms: [propext, Classical.choice, Quot.sound]
@[reducible] def OAI.PlanarAnderson.Site : Type :=
ℤ × ℤ
@[reducible] def OAI.PlanarAnderson.Configuration : Type :=
OAI.PlanarAnderson.Site → ℝ
@[reducible] def OAI.PlanarAnderson.Hilbert : AddSubgroup (PreLp fun x => ℂ) :=
lp (fun x => ℂ) 2
def OAI.PlanarAnderson.disorderLaw : ℝ → MeasureTheory.Measure OAI.PlanarAnderson.Configuration :=
fun h => MeasureTheory.Measure.infinitePi fun x => MeasureTheory.volume[|Set.Icc (-h) h]
def OAI.PlanarAnderson.IsAndersonOperator : OAI.PlanarAnderson.Configuration →
  (↥OAI.PlanarAnderson.Hilbert →L[ℂ] ↥OAI.PlanarAnderson.Hilbert) → Prop :=
fun v H =>
  ∀ (u : ↥OAI.PlanarAnderson.Hilbert) (x : OAI.PlanarAnderson.Site),
    ↑(H u) x = ↑u (x.1 + 1, x.2) + ↑u (x.1 - 1, x.2) + ↑u (x.1, x.2 + 1) + ↑u (x.1, x.2 - 1) + ↑(v x) * ↑u x
def OAI.PlanarAnderson.spectralInterval : ℝ → Set ℂ :=
fun h => {z | z.im = 0 ∧ -4 - h ≤ z.re ∧ z.re ≤ 4 + h}
