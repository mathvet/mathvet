    OAI.HarmonicArtin.alternating st.1 st.2 (M.M st.1 st.2) *

def OAI.HarmonicArtin.ParabolicIntersections.gromovProduct : {X : Type} → [MetricSpace X] → X → X → X → ℝ :=
fun {X} [MetricSpace X] o x y => (dist o x + dist o y - dist x y) / 2
def OAI.HarmonicArtin.ParabolicIntersections.IsGromovHyperbolic : (X : Type) → [MetricSpace X] → Prop :=
fun X [MetricSpace X] =>
  ∃ δ,
    0 ≤ δ ∧
      ∀ (o x y z : X),
        min (OAI.HarmonicArtin.ParabolicIntersections.gromovProduct o x y)
            (OAI.HarmonicArtin.ParabolicIntersections.gromovProduct o y z) ≤
          OAI.HarmonicArtin.ParabolicIntersections.gromovProduct o x z + δ
def OAI.HarmonicArtin.ParabolicIntersections.IsGeodesicMetric : (X : Type) → [MetricSpace X] → Prop :=
fun X [MetricSpace X] => ∀ (x y : X), ∃ γ, Isometry γ ∧ γ ⟨0, ⋯⟩ = x ∧ γ ⟨dist x y, ⋯⟩ = y
def OAI.HarmonicArtin.ParabolicIntersections.IsGromovSequence : {X : Type} → [MetricSpace X] → X → (ℕ → X) → Prop :=

  ∀ (R : ℝ),
    ∃ N, ∀ (m : ℕ), N ≤ m → ∀ (n : ℕ), N ≤ n → R ≤ OAI.HarmonicArtin.ParabolicIntersections.gromovProduct o (u m) (u n)
def OAI.HarmonicArtin.ParabolicIntersections.IsAcylindrical : (G X : Type) →
  [inst : Group G] → [MetricSpace X] → [MulAction G X] → Prop :=
fun G X [Group G] [MetricSpace X] [MulAction G X] =>
  ∀ (ε : ℝ),
    0 < ε →
      ∃ R N,
        ∀ (x y : X),
          R ≤ dist x y →
            {g | dist x (g • x) ≤ ε ∧ dist y (g • y) ≤ ε}.Finite ∧
              {g | dist x (g • x) ≤ ε ∧ dist y (g • y) ≤ ε}.ncard ≤ N
