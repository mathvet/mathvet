OAI.SquareDifference.power_saving : ∃ c C,
  0 < c ∧
    ∀ (N : ℕ), 1 ≤ N → ∀ A ⊆ Finset.Icc 1 ↑N, OAI.SquareDifference.IsSquareDifferenceFree A → ↑A.card ≤ C * ↑N ^ (1 - c)
'OAI.SquareDifference.power_saving' depends on axioms: [propext, Classical.choice, Quot.sound]
def OAI.SquareDifference.IsSquareDifferenceFree : Finset ℤ → Prop :=
fun A => ∀ a ∈ A, ∀ b ∈ A, ∀ (m : ℕ), 1 ≤ m → a - b ≠ ↑m ^ 2
