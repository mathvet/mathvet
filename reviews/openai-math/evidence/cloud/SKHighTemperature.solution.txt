def OAI.SKGap.spinValue : Bool → ℝ :=
fun b => if b = true then 1 else -1
def OAI.SKGap.coupling : {n : ℕ} → OAI.SKGap.Disorder n → Fin n → Fin n → ℝ :=
fun {n} g i k => if h : i < k then g ⟨(i, k), h⟩ else if h : k < i then g ⟨(k, i), h⟩ else 0
def OAI.SKGap.hamiltonian : {n : ℕ} → OAI.SKGap.Disorder n → (Fin n → ℝ) → OAI.SKGap.Spin n → ℝ :=
fun {n} g h x =>
  1 / 2 * ∑ i, ∑ k, OAI.SKGap.spinValue (x i) * OAI.SKGap.coupling g i k * OAI.SKGap.spinValue (x k) +
    ∑ i, h i * OAI.SKGap.spinValue (x i)
def OAI.SKGap.weight : {n : ℕ} → OAI.SKGap.Disorder n → (Fin n → ℝ) → OAI.SKGap.Spin n → ℝ :=
fun {n} g h x => Real.exp (OAI.SKGap.hamiltonian g h x)
def OAI.SKGap.partition : {n : ℕ} → OAI.SKGap.Disorder n → (Fin n → ℝ) → ℝ :=
fun {n} g h => ∑ x, OAI.SKGap.weight g h x

fun {n} g h x => OAI.SKGap.weight g h x / OAI.SKGap.partition g h
def OAI.SKGap.expectation : {n : ℕ} → OAI.SKGap.Disorder n → (Fin n → ℝ) → (OAI.SKGap.Spin n → ℝ) → ℝ :=
fun {n} g h f => ∑ x, OAI.SKGap.mass g h x * f x
def OAI.SKGap.variance : {n : ℕ} → OAI.SKGap.Disorder n → (Fin n → ℝ) → (OAI.SKGap.Spin n → ℝ) → ℝ :=
fun {n} g h f => OAI.SKGap.expectation g h fun x => (f x - OAI.SKGap.expectation g h f) ^ 2
def OAI.SKGap.flip : {n : ℕ} → Fin n → OAI.SKGap.Spin n → OAI.SKGap.Spin n :=
fun {n} i x => Function.update x i !x i
def OAI.SKGap.conditionalExpectation : {n : ℕ} →
  OAI.SKGap.Disorder n → (Fin n → ℝ) → Fin n → (OAI.SKGap.Spin n → ℝ) → OAI.SKGap.Spin n → ℝ :=
fun {n} g h i f x =>
  (OAI.SKGap.weight g h x * f x + OAI.SKGap.weight g h (OAI.SKGap.flip i x) * f (OAI.SKGap.flip i x)) /
    (OAI.SKGap.weight g h x + OAI.SKGap.weight g h (OAI.SKGap.flip i x))
def OAI.SKGap.dirichlet : {n : ℕ} → OAI.SKGap.Disorder n → (Fin n → ℝ) → (OAI.SKGap.Spin n → ℝ) → ℝ :=
fun {n} g h f => ∑ i, OAI.SKGap.expectation g h fun x => (f x - OAI.SKGap.conditionalExpectation g h i f x) ^ 2
def OAI.SKGap.disorderLaw : ℝ → (n : ℕ) → MeasureTheory.Measure (OAI.SKGap.Disorder n) :=
fun β n => MeasureTheory.Measure.pi fun x => ProbabilityTheory.gaussianReal 0 (β ^ 2 / ↑n).toNNReal
def OAI.SKGap.poincareEvent : (n : ℕ) → ℝ → Set (OAI.SKGap.Disorder n) :=
fun n C => {g | ∀ (f : OAI.SKGap.Spin n → ℝ), OAI.SKGap.variance g 0 f ≤ C * OAI.SKGap.dirichlet g 0 f}
def OAI.SKGap.discreteGap : {n : ℕ} → OAI.SKGap.Disorder n → ℝ :=
fun {n} g =>
  sInf {r | ∃ f, 0 < OAI.SKGap.variance g 0 f ∧ r = OAI.SKGap.dirichlet g 0 f / (↑n * OAI.SKGap.variance g 0 f)}
def OAI.SKGap.MainStatement : ℝ → Prop :=
fun β =>
  ∃ C,
    0 < C ∧
      Filter.Tendsto (fun n => (OAI.SKGap.disorderLaw β n) (OAI.SKGap.poincareEvent n C)) Filter.atTop (nhds 1) ∧
        Filter.Tendsto (fun n => (OAI.SKGap.disorderLaw β n) {g | 1 / (C * ↑n) ≤ OAI.SKGap.discreteGap g}) Filter.atTop
          (nhds 1)
