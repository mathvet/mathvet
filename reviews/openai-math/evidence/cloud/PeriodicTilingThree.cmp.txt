Chal.PeriodicTilingThree.periodic_tiling_counterexample_and_minimality : (∃ T,
    T.Nonempty ∧
      (∃ A, Chal.PeriodicTilingThree.Tiles T A) ∧
        (∀ (A : Set (Chal.PeriodicTilingThree.Lattice 3)),
            Chal.PeriodicTilingThree.Tiles T A → ¬Chal.PeriodicTilingThree.FullyPeriodic A) ∧
          (∃ A, Chal.PeriodicTilingThree.AETiles (Chal.PeriodicTilingThree.Thickening T) A) ∧
            ∀ (A : Set (Chal.PeriodicTilingThree.Space 3)),

                ¬Chal.PeriodicTilingThree.EuclideanFullyPeriodic A) ∧
