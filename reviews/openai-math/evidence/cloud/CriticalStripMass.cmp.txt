def Chal.CriticalStrip.archWeight : ℕ → ℤ → List Chal.CriticalStrip.Triangle → ℝ :=
fun N k p =>
  if k ≠ 0 ∧ Chal.CriticalStrip.IsPath N (Chal.CriticalStrip.up k 0) p then Chal.CriticalStrip.rho ^ p.length else 0
def Chal.CriticalStrip.bridgeWeight : ℕ → ℤ → List Chal.CriticalStrip.Triangle → ℝ :=
fun N k p =>
  if 0 < N ∧ Chal.CriticalStrip.IsPath N (Chal.CriticalStrip.down k (N - 1)) p then Chal.CriticalStrip.rho ^ p.length
  else 0
def Chal.CriticalStrip.K : ℕ → ℤ → ℝ :=

def Chal.CriticalStrip.bridgeMass : ℕ → ℝ :=

def Chal.CriticalStrip.moment : ℕ → ℝ :=

def Chal.CriticalStrip.StripSumsFinite : ℕ → Prop :=
fun N =>
  (∀ (k : ℤ), Summable (Chal.CriticalStrip.archWeight N k)) ∧
    (∀ (k : ℤ), Summable (Chal.CriticalStrip.bridgeWeight N k)) ∧
      Summable (Chal.CriticalStrip.K N) ∧
        Summable (Chal.CriticalStrip.bridgeKernel N) ∧ Summable fun k => (↑k + 1) * Chal.CriticalStrip.K N (↑k + 1)
def Chal.CriticalStrip.Comparable : (ℕ → ℝ) → (ℕ → ℝ) → Prop :=
