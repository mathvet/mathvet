def Chal.UniversalCoverSplitting.CompatibleProduct.{u, v} : {n r₁ r₂ : ℕ} →
  {X : Chal.UniversalCoverSplitting.ComplexManifold n} →
    {Z : Chal.UniversalCoverSplitting.ComplexManifold n} →
      Chal.UniversalCoverSplitting.HolomorphicSplitting X r₁ r₂ →
        Chal.UniversalCoverSplitting.OrdinaryUniversalCover X Z →
          Chal.UniversalCoverSplitting.ComplexManifold r₁ → Chal.UniversalCoverSplitting.ComplexManifold r₂ → Prop :=
fun {n r₁ r₂} {X} {Z} S π Y₁ Y₂ =>
  ∃ Φ,
    MDiff ⇑Φ ∧
      MDiff ⇑Φ.symm ∧
        (∀ (z : Z.Carrier),
            Submodule.map (↑(mfderiv% ⇑Φ z))
                (Submodule.comap (↑(mfderiv% π.projection z)) (↑(S.projection (π.projection z))).range) =
              (↑(ContinuousLinearMap.snd ℂ (Chal.UniversalCoverSplitting.Model r₁)
                    (Chal.UniversalCoverSplitting.Model r₂))).ker) ∧
          ∀ (z : Z.Carrier),
            Submodule.map (↑(mfderiv% ⇑Φ z))
                (Submodule.comap (↑(mfderiv% π.projection z)) (↑(S.projection (π.projection z))).ker) =
              (↑(ContinuousLinearMap.fst ℂ (Chal.UniversalCoverSplitting.Model r₁)
                    (Chal.UniversalCoverSplitting.Model r₂))).ker
