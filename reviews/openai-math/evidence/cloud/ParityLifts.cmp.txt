def Chal.ParityWL.ParityReductionStatement : Prop :=
∀ (k : ℕ),
  4 ≤ k →
    ∀ (c : Chal.ParityWL.Convention) (C : Chal.ParityWL.ChoiceSystem (Chal.ParityWL.bagParameter c k)),
      let T := Chal.ParityWL.template (Chal.ParityWL.bagParameter c k);
      let G := Chal.ParityWL.baseGraph C;
      have τ := Chal.ParityWL.baseProjection C;
      let V₀ := Chal.ParityWL.LiftVertex T G τ fun x => 0;
      let V₁ := Chal.ParityWL.LiftVertex T G τ (Chal.ParityWL.starParity (Chal.ParityWL.bagParameter c k));
      Fintype.card V₀ = Fintype.card V₁ ∧
        (∀ (A : ℕ),
            1 ≤ A → C.SizeAtMost A → Fintype.card V₀ ≤ A * Chal.ParityWL.orderFactor (Chal.ParityWL.bagParameter c k)) ∧
          (Chal.ParityWL.WLEquivalent (Chal.ParityWL.zeroLift C) (Chal.ParityWL.starLift C) c k ↔ ¬C.Successful) ∧
            (Chal.ParityWL.PureWLEquivalent (Chal.ParityWL.zeroLift C) (Chal.ParityWL.starLift C) c k ↔ ¬C.Successful) ∧
              (C.Successful →
                ¬Chal.ParityWL.sameHistogram (Chal.ParityWL.zeroLift C) (Chal.ParityWL.starLift C) c k
                      (Chal.ParityWL.detectionRound c) ∧
                  ¬Chal.ParityWL.samePureHistogram (Chal.ParityWL.zeroLift C) (Chal.ParityWL.starLift C) c k
                      (Chal.ParityWL.detectionRound c))
