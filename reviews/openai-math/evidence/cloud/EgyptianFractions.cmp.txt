def Chal.Problem337.IsOneExpansion : {k : ℕ} → (Fin k → ℕ) → Prop :=
fun {k} n => (∀ (i : Fin k), 1 ≤ n i) ∧ StrictMono n ∧ ∑ i, 1 / ↑(n i) = 1
