def Chal.MetricKMedian.encodeRat : ℚ → List Bool :=

def Chal.MetricKMedian.encodeInstance : Chal.MetricKMedian.Instance → List Bool :=
fun I =>
  ((Chal.MetricKMedian.encodeNat I.n ++ Chal.MetricKMedian.encodeNat I.k ++
        List.ofFn fun i => decide (i ∈ I.clients)) ++
      List.ofFn fun i => decide (i ∈ I.facilities)) ++

def Chal.MetricKMedian.nearest : (I : Chal.MetricKMedian.Instance) → Fin I.n → Finset (Fin I.n) → ℝ :=
fun I j S => if h : S.Nonempty then (Finset.image (fun i => ↑(I.distance j i)) S).min' ⋯ else 0
def Chal.MetricKMedian.cost : (I : Chal.MetricKMedian.Instance) → Finset (Fin I.n) → ℝ :=
fun I S => ∑ j ∈ I.clients, Chal.MetricKMedian.nearest I j S
def Chal.MetricKMedian.feasibleSets : (I : Chal.MetricKMedian.Instance) → Finset (Finset (Fin I.n)) :=
fun I => {S ∈ I.facilities.powerset | S.Nonempty ∧ S.card ≤ I.k}
def Chal.MetricKMedian.optimum : Chal.MetricKMedian.Instance → ℝ :=
fun I =>
  if h : (Chal.MetricKMedian.feasibleSets I).Nonempty then
    (Finset.image (Chal.MetricKMedian.cost I) (Chal.MetricKMedian.feasibleSets I)).min' ⋯
  else 0
def Chal.MetricKMedian.Correct : ℝ → Chal.MetricKMedian.Instance → List Bool → Prop :=
fun epsilon I output =>
  ∃ S ⊆ I.facilities,
    S.Nonempty ∧
      S.card ≤ I.k ∧
        (output = List.ofFn fun i => decide (i ∈ S)) ∧
          Chal.MetricKMedian.cost I S ≤ (1 + 2 / Real.exp 1 + epsilon) * Chal.MetricKMedian.optimum I
