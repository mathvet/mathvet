def OAI.GammaFPP.HasC1CurveChart : Set OAI.PlanarFPP.Plane → OAI.PlanarFPP.Plane → Prop :=
fun S v =>
  ∃ U,
    IsOpen U ∧
      v ∈ U ∧
        v ∈ S ∧
          ∃ γ θ,
            ContDiff ℝ 1 γ ∧
              ContinuousOn θ U ∧
                (∀ (t : ℝ), γ t ∈ S ∩ U) ∧
                  (∀ (t : ℝ), θ (γ t) = t) ∧ (∀ z ∈ S ∩ U, γ (θ z) = z) ∧ ∀ (t : ℝ), ∃ d, HasDerivAt γ d t ∧ d ≠ 0
def OAI.GammaFPP.GammaDifferentiabilityTarget : Prop :=
∀ (shape rate : ℝ),
  0 < shape →
    0 < rate →
      ∀ (Ω : Type) [inst : MeasurableSpace Ω] (P : MeasureTheory.Measure Ω) (τ : OAI.PlanarFPP.Edge → Ω → ℝ),
        OAI.GammaFPP.GammaEnvironment shape rate P τ →
          ∃ μ,
            OAI.PlanarFPP.TimeConstantNorm μ ∧
              OAI.PlanarFPP.IsTimeConstant P τ μ ∧
                OAI.PlanarFPP.DifferentiableAwayFromOrigin μ ∧
                  ∀ v ∈ frontier {z | μ z ≤ 1}, OAI.GammaFPP.HasC1CurveChart (frontier {z | μ z ≤ 1}) v
