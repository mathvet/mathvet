def OAI.KServer.MainStatement.{u} : Prop :=
∃ C,
  0 < C ∧
    ∀ (k : ℕ),
      2 ≤ k →
        ∀ (X : Type u) (x : MetricSpace X),
          (∃ e, Function.Injective e) →
            ∀ (s : OAI.KServer.Configuration k X),
              ∃ A B,
                0 ≤ B ∧
                  (Function.Injective s → B = 0) ∧
                    ∀ (σ : List X),
                      OAI.KServer.expectedCost A s σ ≤ C * Real.log (↑k + 1) ^ 2 * OAI.KServer.optimalCost s σ + B
