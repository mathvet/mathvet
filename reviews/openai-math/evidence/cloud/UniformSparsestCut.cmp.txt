def Chal.UniformSparsestCut.pairSum : {n : ℕ} → (Fin n → Fin n → ℝ) → ℝ :=
fun {n} f => ∑ i, ∑ j, if i < j then f i j else 0
def Chal.UniformSparsestCut.cutRatio : {n : ℕ} → Chal.UniformSparsestCut.Capacity n → Finset (Fin n) → ℝ :=
fun {n} C B => (∑ i ∈ B, ∑ j ∈ Bᶜ, C.cap i j) / (↑B.card * (↑n - ↑B.card))
def Chal.UniformSparsestCut.OPT : {n : ℕ} → Chal.UniformSparsestCut.Capacity n → ℝ :=
fun {n} C => sInf {r | ∃ B, B.Nonempty ∧ B ≠ Finset.univ ∧ r = Chal.UniformSparsestCut.cutRatio C B}
def Chal.UniformSparsestCut.NegativeType : {n : ℕ} → (Fin n → Fin n → ℝ) → Prop :=
fun {n} d => (∃ x, ∀ (i j : Fin n), d i j = ‖x i - x j‖ ^ 2) ∧ ∀ (i j k : Fin n), d i k ≤ d i j + d j k
def Chal.UniformSparsestCut.Feasible : {n : ℕ} → (Fin n → Fin n → ℝ) → Prop :=
fun {n} d => Chal.UniformSparsestCut.NegativeType d ∧ Chal.UniformSparsestCut.pairSum d = 1
def Chal.UniformSparsestCut.glValue : {n : ℕ} → Chal.UniformSparsestCut.Capacity n → ℝ :=
fun {n} C =>
  sInf {r | ∃ d, Chal.UniformSparsestCut.Feasible d ∧ r = Chal.UniformSparsestCut.pairSum fun i j => C.cap i j * d i j}
