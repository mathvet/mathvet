def OAI.DirectionalTransience.MainStatement : Prop :=
∀ (d : ℕ),
  2 ≤ d →
    ∀ (ν : MeasureTheory.Measure (OAI.DirectionalTransience.Row d)),
      MeasureTheory.IsProbabilityMeasure ν →
        OAI.DirectionalTransience.UniformElliptic ν →
          ∀ (ℓ : OAI.DirectionalTransience.Vector d),
            OAI.DirectionalTransience.dot ℓ ℓ = 1 →
              OAI.DirectionalTransience.DirectionallyTransient ν ℓ →
                ∃ v, 0 < OAI.DirectionalTransience.dot v ℓ ∧ OAI.DirectionalTransience.HasVelocity ν v
