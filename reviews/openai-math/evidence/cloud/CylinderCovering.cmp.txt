def Chal.RuledApproximation.SmoothNear : WithTop ℕ∞ →
  Set Chal.RuledApproximation.Plane → (Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane) → Prop :=
fun n D v => ∃ U, IsOpen U ∧ D ⊆ U ∧ ContDiffOn ℝ n v U
def Chal.RuledApproximation.HyperbolicAt : ℝ →
  (Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane) → Chal.RuledApproximation.Plane → Prop :=
fun L v p =>
  ∃ lam,
    0 < L * lam ∧
      L * lam < 1 ∧ Module.End.HasEigenvalue (↑(fderiv ℝ v p)) lam ∧ Module.End.HasEigenvalue (↑(fderiv ℝ v p)) (-lam)

fun v => 1 / √(1 + ‖v‖ ^ 2)
def Chal.RuledApproximation.physicalWeight : ℝ → Chal.RuledApproximation.Plane → ℝ :=
fun h w => 1 / √(1 + w.ofLp 0 ^ 2 + h ^ 2 * w.ofLp 1 ^ 2)

  (Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane) → ℝ → Set Chal.RuledApproximation.Space :=
fun D v L =>
  {z |
    ∃ p ∈ D,
      ∃ r, |r| ≤ L ∧ z = Chal.RuledApproximation.vec3 (p.ofLp 0 + r * (v p).ofLp 0) (p.ofLp 1 + r * (v p).ofLp 1) r}

  Set Chal.RuledApproximation.Plane →
    (Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane) → ℝ → Set Chal.RuledApproximation.Space :=
fun h D v L =>
  {z |
    ∃ p ∈ D,
      ∃ s,
        |s| ≤ L ∧ z = Chal.RuledApproximation.vec3 s (p.ofLp 0 + s * (v p).ofLp 0) (h * (p.ofLp 1 + s * (v p).ofLp 1))}
def Chal.RuledApproximation.HyperbolicStatement : Prop :=
∀ (D : Set Chal.RuledApproximation.Plane) (v : Chal.RuledApproximation.Plane → Chal.RuledApproximation.Plane) (L : ℝ),

                              (C₂ i).parallelogram ∧

                                    Chal.RuledApproximation.firstPhysicalMap h ''
                                      Chal.RuledApproximation.auxiliaryCylinder (T i) (g i) ∧

                                      Chal.RuledApproximation.secondPhysicalMap h a ''
                                        Chal.RuledApproximation.auxiliaryCylinder (T i) (g i) ∧
                                    (C₁ i).area =
                                        h * MeasureTheory.volume.real (T i) *
                                          Chal.RuledApproximation.physicalWeight h (g i) ∧
                                      (C₂ i).area =
                                        h * MeasureTheory.volume.real (T i) *
                                          Chal.RuledApproximation.physicalWeight h (g i)
