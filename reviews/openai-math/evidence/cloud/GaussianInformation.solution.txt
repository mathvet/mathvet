def OAI.RepeatedLocalization.cellSet : {n : ℕ} →
  OAI.RepeatedLocalization.Cell n → Set (OAI.RepeatedLocalization.Coordinate n) :=
fun {n} Q =>
  {z |

      -1 / (2 * √↑n) + ↑↑(Q.snd a) / (√↑n * 2 ^ Q.fst) ≤ z a ∧
        z a < -1 / (2 * √↑n) + (↑↑(Q.snd a) + 1) / (√↑n * 2 ^ Q.fst)}

def OAI.RepeatedLocalization.ValidDisclosure : {n N t : ℕ} → OAI.RepeatedLocalization.Disclosure n N t → Prop :=
fun {n N t} Q =>
  (∀ (i : Fin t), Measurable fun zh => Q i zh.1 zh.2) ∧
    (∀ (i : Fin t), (Set.range fun zh => Q i zh.1 zh.2).Finite) ∧
      (∀ z ∈ OAI.RepeatedLocalization.initialCube n,
          ∀ (w : OAI.RepeatedLocalization.History N t) (i : Fin t),
            z ∈ OAI.RepeatedLocalization.cellSet (Q i z (OAI.RepeatedLocalization.through w i)) ∧
              OAI.RepeatedLocalization.cellSet (Q i z (OAI.RepeatedLocalization.through w i)) ⊆
                OAI.RepeatedLocalization.initialCube n) ∧
        ∀ z ∈ OAI.RepeatedLocalization.initialCube n,
          ∀ (w : OAI.RepeatedLocalization.History N t) (i j : Fin t),
            i ≤ j →
              OAI.RepeatedLocalization.cellSet (Q j z (OAI.RepeatedLocalization.through w j)) ⊆
                OAI.RepeatedLocalization.cellSet (Q i z (OAI.RepeatedLocalization.through w i))

      OAI.RepeatedLocalization.Coordinate n ×
        OAI.RepeatedLocalization.History N t × (Fin t → OAI.RepeatedLocalization.Cell n) :=
fun {n N t} Q zw => (zw.1, zw.2, fun i => Q i zw.1 (OAI.RepeatedLocalization.through zw.2 i))

  if ht : 0 < t then

def OAI.RepeatedLocalization.MainStatement : Prop :=
∃ c,
  0 < c ∧
    ∃ n₀,
      ∀ (C₀ : ℝ),
        0 < C₀ →
          ∃ C,
            0 < C ∧
              ∀ (n N t : ℕ),
                n₀ ≤ n →
                  1 ≤ N →
                    ∀ (rules : OAI.RepeatedLocalization.Rules n N t),
                      (2 / ↑(n / 8) + Real.exp (-c * ↑(n / 16))) * Real.log ↑N ≤ C₀ * (↑n + 1) →
                        ∃ Q,

                            OAI.RepeatedLocalization.mutualInformation (OAI.RepeatedLocalization.augmentedLaw rules Q) ≤
                                ENNReal.ofReal (C * (↑n + 1) * ↑t) ∧

                                  ↑(OAI.RepeatedLocalization.finalLevel Q

def OAI.KernelPotential.CoercivityStatement : Prop :=
∃ C,
  0 < C ∧
    ∃ d₀,
      ∀ (d : ℕ),
        d₀ ≤ d →
          (∀ (ν : MeasureTheory.Measure (OAI.KernelPotential.Ambient d)),
              MeasureTheory.IsProbabilityMeasure ν →
                OAI.KernelPotential.BoundedDensity ν →
                  1 / 2 * (InformationTheory.klDiv ν (OAI.KernelPotential.sphereProbability d)).toReal - C * ↑d ≤
                    OAI.KernelPotential.Phi ν) ∧
            OAI.KernelPotential.Phi (OAI.KernelPotential.sphereProbability d) ≤ ↑(2 * (d / 16)) * Real.log 2 ∧
              ∀ (q : ℕ) (w : Fin q → ℝ),
                (∀ (i : Fin q), 0 ≤ w i) →
                  ∑ i, w i = 1 →
                    ∀ (ν : Fin q → MeasureTheory.Measure (OAI.KernelPotential.Ambient d)),
                      (∀ (i : Fin q), MeasureTheory.IsProbabilityMeasure (ν i)) →
                        (∀ (i : Fin q), OAI.KernelPotential.BoundedDensity (ν i)) →
                          OAI.KernelPotential.Phi (∑ i, ENNReal.ofReal (w i) • ν i) ≤
                            ∑ i, w i * OAI.KernelPotential.Phi (ν i)
def OAI.KernelPotential.DriftStatement : Prop :=
∃ C,
  0 < C ∧
    ∃ d₀,
      ∀ (d : ℕ),
        d₀ ≤ d →
          ∀ (N : ℕ),
            1 ≤ N →
              Real.log ↑N ≤ ↑d ^ 2 →
                ∀ (μ : MeasureTheory.Measure (OAI.KernelPotential.Ambient d × Fin N)),
                  MeasureTheory.IsProbabilityMeasure μ →
                    MeasureTheory.Measure.map Prod.fst μ = OAI.KernelPotential.sphereProbability d →
                      ∀ (rule : OAI.KernelPotential.BlockRule d N),
                        OAI.KernelPotential.averagePhi (OAI.KernelPotential.outgoingLaw μ rule) ≤
                          OAI.KernelPotential.averagePhi μ + C * ↑d
