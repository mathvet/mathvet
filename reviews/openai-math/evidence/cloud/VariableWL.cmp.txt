def Chal.VariableWL.wordCells : (a : ℕ) → Chal.VariableWL.Word → ℤ → Fin (a + 3) :=
fun a w i =>
  if 0 ≤ i then
    match w[i.toNat]? with
    | none => 0
    | some false => 1
    | some true => 2
  else 0
def Chal.VariableWL.initial : (M : Chal.VariableWL.Machine) → Chal.VariableWL.Word → Chal.VariableWL.Configuration M :=
fun M w =>
  { control := 0, heads := fun x => 0,
    cells := fun j => if j = 0 then Chal.VariableWL.wordCells M.workSymbols w else fun x => 0 }
def Chal.VariableWL.scanned : {M : Chal.VariableWL.Machine} →
  Chal.VariableWL.Configuration M → Fin (M.extraTapes + 1) → Fin (M.workSymbols + 3) :=
fun {M} c j => c.cells j (c.heads j)
def Chal.VariableWL.step : (M : Chal.VariableWL.Machine) →
  Chal.VariableWL.Configuration M → Option (Chal.VariableWL.Configuration M) :=
fun M c =>
  Option.map
    (fun data =>
      { control := data.1, heads := fun j => c.heads j + (data.2 j).2.displacement,
        cells := fun j i => if i = c.heads j then (data.2 j).1 else c.cells j i })
    (M.transition c.control (Chal.VariableWL.scanned c))

fun M w x => Nat.brecOn x (Chal.VariableWL.run._f M w)
def Chal.VariableWL.OutputsWithin : Chal.VariableWL.Machine → Chal.VariableWL.Word → Chal.VariableWL.Word → ℕ → Prop :=
fun M w out bound =>
  ∃ t ≤ bound,
    Chal.VariableWL.step M (Chal.VariableWL.run M w t) = none ∧
      (Chal.VariableWL.run M w t).cells 0 = Chal.VariableWL.wordCells M.workSymbols out

def Chal.VariableWL.InEXPTIME : Chal.VariableWL.Language → Prop :=
fun L =>
  ∃ M c d,
    ∀ (w : Chal.VariableWL.Word),
      ∃ b, (b = true ↔ w ∈ L) ∧ Chal.VariableWL.OutputsWithin M w [b] (2 ^ (c * (List.length w + 1) ^ d))
def Chal.VariableWL.PolytimeReduces : Chal.VariableWL.Language → Chal.VariableWL.Language → Prop :=
fun L R =>
  ∃ M c d,
    ∀ (w : Chal.VariableWL.Word),
      ∃ out, Chal.VariableWL.OutputsWithin M w out (c * (List.length w + 1) ^ d) ∧ (w ∈ L ↔ out ∈ R)
def Chal.VariableWL.EXPTIMEComplete : Chal.VariableWL.Language → Prop :=
fun L =>
  Chal.VariableWL.InEXPTIME L ∧
    ∀ (R : Chal.VariableWL.Language), Chal.VariableWL.InEXPTIME R → Chal.VariableWL.PolytimeReduces R L

{w |
  ∃ G H k,
    w = Chal.VariableWL.pairCode G H k ∧
      2 ≤ k ∧
        0 < G.order ∧
          G.order = H.order ∧
            G.graph.Connected ∧
              H.graph.Connected ∧
