  Chal.FiniteCongruence.Algebra Carrier → Setoid Carrier → Prop :=
fun {Carrier} alg rel =>
  ∀ (index : Fin alg.numOps) (left right : Fin (alg.arity index) → Carrier),
    (∀ (coord : Fin (alg.arity index)), rel (left coord) (right coord)) → rel (alg.op index left) (alg.op index right)

  [LE Lat] → (Carrier → Carrier → Lat) → (Carrier → Carrier) → Prop :=
fun {Lat Carrier} [LE Lat] color placement =>
  ∀ (left right : Carrier), color (placement left) (placement right) ≤ color left right
def Chal.FiniteCongruence.Marked : {Lat Carrier : Type} →
  [LE Lat] → (Carrier → Carrier → Lat) → Finset (Carrier × Carrier) → Carrier → Carrier → Prop :=
fun {Lat Carrier} [LE Lat] color seeds left right =>
  ∃ placement,
    Chal.FiniteCongruence.Admissible color placement ∧
      ∃ edge ∈ seeds, placement edge.1 = left ∧ placement edge.2 = right
def Chal.FiniteCongruence.Connected : {Lat Carrier : Type} →
  [LE Lat] → (Carrier → Carrier → Lat) → Finset (Carrier × Carrier) → Carrier → Carrier → Prop :=
fun {Lat Carrier} [LE Lat] color seeds left right =>
  Relation.EqvGen (Chal.FiniteCongruence.Marked color seeds) left right
def Chal.FiniteCongruence.IsGraphWitness : {Lat Carrier : Type} →
  [inst : Lattice Lat] → [OrderBot Lat] → (Carrier → Carrier → Lat) → Prop :=
fun {Lat Carrier} [Lattice Lat] [OrderBot Lat] color =>
  (∀ (left right : Carrier), color left right = color right left) ∧
    (∀ (left right : Carrier), color left right = ⊥ ↔ left = right) ∧
      (∀ (left right middle : Carrier), color left right ≤ color left middle ⊔ color middle right) ∧
        (∀ (lower upper : Lat), ¬lower ≤ upper → ∃ left right, color left right ≤ lower ∧ ¬color left right ≤ upper) ∧
          ∀ (seeds : Finset (Carrier × Carrier)),
            seeds.Nonempty →
              ∀ (left right : Carrier),
                (color left right ≤ seeds.sup fun edge => color edge.1 edge.2) →
                  Chal.FiniteCongruence.Connected color seeds left right
def Chal.FiniteCongruence.HasGraphWitness : (Lat : Type) → [inst : Lattice Lat] → [OrderBot Lat] → Prop :=
fun Lat [Lattice Lat] [OrderBot Lat] => ∃ size, 0 < size ∧ ∃ color, Chal.FiniteCongruence.IsGraphWitness color
