def Chal.CommonBasesFPRAS.machineStep : {k s n : ℕ} →
  Chal.CommonBasesFPRAS.Program k s →
    Chal.CommonBasesFPRAS.IndependenceOracle n →
      Bool → Chal.CommonBasesFPRAS.MachineConfig k s → Chal.CommonBasesFPRAS.MachineConfig k s :=
fun {k s n} P O randomBit c =>
  if c.output.isSome = true then c
  else
    match P c.control with
    | Chal.CommonBasesFPRAS.Instruction.push i b next =>
      { registers := Function.update c.registers i (b :: c.registers i), control := next, output := c.output,
        oracleCalls := c.oracleCalls, bitOperations := c.bitOperations + 1 }
    | Chal.CommonBasesFPRAS.Instruction.pop i empty zero one =>
      match c.registers i with
      | [] =>
        { registers := c.registers, control := empty, output := c.output, oracleCalls := c.oracleCalls,
          bitOperations := c.bitOperations + 1 }
      | b :: l =>
        { registers := Function.update c.registers i l, control := if b = true then one else zero, output := c.output,
          oracleCalls := c.oracleCalls, bitOperations := c.bitOperations + 1 }
    | Chal.CommonBasesFPRAS.Instruction.coin zero one =>
      { registers := c.registers, control := if randomBit = true then one else zero, output := c.output,
        oracleCalls := c.oracleCalls, bitOperations := c.bitOperations + 1 }
    | Chal.CommonBasesFPRAS.Instruction.query which i no yes =>
      if (c.registers i).length = n then
        { registers := c.registers,
          control := if (O which fun j => (c.registers i)[↑j]?.getD false) = true then yes else no, output := c.output,
          oracleCalls := c.oracleCalls + 1, bitOperations := c.bitOperations + (c.registers i).length + n + 1 }
      else
        { registers := c.registers, control := c.control, output := some 0, oracleCalls := c.oracleCalls,
          bitOperations := c.bitOperations + (c.registers i).length + n + 1 }
    | Chal.CommonBasesFPRAS.Instruction.halt =>
      { registers := c.registers, control := c.control,
        output :=
          some
            (↑(Chal.CommonBasesFPRAS.binaryValue (c.registers ⟨6, ⋯⟩)) /
              ↑(Chal.CommonBasesFPRAS.binaryValue (c.registers ⟨7, ⋯⟩))),
        oracleCalls := c.oracleCalls,
        bitOperations := c.bitOperations + (c.registers ⟨6, ⋯⟩).length + (c.registers ⟨7, ⋯⟩).length + 1 }

def Chal.CommonBasesFPRAS.finiteBits : {B : ℕ} → (Fin B → Bool) → ℕ → Bool :=
fun {B} bits t => if h : t < B then bits ⟨t, h⟩ else false
def Chal.CommonBasesFPRAS.relativeAccuracy : ℚ → ℕ → ℚ → Prop :=
fun ε Z z => (1 - ε) * ↑Z ≤ z ∧ z ≤ (1 + ε) * ↑Z
def Chal.CommonBasesFPRAS.resourceBound : ℕ → ℕ → ℕ → ℕ → ℚ → ℚ → ℕ :=
fun C d n r ε δ => C * (1 + n + Chal.CommonBasesFPRAS.inputLength n r ε δ + ⌈ε⁻¹⌉₊ + Nat.clog 2 ⌈δ⁻¹⌉₊) ^ d
def Chal.CommonBasesFPRAS.MainTheorem : Prop :=
∃ k s P C d,
  0 < C ∧
    ∀ (n r : ℕ) (M₁ M₂ : Matroid (Fin n)) (O : Chal.CommonBasesFPRAS.IndependenceOracle n) (ε δ : ℚ),
      M₁.E = Set.univ →
        M₂.E = Set.univ →
          M₁.eRank = ↑r →
            M₂.eRank = ↑r →
              Chal.CommonBasesFPRAS.ExactOracles M₁ M₂ O →
                0 < ε →
                  ε < 1 →
                    0 < δ →
                      δ < 1 →
                        have B := Chal.CommonBasesFPRAS.resourceBound C d n r ε δ;
                        have Z := Chal.CommonBasesFPRAS.commonBaseCount M₁ M₂;
                        (∀ (bits : ℕ → Bool),
                            have out :=
                              Chal.CommonBasesFPRAS.execution P O (Chal.CommonBasesFPRAS.initialConfig k s n r ε δ) bits
                                B;
                            out.output.isSome = true ∧
                              out.oracleCalls ≤ B ∧
                                out.bitOperations ≤ B ∧ ∀ (z : ℚ), out.output = some z → 0 ≤ z ∧ (Z = 0 → z = 0)) ∧
                          (1 - δ) * 2 ^ B ≤
                            ↑(Nat.card
                                { bits //
                                  ∃ z,
                                    (Chal.CommonBasesFPRAS.execution P O
                                            (Chal.CommonBasesFPRAS.initialConfig k s n r ε δ)
                                            (Chal.CommonBasesFPRAS.finiteBits bits) B).output =
                                        some z ∧
                                      Chal.CommonBasesFPRAS.relativeAccuracy ε Z z })
