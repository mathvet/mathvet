Chal.NCHitting.universal_hitting : ∀ (n s : ℕ),
  1 ≤ n →
    1 ≤ s →
      ∀ (F : Type u_1) [inst : Field F] [CharZero F] (f : Chal.NCHitting.Formula F n),
        f.size ≤ s →
          (∃ w, f.coeff w ≠ 0) →
            Chal.NCHitting.Formula.eval (⇑(Matrix.scalar (Fin (Chal.NCHitting.dimension n s))))
                (fun i r q => ↑(Chal.NCHitting.generator n s i r q)) f ≠
              0
