def Chal.PowerSaving.DFTProgram : Chal.PowerSaving.RAM.Prog false Chal.PowerSaving.RAM.Ty.w Chal.PowerSaving.RAM.Ty.w →
  Chal.PowerSaving.RAM.Prog false Chal.PowerSaving.dftInKind Chal.PowerSaving.dftOutKind → (ℕ → ℕ) → Prop :=
fun order solve W =>
  ∃ cBound,
    ∀ (n : ℕ),
      0 < n →
        have cap := (n + 2) ^ cBound;
        have d := Chal.PowerSaving.RAM.run order n;
        W n + 10 * (n + 2) ≤ cap ∧
          d.peak ≤ cap ∧
            d.valid ∧
              0 < d.val ∧
                d.val < 1024 * n ^ 3 ∧
                  d.val ≤ cap ∧
                    ∀ (x : Chal.PowerSaving.Vec n),
                      have result :=
                        Chal.PowerSaving.RAM.run solve (n, Chal.PowerSaving.root d.val, { len := n, pos := x });
                      result.valid ∧
                        result.peak ≤ cap ∧
                          d.work + 1 + result.work ≤ W n ∧ result.val = { len := n, pos := Chal.PowerSaving.dft n x }

          d.peak ≤ cap ∧

                        Chal.PowerSaving.RAM.run solve

fun W =>
