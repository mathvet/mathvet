def Chal.WLIdentification.initial : (M : Chal.WLIdentification.Machine) →
  Chal.WLIdentification.Word → Chal.WLIdentification.Configuration M :=
fun M w =>
  { state := 0, head := 0,
    tape := fun z => if 0 ≤ z then w[z.toNat]?.elim 0 (Chal.WLIdentification.inputSymbol M) else 0 }
def Chal.WLIdentification.Halted : {M : Chal.WLIdentification.Machine} → Chal.WLIdentification.Configuration M → Prop :=
fun {M} c => c.state = 1 ∨ c.state = 2
def Chal.WLIdentification.step : (M : Chal.WLIdentification.Machine) →
  Chal.WLIdentification.Configuration M → Chal.WLIdentification.Configuration M :=
fun M c =>
  if c.state = 1 ∨ c.state = 2 then c
  else
    have t := M.transition c.state (c.tape c.head);
    { state := t.1, head := c.head + t.2.2.displacement, tape := Function.update c.tape c.head t.2.1 }
def Chal.WLIdentification.run : (M : Chal.WLIdentification.Machine) →
  Chal.WLIdentification.Word → ℕ → Chal.WLIdentification.Configuration M :=
fun M w t => (Chal.WLIdentification.step M)^[t] (Chal.WLIdentification.initial M w)
def Chal.WLIdentification.Output : {M : Chal.WLIdentification.Machine} →
  Chal.WLIdentification.Configuration M → Chal.WLIdentification.Word → Prop :=
fun {M} c w =>
  (∀ (i : Fin (List.length w)), c.tape ↑↑i = Chal.WLIdentification.inputSymbol M w[i]) ∧ c.tape ↑(List.length w) = 0
def Chal.WLIdentification.DecidesWithin : Chal.WLIdentification.Machine →
  Chal.WLIdentification.BinaryLanguage → (ℕ → ℕ) → Prop :=
fun M L bound =>
  ∀ (w : Chal.WLIdentification.Word),
    ∃ t ≤ bound (List.length w),
      Chal.WLIdentification.Halted (Chal.WLIdentification.run M w t) ∧
        ((Chal.WLIdentification.run M w t).state = 1 ↔ w ∈ L)
def Chal.WLIdentification.InEXPTIME : Chal.WLIdentification.BinaryLanguage → Prop :=
fun L => ∃ M C d, Chal.WLIdentification.DecidesWithin M L fun n => 2 ^ (C * (n + 1) ^ d)
def Chal.WLIdentification.ComputesWithin : Chal.WLIdentification.Machine →
  (Chal.WLIdentification.Word → Chal.WLIdentification.Word) → (ℕ → ℕ) → Prop :=
fun M f bound =>
  ∀ (w : Chal.WLIdentification.Word),
    ∃ t ≤ bound (List.length w),
      (Chal.WLIdentification.run M w t).state = 1 ∧ Chal.WLIdentification.Output (Chal.WLIdentification.run M w t) (f w)
def Chal.WLIdentification.PolytimeManyOne : Chal.WLIdentification.BinaryLanguage →
  Chal.WLIdentification.BinaryLanguage → Prop :=
fun L S =>
  ∃ f M C d,
    (Chal.WLIdentification.ComputesWithin M f fun n => C * (n + 1) ^ d) ∧
      ∀ (w : Chal.WLIdentification.Word), w ∈ L ↔ f w ∈ S
def Chal.WLIdentification.EXPTIMEComplete : Chal.WLIdentification.BinaryLanguage → Prop :=
fun S =>
  Chal.WLIdentification.InEXPTIME S ∧
    ∀ (L : Chal.WLIdentification.BinaryLanguage),
      Chal.WLIdentification.InEXPTIME L → Chal.WLIdentification.PolytimeManyOne L S
