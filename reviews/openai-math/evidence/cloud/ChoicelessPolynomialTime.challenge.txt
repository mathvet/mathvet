  OAI.CPTSeparation.FullCPT.Program R →

  P.active rel h ∪

      match P.run rel j with

def OAI.CPTSeparation.FullCPT.Program.evaluationAccepts : {A R : Type} →
  OAI.CPTSeparation.FullCPT.Program R → [Fintype A] → Polynomial ℕ → Polynomial ℕ → (R → A → A → Bool) → Prop :=
fun {A R} P [Fintype A] time space rel =>
  ∃ h st,
    h ≤ Polynomial.eval (Fintype.card A) time ∧
      P.run rel h = some st ∧
        P.flag st OAI.CPTSeparation.Operational.Control.halt ∧
          P.flag st OAI.CPTSeparation.Operational.Control.accept ∧
            (P.occurring rel h).card ≤ Polynomial.eval (Fintype.card A) space

  Prop :=
fun Q =>
  ∃ P time space,
    ∀ (A : Type) [inst : Fintype A] (I : OAI.CPTSeparation.Input A), P.evaluationAccepts time space I.rel ↔ Q I
