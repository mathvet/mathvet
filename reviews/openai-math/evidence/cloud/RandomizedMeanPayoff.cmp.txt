def Chal.MeanPayoff.Strategy : Chal.MeanPayoff.Game → Type :=
fun G => List (Fin G.m) → (v : Fin G.n) → { e // G.tail e = v }
def Chal.MeanPayoff.move : (G : Chal.MeanPayoff.Game) →
  Chal.MeanPayoff.Strategy G → Chal.MeanPayoff.Strategy G → Fin G.n → List (Fin G.m) → Fin G.m :=
fun G σ τ v history => if G.maxOwner v = true then ↑(σ history v) else ↑(τ history v)
def Chal.MeanPayoff.position : (G : Chal.MeanPayoff.Game) →
  Chal.MeanPayoff.Strategy G → Chal.MeanPayoff.Strategy G → Fin G.n → ℕ → Fin G.n × List (Fin G.m) :=
fun G σ τ start x => Nat.brecOn x (Chal.MeanPayoff.position._f G σ τ start)
def Chal.MeanPayoff.play : (G : Chal.MeanPayoff.Game) →
  Chal.MeanPayoff.Strategy G → Chal.MeanPayoff.Strategy G → Fin G.n → ℕ → Fin G.m :=
fun G σ τ start t =>
  have p := Chal.MeanPayoff.position G σ τ start t;
  Chal.MeanPayoff.move G σ τ p.1 p.2
def Chal.MeanPayoff.meanPayoff : (G : Chal.MeanPayoff.Game) → (ℕ → Fin G.m) → ℝ :=
fun G π => Filter.liminf (fun T => (∑ j ∈ Finset.range T, ↑(G.weight (π j))) / ↑T) Filter.atTop
def Chal.MeanPayoff.Winning : (G : Chal.MeanPayoff.Game) → Fin G.n → Prop :=
fun G v => ∃ σ, ∀ (τ : Chal.MeanPayoff.Strategy G), 0 ≤ Chal.MeanPayoff.meanPayoff G (Chal.MeanPayoff.play G σ τ v)
def Chal.MeanPayoff.encodeNat : ℕ → List Bool :=
fun n =>
  have bits := (n + 1).bits;
  List.replicate bits.length true ++ [false] ++ bits
def Chal.MeanPayoff.encodeInt : ℤ → List Bool :=

def Chal.MeanPayoff.initial : (M : Chal.MeanPayoff.Machine) → List Bool → Chal.MeanPayoff.Configuration M :=
fun M code =>
  { control := some M.start, head := fun x => 0, tape := fun k j => if k = 0 ∧ 0 ≤ j then code[j.toNat]? else none }
def Chal.MeanPayoff.step : (M : Chal.MeanPayoff.Machine) →
  Chal.MeanPayoff.Configuration M → Bool → Chal.MeanPayoff.Configuration M :=
fun M c bit =>
  match c.control with
  | none => c
  | some s =>
    match M.transition s (fun k => c.tape k (c.head k)) bit with
    | none => { control := none, head := c.head, tape := c.tape }
    | some a =>
      { control := some a.next, head := fun k => c.head k + (a.shift k).offset,
        tape := fun k => Function.update (c.tape k) (c.head k) (a.write k) }
def Chal.MeanPayoff.run : (M : Chal.MeanPayoff.Machine) →
  List Bool → (ℕ → Bool) → ℕ → Chal.MeanPayoff.Configuration M :=
fun M code random x => Nat.brecOn x (Chal.MeanPayoff.run._f M code random)
def Chal.MeanPayoff.padBits : {T : ℕ} → (Fin T → Bool) → ℕ → Bool :=
fun {T} bits j => if h : j < T then bits ⟨j, h⟩ else false
def Chal.MeanPayoff.result : (M : Chal.MeanPayoff.Machine) →
  Chal.MeanPayoff.Game → (T : ℕ) → (Fin T → Bool) → Chal.MeanPayoff.Configuration M :=
fun M G T bits => Chal.MeanPayoff.run M (Chal.MeanPayoff.input G) (Chal.MeanPayoff.padBits bits) T
def Chal.MeanPayoff.CorrectOutput : Chal.MeanPayoff.Game →
  {M : Chal.MeanPayoff.Machine} → Chal.MeanPayoff.Configuration M → Prop :=
fun G {M} c =>
  c.control = none ∧
    (∀ (v : Fin G.n), c.tape 2 ↑↑v = some true ↔ Chal.MeanPayoff.Winning G v) ∧
      (∀ (v : Fin G.n), c.tape 2 ↑↑v = some false ↔ ¬Chal.MeanPayoff.Winning G v) ∧
        ∀ (j : ℤ), j < 0 ∨ ↑G.n ≤ j → c.tape 2 j = none
def Chal.MeanPayoff.MainClaim : Prop :=
∃ M C,
  0 < C ∧
    ∀ (G : Chal.MeanPayoff.Game),
      ∃ T,
        ↑T ≤ 2 ^ (C * Real.logb 2 (↑(Chal.MeanPayoff.input G).length + 2) ^ 2) ∧
          (∀ (bits : Fin T → Bool), (Chal.MeanPayoff.result M G T bits).control = none) ∧
