def Chal.OrdinaryCorrelations.OneBounded : (ℕ → ℂ) → Prop :=

def Chal.OrdinaryCorrelations.Multiplicative : (ℕ → ℂ) → Prop :=
fun f => ∀ (m n : ℕ), 0 < m → 0 < n → m.Coprime n → f (m * n) = f m * f n
def Chal.OrdinaryCorrelations.distanceSq : (ℕ → ℂ) → {q : ℕ} → DirichletCharacter ℂ q → ℝ → ℝ → ℝ :=
