def Chal.CHObstruction.ClassData.Categorical.{u} : {L : ℕ → Type} →
  Chal.CHObstruction.ClassData L → Cardinal.{u} → Prop :=
fun {L} K μ =>
  (∃ M, K.objects M ∧ M.carrier.card = μ) ∧
    ∀ (M N : Chal.CHObstruction.RelModel L),
      K.objects M → K.objects N → M.carrier.card = μ → N.carrier.card = μ → Nonempty (M.Iso N)
def Chal.CHObstruction.ClassData.TwoModels.{u} : {L : ℕ → Type} →
  Chal.CHObstruction.ClassData L → Cardinal.{u} → Prop :=
fun {L} K μ => ∃ M N, K.objects M ∧ K.objects N ∧ M.carrier.card = μ ∧ N.carrier.card = μ ∧ ¬Nonempty (M.Iso N)
