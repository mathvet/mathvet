@Chal.BufferedIsing.finite_graph_comparison : ∀ {n m : ℕ} (G : Chal.BufferedIsing.FiniteGraph n m) (h : Fin n → ℝ)
  (I J F : Finset (Fin n)),
  Disjoint I J →
    Disjoint I F →
      Disjoint J F →
        ∀ (i j : Chal.BufferedIsing.KeptVertex F),
          ↑i ∈ I →
            ↑j ∈ J →
              ∀ (a τ : Chal.BufferedIsing.Spins n) (mix mix' : Chal.BufferedIsing.Spins n → ℝ),
                (∀ (b : Chal.BufferedIsing.Spins n), 0 ≤ mix b) →
                  (∀ (b : Chal.BufferedIsing.Spins n), 0 ≤ mix' b) →
                    ∑ b, mix b = 1 →
                      ∑ b, mix' b = 1 →
                        Real.exp (-4 * Real.artanh (Chal.BufferedIsing.qZero G I J F i j)) ≤
                            Chal.BufferedIsing.mixtureProb G h I J F a τ mix /
                              Chal.BufferedIsing.mixtureProb G h I J F a τ mix' ∧
                          Chal.BufferedIsing.mixtureProb G h I J F a τ mix /
                              Chal.BufferedIsing.mixtureProb G h I J F a τ mix' ≤
                            Real.exp (4 * Real.artanh (Chal.BufferedIsing.qZero G I J F i j))

  Chal.BufferedIsing.FiniteGraph n m → (Fin n → ℝ) → (Chal.BufferedIsing.Spins n → Prop) → ℝ :=

def Chal.BufferedIsing.conditionalProb : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m →
    (Fin n → ℝ) →
      Finset (Fin n) →
        Finset (Fin n) →
          Finset (Fin n) → Chal.BufferedIsing.Spins n → Chal.BufferedIsing.Spins n → Chal.BufferedIsing.Spins n → ℝ :=
fun {n m} G h I J F a b τ =>
  (Chal.BufferedIsing.constrainedSum G h fun σ =>
      Chal.BufferedIsing.AgreesOn I σ a ∧ Chal.BufferedIsing.AgreesOn J σ b ∧ Chal.BufferedIsing.AgreesOn F σ τ) /
    Chal.BufferedIsing.constrainedSum G h fun σ => Chal.BufferedIsing.AgreesOn J σ b ∧ Chal.BufferedIsing.AgreesOn F σ τ
def Chal.BufferedIsing.keptEdges : {n m : ℕ} → Chal.BufferedIsing.FiniteGraph n m → Finset (Fin n) → Finset (Fin m) :=
fun {n m} G F => {e | G.left e ∉ F ∧ G.right e ∉ F}
@[reducible] def Chal.BufferedIsing.KeptVertex : {n : ℕ} → Finset (Fin n) → Type :=
fun {n} F => { v // v ∉ F }
def Chal.BufferedIsing.comparisonRel : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m →
    Finset (Fin n) →
      Finset (Fin n) →
        (F : Finset (Fin n)) →
          Finset (Fin m) → Chal.BufferedIsing.KeptVertex F → Chal.BufferedIsing.KeptVertex F → Prop :=
fun {n m} G I J F ω u v =>
  ↑u ∈ I ∧ ↑v ∈ I ∨ ↑u ∈ J ∧ ↑v ∈ J ∨ ∃ e ∈ ω, G.left e = ↑u ∧ G.right e = ↑v ∨ G.left e = ↑v ∧ G.right e = ↑u
def Chal.BufferedIsing.comparisonSetoid : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m →
    Finset (Fin n) →
      Finset (Fin n) → (F : Finset (Fin n)) → Finset (Fin m) → Setoid (Chal.BufferedIsing.KeptVertex F) :=
fun {n m} G I J F ω => Relation.EqvGen.setoid (Chal.BufferedIsing.comparisonRel G I J F ω)
def Chal.BufferedIsing.clusterCount : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m → Finset (Fin n) → Finset (Fin n) → Finset (Fin n) → Finset (Fin m) → ℕ :=
fun {n m} G I J F ω => Nat.card (Quotient (Chal.BufferedIsing.comparisonSetoid G I J F ω))
def Chal.BufferedIsing.fkWeight : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m → Finset (Fin n) → Finset (Fin n) → Finset (Fin n) → Finset (Fin m) → ℝ :=
fun {n m} G I J F ω =>
  2 ^ Chal.BufferedIsing.clusterCount G I J F ω *
    ∏ e ∈ Chal.BufferedIsing.keptEdges G F,
      if e ∈ ω then 1 - Real.exp (-2 * G.coupling e) else Real.exp (-2 * G.coupling e)
def Chal.BufferedIsing.qZero : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m →
    Finset (Fin n) →
      Finset (Fin n) → (F : Finset (Fin n)) → Chal.BufferedIsing.KeptVertex F → Chal.BufferedIsing.KeptVertex F → ℝ :=
fun {n m} G I J F i j =>
  (∑ ω ∈ (Chal.BufferedIsing.keptEdges G F).powerset,
      if (Chal.BufferedIsing.comparisonSetoid G I J F ω) i j then Chal.BufferedIsing.fkWeight G I J F ω else 0) /
    ∑ ω ∈ (Chal.BufferedIsing.keptEdges G F).powerset, Chal.BufferedIsing.fkWeight G I J F ω
def Chal.BufferedIsing.mixtureProb : {n m : ℕ} →
  Chal.BufferedIsing.FiniteGraph n m →
    (Fin n → ℝ) →
      Finset (Fin n) →
        Finset (Fin n) →
          Finset (Fin n) →
            Chal.BufferedIsing.Spins n → Chal.BufferedIsing.Spins n → (Chal.BufferedIsing.Spins n → ℝ) → ℝ :=
fun {n m} G h I J F a τ mix => ∑ b, mix b * Chal.BufferedIsing.conditionalProb G h I J F a b τ
