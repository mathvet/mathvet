@[reducible] def Chal.Thorp.Position : ℕ → Type :=
fun d => Fin d → Bool
@[reducible] def Chal.Thorp.State : ℕ → Type :=
fun d => Equiv.Perm (Chal.Thorp.Position d)
def Chal.Thorp.Coins : ℕ → Type :=
fun x =>
  match x with
  | 0 => Unit
  | d.succ => Chal.Thorp.Position d → Bool
def Chal.Thorp.pairSwitch : (d : ℕ) → (Chal.Thorp.Position d → Bool) → Equiv.Perm (Bool × Chal.Thorp.Position d) :=
fun d c =>
  { toFun := fun p => (p.1 ^^ c p.2, p.2), invFun := fun p => (p.1 ^^ c p.2, p.2), left_inv := ⋯, right_inv := ⋯ }
def Chal.Thorp.rotate : (d : ℕ) → Chal.Thorp.State d :=
fun d => Equiv.piCongrLeft (fun x => Bool) (Equiv.symm (finRotate d))
def Chal.Thorp.step : (d : ℕ) → Chal.Thorp.Coins d → Chal.Thorp.State d :=
fun x x_1 =>
  match x, x_1 with
  | 0, x => Equiv.refl (Chal.Thorp.Position 0)
  | d.succ, c =>
    (((Fin.consEquiv fun x => Bool).symm.trans (Chal.Thorp.pairSwitch d c)).trans (Fin.consEquiv fun x => Bool)).trans
      (Chal.Thorp.rotate (d + 1))

fun d t => Fin t → Chal.Thorp.Coins d

fun d t ω => (List.ofFn fun i => Chal.Thorp.step d (ω i)).reverse.prod
def Chal.Thorp.fairMass.{u, v} : {Ω : Type u} → {α : Type v} → [Fintype Ω] → (Ω → α) → α → ℝ :=
fun {Ω} {α} [Fintype Ω] f a => (∑ ω, if f ω = a then 1 else 0) / ↑(Fintype.card Ω)

fun {α} [Fintype α] μ ν => 1 / 2 * ∑ a, |μ a - ν a|
