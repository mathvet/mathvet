def OAI.ShortEgyptian.IsExpansion : ℕ → ℕ → List ℕ → Prop :=
fun a b ns =>
  List.Pairwise (fun x1 x2 => x1 < x2) ns ∧
    (∀ n ∈ ns, 2 ≤ n) ∧
      (List.map (fun n => 1 / n) do
            let a ← ns
            pure ↑a).sum =
        ↑a / ↑b
def OAI.ShortEgyptian.minLength : ℕ → ℕ → ℕ :=
fun a b => sInf {k | ∃ ns, OAI.ShortEgyptian.IsExpansion a b ns ∧ ns.length = k}
def OAI.ShortEgyptian.maxMinLength : ℕ → ℕ :=
fun b => (Finset.Ico 1 b).sup fun a => OAI.ShortEgyptian.minLength a b
