def Chal.ThorpCompatibility.weightedCompatibility : (A D : ℕ) →
  (Fin A → Equiv.Perm (Fin D) → ℝ) → (Fin D → Equiv.Perm (Fin A) → ℝ) → ℝ :=
fun A D w v =>
  ↑(A * D).factorial / (↑D.factorial ^ A * ↑A.factorial ^ D) *
    Chal.ThorpCompatibility.uniformMean fun rc =>
      if Chal.ThorpCompatibility.Compatible rc.1 rc.2 then (∏ i, w i (rc.1 i)) * ∏ k, v k (rc.2 k) else 0
def Chal.ThorpCompatibility.WeightedCompatibilityTheorem : Prop :=
∃ L C₀ m₀,
  0 < L ∧
    0 < C₀ ∧
      0 < m₀ ∧
        ∀ (A D : ℕ),
          have n := ↑A * ↑D;
          have m := √n;
          have θ := 1 - L / Real.log m;
          m₀ ≤ m →
            0 < θ →
              m / ↑2 ≤ ↑A →
                ↑A ≤ ↑2 * m →
                  m / ↑2 ≤ ↑D →
                    ↑D ≤ ↑2 * m →
                      ∀ (w : Fin A → Equiv.Perm (Fin D) → ℝ) (v : Fin D → Equiv.Perm (Fin A) → ℝ),
                        (∀ (i : Fin A) (σ : Equiv.Perm (Fin D)), 0 ≤ w i σ) →
                          (∀ (k : Fin D) (σ : Equiv.Perm (Fin A)), 0 ≤ v k σ) →
