OAI.RadialTransition.densityInterval : ∃ p φ B C η βc f,
  0 < p ∧
    OAI.RadialTransition.IsPackingDensity p ∧
      OAI.RadialTransition.IsBoundedPotential φ ∧
        Continuous φ ∧
          0 < B ∧
            0 < C ∧
              OAI.RadialTransition.StableWith φ B ∧
                OAI.RadialTransition.AlgebraicDecayWith φ C ∧
                  0 < η ∧
                    Set.Ioo (5 * p / 3 - η) (5 * p / 3 + η) ⊆ Set.Ioi 0 ∧
                      βc ∈ Set.Icc (7 / 8) (9 / 8) ∧
                        ∀ ρ ∈ Set.Ioo (5 * p / 3 - η) (5 * p / 3 + η),
                          (OAI.RadialTransition.IsCanonicalFreeEnergy φ ρ fun β => f β ρ) ∧
                            OAI.RadialTransition.StrictTemperatureCorner (fun β => f β ρ) βc

@[reducible] def OAI.RadialTransition.Position : Type :=
EuclideanSpace ℝ (Fin 3)
@[reducible] def OAI.RadialTransition.Potential : Type :=
NNReal → ℝ
def OAI.RadialTransition.cube : ℝ → Set OAI.RadialTransition.Position :=
fun L => {x | ∀ (k : Fin 3), 0 ≤ x.ofLp k ∧ x.ofLp k ≤ L}
def OAI.RadialTransition.configurationCube : ℝ → (N : ℕ) → Set (Fin N → OAI.RadialTransition.Position) :=
fun L N => {x | ∀ (i : Fin N), x i ∈ OAI.RadialTransition.cube L}
def OAI.RadialTransition.energy : OAI.RadialTransition.Potential →
  {N : ℕ} → (Fin N → OAI.RadialTransition.Position) → ℝ :=
fun φ {N} x => ∑ i, ∑ j with i < j, φ (nndist (x i) (x j))
def OAI.RadialTransition.partition : OAI.RadialTransition.Potential → ℝ → ℕ → ℝ → ℝ :=
fun φ L N β =>
  (↑N.factorial)⁻¹ *
    ∫ (x : Fin N → OAI.RadialTransition.Position) in OAI.RadialTransition.configurationCube L N,
      Real.exp (-β * OAI.RadialTransition.energy φ x)
def OAI.RadialTransition.packingNumber : ℝ → ℕ :=
fun L =>
  sSup {n | ∃ X, X.card = n ∧ (∀ x ∈ X, x ∈ OAI.RadialTransition.cube L) ∧ ∀ x ∈ X, ∀ y ∈ X, x ≠ y → 1 ≤ dist x y}
def OAI.RadialTransition.IsPackingDensity : ℝ → Prop :=
fun p => Filter.Tendsto (fun L => ↑(OAI.RadialTransition.packingNumber L) / L ^ 3) Filter.atTop (nhds p)
def OAI.RadialTransition.IsBoundedPotential : OAI.RadialTransition.Potential → Prop :=
fun φ => ∃ M, 0 ≤ M ∧ ∀ (r : NNReal), |φ r| ≤ M
def OAI.RadialTransition.StableWith : OAI.RadialTransition.Potential → ℝ → Prop :=
fun φ B => ∀ (N : ℕ) (x : Fin N → OAI.RadialTransition.Position), -(B * ↑N) ≤ OAI.RadialTransition.energy φ x
def OAI.RadialTransition.AlgebraicDecayWith : OAI.RadialTransition.Potential → ℝ → Prop :=
fun φ C => ∀ (r : NNReal), 1 ≤ r → |φ r| ≤ C * ↑r ^ (-3 - 1 / 32)
def OAI.RadialTransition.IsCanonicalFreeEnergy : OAI.RadialTransition.Potential → ℝ → (ℝ → ℝ) → Prop :=
fun φ ρ f =>
  ∀ (β : ℝ),
    0 < β →
      Filter.Tendsto (fun L => -(Real.log (OAI.RadialTransition.partition φ L ⌊ρ * L ^ 3⌋₊ β) / (β * L ^ 3)))
        Filter.atTop (nhds (f β))
def OAI.RadialTransition.StrictTemperatureCorner : (ℝ → ℝ) → ℝ → Prop :=
fun f βc =>
  ∃ dLeft dRight,
    HasDerivWithinAt f dLeft (Set.Iic βc) βc ∧
      HasDerivWithinAt f dRight (Set.Ici βc) βc ∧ dRight < dLeft ∧ ¬DifferentiableAt ℝ f βc
