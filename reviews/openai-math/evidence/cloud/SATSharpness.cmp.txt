@[reducible] def Chal.RandomKSAT.Assignment : ℕ → Type :=
fun n => Fin n → Bool
def Chal.RandomKSAT.Clause : ℕ → ℕ → Type :=
fun n k => (s : { s // s.card = k }) × (↥↑s → Bool)
def Chal.RandomKSAT.Satisfies : {n k : ℕ} → Chal.RandomKSAT.Clause n k → Chal.RandomKSAT.Assignment n → Prop :=
fun {n k} c a => ∃ v, a ↑v = c.snd v
@[reducible] def Chal.RandomKSAT.Stream : ℕ → ℕ → Type :=
fun n k => ℕ → Chal.RandomKSAT.Clause n k
def Chal.RandomKSAT.clauseLaw : (n k : ℕ) → MeasureTheory.Measure (Chal.RandomKSAT.Clause n k) :=
fun n k => ProbabilityTheory.uniformOn Set.univ
def Chal.RandomKSAT.streamLaw : (n k : ℕ) → MeasureTheory.Measure (Chal.RandomKSAT.Stream n k) :=
fun n k => MeasureTheory.Measure.infinitePi fun x => Chal.RandomKSAT.clauseLaw n k
def Chal.RandomKSAT.SetSAT : {u s : ℕ} →
  Finset (Chal.RandomKSAT.Assignment u) → Chal.RandomKSAT.Stream u s → ℕ → Prop :=
fun {u s} S ω g => ∃ a ∈ S, ∀ i < g, Chal.RandomKSAT.Satisfies (ω i) a
def Chal.RandomKSAT.blockKill : (u : ℕ) → ℕ → Finset (Chal.RandomKSAT.Assignment u) → ℕ → ℝ :=
fun u s S g => ((Chal.RandomKSAT.streamLaw u s) {ω | ¬Chal.RandomKSAT.SetSAT S ω g}).toReal
def Chal.RandomKSAT.favg.{u_1} : {α : Type u_1} → [Fintype α] → (α → ℝ) → ℝ :=
fun {α} [Fintype α] f => (∑ a, f a) / ↑(Fintype.card α)
def Chal.RandomKSAT.clauseStep : (u : ℕ) →
  ℕ → (Finset (Chal.RandomKSAT.Assignment u) → ℝ) → Finset (Chal.RandomKSAT.Assignment u) → ℝ :=
fun u s f S => Chal.RandomKSAT.favg fun c => f (Finset.filter (Chal.RandomKSAT.Satisfies c) S)
def Chal.RandomKSAT.alive : (u : ℕ) → Finset (Chal.RandomKSAT.Assignment u) → ℝ :=
fun u S => if S.Nonempty then 1 else 0
def Chal.RandomKSAT.survival : (u : ℕ) → ℕ → Finset (Chal.RandomKSAT.Assignment u) → ℕ → ℝ :=
fun u s S m => (Chal.RandomKSAT.clauseStep u s)^[m] (Chal.RandomKSAT.alive u) S
def Chal.RandomKSAT.lifetime : (u : ℕ) → ℕ → Finset (Chal.RandomKSAT.Assignment u) → ℝ :=
fun u s S => ∑' (m : ℕ), Chal.RandomKSAT.survival u s S m
