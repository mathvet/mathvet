def Chal.FiniteFactor.HasScalarCenter.{u} : {H : Type u} →
  [inst : NormedAddCommGroup H] →
    [inst_1 : InnerProductSpace ℂ H] → [inst_2 : CompleteSpace H] → VonNeumannAlgebra H → Prop :=
fun {H} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H] M =>
  ∀ a ∈ M, (∀ b ∈ M, a * b = b * a) → ∃ z, a = z • 1
def Chal.FiniteFactor.IsFiniteAlgebra.{u} : {H : Type u} →
  [inst : NormedAddCommGroup H] →
    [inst_1 : InnerProductSpace ℂ H] → [inst_2 : CompleteSpace H] → VonNeumannAlgebra H → Prop :=
fun {H} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H] M => ∀ v ∈ M, star v * v = 1 → v * star v = 1
def Chal.FiniteFactor.HasNoMinimalProjection.{u} : {H : Type u} →
  [inst : NormedAddCommGroup H] →
    [inst_1 : InnerProductSpace ℂ H] → [inst_2 : CompleteSpace H] → VonNeumannAlgebra H → Prop :=
fun {H} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H] M =>
  ∀ p ∈ M, IsStarProjection p → p ≠ 0 → ∃ q ∈ M, IsStarProjection q ∧ q ≠ 0 ∧ q ≠ p ∧ q * p = q
def Chal.FiniteFactor.HasSeparablePredual.{u} : {H : Type u} →
  [inst : NormedAddCommGroup H] →
    [inst_1 : InnerProductSpace ℂ H] → [inst_2 : CompleteSpace H] → VonNeumannAlgebra H → Prop :=
fun {H} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H] M =>
  ∃ E x x_1,
    ∃ (_ : CompleteSpace E) (_ : TopologicalSpace.SeparableSpace E),
      Nonempty (StrongDual ℂ E ≃ₗᵢ⋆[ℂ] ↥M.toStarSubalgebra)
def Chal.FiniteFactor.MainTarget.{u} : Prop :=
∃ H x x_1,
  ∃ (x_2 : CompleteSpace H),
    ∃ M,
      Chal.FiniteFactor.HasScalarCenter M ∧
        Chal.FiniteFactor.IsFiniteAlgebra M ∧
          Chal.FiniteFactor.HasNoMinimalProjection M ∧
            Chal.FiniteFactor.HasSeparablePredual M ∧
              ∃ T ∈ M,
                T ≠ 0 ∧
                  Filter.Tendsto (fun n => ‖T ^ n‖ ^ (1 / ↑n)) Filter.atTop (nhds 0) ∧
                    ∀ p ∈ M, IsStarProjection p → (1 - p) * T * p = 0 → p = 0 ∨ p = 1
