def Chal.Zarankiewicz.ProperCrossing : Set Chal.Zarankiewicz.Plane →
  Set Chal.Zarankiewicz.Plane → Chal.Zarankiewicz.Plane → Prop :=
fun A B x =>
  ∃ chart,
    x ∈ chart.source ∧
      ↑chart x = (0, 0) ∧
        ↑chart '' (chart.source ∩ A) = chart.target ∩ {y | y.2 = 0} ∧
          ↑chart '' (chart.source ∩ B) = chart.target ∩ {y | y.1 = 0}

fun {m n} D => {x | ∃ e f, e ≠ f ∧ x ∈ Chal.Zarankiewicz.edgeInterior D e ∧ x ∈ Chal.Zarankiewicz.edgeInterior D f}

def Chal.Zarankiewicz.crossingCount : {m n : ℕ} → Chal.Zarankiewicz.AdmissibleDrawing m n → ℕ :=
fun {m n} D => ⋯.toFinset.card
def Chal.Zarankiewicz.MainTarget : Prop :=
∀ (m n : ℕ),
  0 < m →
    0 < n →
      (∃ D, Chal.Zarankiewicz.crossingCount D = Chal.Zarankiewicz.axisPairs m * Chal.Zarankiewicz.axisPairs n) ∧
        ∀ (D : Chal.Zarankiewicz.AdmissibleDrawing m n),
          Chal.Zarankiewicz.axisPairs m * Chal.Zarankiewicz.axisPairs n ≤ Chal.Zarankiewicz.crossingCount D
