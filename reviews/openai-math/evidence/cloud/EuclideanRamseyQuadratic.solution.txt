@[reducible] def OAI.EuclideanRamsey.Space : ℕ → Type :=
fun d => EuclideanSpace ℝ (Fin d)
def OAI.EuclideanRamsey.Congruent : {s d D : ℕ} →
  (Fin s → OAI.EuclideanRamsey.Space d) → (Fin s → OAI.EuclideanRamsey.Space D) → Prop :=
fun {s d D} a b => ∀ (i j : Fin s), dist (b i) (b j) = dist (a i) (a j)
def OAI.EuclideanRamsey.Ramsey : {s d : ℕ} → (Fin s → OAI.EuclideanRamsey.Space d) → Prop :=
fun {s d} a =>
  ∀ (r : ℕ),
    2 ≤ r →
      ∃ D,
        1 ≤ D ∧
          ∀ (c : OAI.EuclideanRamsey.Space D → Fin r),
            ∃ b, OAI.EuclideanRamsey.Congruent a b ∧ ∃ k, ∀ (i : Fin s), c (b i) = k
def OAI.EuclideanRamsey.coordinateField : {s d : ℕ} → (Fin s → OAI.EuclideanRamsey.Space d) → IntermediateField ℚ ℝ :=
fun {s d} a => IntermediateField.adjoin ℚ (Set.range fun ij => (a ij.1).ofLp ij.2)
@[reducible] def OAI.EuclideanRamsey.Coeff : {s d : ℕ} → (Fin s → OAI.EuclideanRamsey.Space d) → Type :=
fun {s d} a => ↥(OAI.EuclideanRamsey.coordinateField a)
def OAI.EuclideanRamsey.coordinate : {s d : ℕ} →
  (a : Fin s → OAI.EuclideanRamsey.Space d) → Fin s → Fin d → OAI.EuclideanRamsey.Coeff a :=
fun {s d} a i j => ⟨(a i).ofLp j, ⋯⟩
def OAI.EuclideanRamsey.augmented : {s d : ℕ} →
  (a : Fin s → OAI.EuclideanRamsey.Space d) → Fin s → Option (Fin d) → OAI.EuclideanRamsey.Coeff a :=
fun {s d} a i x =>
  match x with
  | none => 1
  | some j => OAI.EuclideanRamsey.coordinate a i j
