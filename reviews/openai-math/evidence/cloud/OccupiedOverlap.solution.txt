@[reducible] def OAI.RowColumn.OccupiedOverlapEndpoint : Prop :=
∃ C,
  0 < C ∧
    ∀ (m n : ℕ) (Ω : Finset (OAI.RowColumn.Board m n)) (a : YoungDiagram) (e : ↥Ω ≃ OAI.CubeShuffle.Specht.Cell a)
      (h : ℕ),
      1 ≤ h →
        h ≤ m * n →
          OAI.RowColumn.InHook a h →
            ∀ (r c u v : ℕ) (ρ : Representation ℂ (↥(OAI.RowColumn.rowGroup Ω)) (EuclideanSpace ℂ (Fin r)))
              (τ : Representation ℂ (↥(OAI.RowColumn.columnGroup Ω)) (EuclideanSpace ℂ (Fin c))),
              ρ.IsIrreducible →
                τ.IsIrreducible →
                  OAI.CubeShuffle.UnitaryFinite.IsUnitary ρ →
                    OAI.CubeShuffle.UnitaryFinite.IsUnitary τ →
                      ∀ (I : Fin u → EuclideanSpace ℂ (Fin r) →ₗᵢ[ℂ] ↥(OAI.CubeShuffle.Specht.hilbertSpace a))
                        (J : Fin v → EuclideanSpace ℂ (Fin c) →ₗᵢ[ℂ] ↥(OAI.CubeShuffle.Specht.hilbertSpace a)),
                        OAI.RowColumn.CompleteCopies ρ
                            (MonoidHom.comp (OAI.CubeShuffle.Specht.relabelledUnitary a e)
                              (OAI.RowColumn.rowGroup Ω).subtype)
                            I →
                          OAI.RowColumn.CompleteCopies τ
                              (MonoidHom.comp (OAI.CubeShuffle.Specht.relabelledUnitary a e)
                                (OAI.RowColumn.columnGroup Ω).subtype)
                              J →
                            OAI.RowColumn.allCopyOverlap I J ≤
                              Real.exp (C * (↑(m + n + 1) * ↑h ^ 2 * Real.log (↑(m * n) + 2) + ↑(m * n - Ω.card))) *
                                min 1 (↑r * ↑c / ↑(Module.finrank ℂ ↥(OAI.CubeShuffle.Specht.space a)))
