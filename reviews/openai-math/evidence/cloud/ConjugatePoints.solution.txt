def OAI.ThreeManifold.MainStatement : Prop :=
∃ M top,
  ∃ (_ : T2Space M) (_ : SecondCountableTopology M) (_ : CompactSpace M) (_ : ConnectedSpace M),
    ∃ charts,
      ∃ (smooth : IsManifold OAI.ThreeManifold.I3 (↑⊤) M),
        OAI.ThreeManifold.OrientedAtlas ∧
          (∃ g, OAI.ThreeManifold.NoConjugatePoints g) ∧ ¬∃ g, OAI.ThreeManifold.NonpositiveSectionalCurvature g
