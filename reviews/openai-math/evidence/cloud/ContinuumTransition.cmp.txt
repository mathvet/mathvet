Chal.ContinuumTemperature.main_theorem : Chal.ContinuumTemperature.MainStatement
@[reducible] def Chal.ContinuumTemperature.Space : Type :=
EuclideanSpace ℝ (Fin 3)
def Chal.ContinuumTemperature.radialPotential : (ℝ → ℝ) → Chal.ContinuumTemperature.Space → EReal :=
fun φ x => if x = 0 then ⊤ else ↑(φ ‖x‖)
def Chal.ContinuumTemperature.energy : (ℝ → ℝ) → {N : ℕ} → (Fin N → Chal.ContinuumTemperature.Space) → EReal :=
fun φ {N} x => ∑ i, ∑ j with i < j, Chal.ContinuumTemperature.radialPotential φ (x i - x j)
def Chal.ContinuumTemperature.Admissible : (ℝ → ℝ) → Prop :=
fun φ =>
  Measurable (Chal.ContinuumTemperature.radialPotential φ) ∧
    (∀ (a b : ℝ), 0 < a → a ≤ b → BddAbove (φ '' Set.Icc a b) ∧ BddBelow (φ '' Set.Icc a b)) ∧
      Filter.Tendsto φ (nhdsWithin 0 (Set.Ioi 0)) Filter.atTop ∧
        (∃ r₁ r₂ a, 0 < r₁ ∧ r₁ < r₂ ∧ 0 < a ∧ ∀ r ∈ Set.Icc r₁ r₂, φ r ≤ -a) ∧
          (∃ R, 0 < R ∧ MeasureTheory.IntegrableOn (fun r => r ^ 2 * |φ r|) (Set.Ioi R) MeasureTheory.volume) ∧
            ∃ B,
              ∀ (N : ℕ) (x : Fin N → Chal.ContinuumTemperature.Space), ↑(-B * ↑N) ≤ Chal.ContinuumTemperature.energy φ x
def Chal.ContinuumTemperature.cube : ℝ → Set Chal.ContinuumTemperature.Space :=
fun L => {x | ∀ (i : Fin 3), x.ofLp i ∈ Set.Icc (-L / 2) (L / 2)}
def Chal.ContinuumTemperature.boltzmann : (ℝ → ℝ) → ℝ → {N : ℕ} → (Fin N → Chal.ContinuumTemperature.Space) → ℝ :=
fun φ β {N} x =>
  if Chal.ContinuumTemperature.energy φ x = ⊤ then 0 else Real.exp (-β * (Chal.ContinuumTemperature.energy φ x).toReal)
def Chal.ContinuumTemperature.partition : (ℝ → ℝ) → ℝ → ℝ → ℕ → ℝ :=
fun φ β L N =>
  if N = 0 then 1
  else
    1 / ↑N.factorial *
      ∫ (x : Fin N → Chal.ContinuumTemperature.Space) in {x | ∀ (i : Fin N), x i ∈ Chal.ContinuumTemperature.cube L},
        Chal.ContinuumTemperature.boltzmann φ β x
def Chal.ContinuumTemperature.HasCanonicalFreeEnergy : (ℝ → ℝ) → (ℝ → ℝ → ℝ) → Prop :=
fun φ f =>
  ∀ (β ρ : ℝ),
    0 < β →
      0 < ρ →
        Filter.Tendsto (fun L => L⁻¹ ^ 3 * Real.log (Chal.ContinuumTemperature.partition φ β L ⌊ρ * L ^ 3⌋₊))
          Filter.atTop (nhds (-β * f β ρ))
def Chal.ContinuumTemperature.MainStatement : Prop :=
∃ φ f I βc,
  Chal.ContinuumTemperature.Admissible φ ∧
    (φ =o[Filter.atTop] fun r => r ^ (-3)) ∧
      Chal.ContinuumTemperature.HasCanonicalFreeEnergy φ f ∧
        IsOpen I ∧
          I.OrdConnected ∧
            I.Nonempty ∧
              I ⊆ Set.Ioi 0 ∧
                βc ∈ Set.Ioo (1 / 2) (3 / 2) ∧
                  ∀ ρ ∈ I,
                    ∃ dminus dplus,
                      HasDerivWithinAt (fun β => f β ρ) dminus (Set.Iic βc) βc ∧
                        HasDerivWithinAt (fun β => f β ρ) dplus (Set.Ici βc) βc ∧ dplus < dminus
