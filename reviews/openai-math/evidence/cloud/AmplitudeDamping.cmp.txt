Chal.GAD.main : ∀ (γ ν : ℝ),
  γ ∈ Set.Icc 0 1 →
    ν ∈ Set.Icc 0 1 →
      (∀ (n : ℕ), 1 ≤ n → Chal.GAD.holevo γ ν n = ↑n * Chal.GAD.holevo γ ν 1) ∧
        (∃ p,
            Chal.GAD.Maximizes γ ν p ∧
              ∀ (n : ℕ), 1 ≤ n → Chal.GAD.holevo γ ν n = ↑n / Real.log 2 * Chal.GAD.objective γ ν p) ∧
          Chal.GAD.capacity γ ν = Chal.GAD.holevo γ ν 1 ∧
            ∀ (p : ℝ),
              Chal.GAD.Maximizes γ ν p →
                ∀ (n : ℕ),
                  1 ≤ n →
                    (∀ (s : Chal.GAD.Basis n), Chal.GAD.IsState (Chal.GAD.phaseState p s)) ∧
                      Chal.GAD.holevo γ ν n = Chal.GAD.phaseValue γ ν p n
@[reducible] def Chal.GAD.Basis : ℕ → Type :=
fun n => Fin n → Fin 2
@[reducible] def Chal.GAD.QMatrix : ℕ → Type :=
fun n => Matrix (Chal.GAD.Basis n) (Chal.GAD.Basis n) ℂ
def Chal.GAD.entropy.{u} : {ι : Type u} → [Fintype ι] → [DecidableEq ι] → Matrix ι ι ℂ → ℝ :=
fun {ι} [Fintype ι] [DecidableEq ι] P => (cfc Real.negMulLog P).trace.re
def Chal.GAD.IsState.{u} : {ι : Type u} → [Fintype ι] → [DecidableEq ι] → Matrix ι ι ℂ → Prop :=
fun {ι} [Fintype ι] [DecidableEq ι] P => P.PosSemidef ∧ P.trace = 1
def Chal.GAD.kraus : ℝ → ℝ → Fin 4 → Matrix (Fin 2) (Fin 2) ℂ :=
fun γ ν r =>
  if r = 0 then !![↑√(1 - ν), 0; 0, ↑√(1 - ν) * ↑√(1 - γ)]
  else
    if r = 1 then !![0, ↑√(γ * (1 - ν)); 0, 0]
    else if r = 2 then !![↑√ν * ↑√(1 - γ), 0; 0, ↑√ν] else !![0, 0; ↑√(γ * ν), 0]
def Chal.GAD.tensorKraus : ℝ → ℝ → (n : ℕ) → (Fin n → Fin 4) → Chal.GAD.QMatrix n :=
fun γ ν n r i j => ∏ k, Chal.GAD.kraus γ ν (r k) (i k) (j k)
def Chal.GAD.channel : ℝ → ℝ → (n : ℕ) → Chal.GAD.QMatrix n → Chal.GAD.QMatrix n :=
fun γ ν n P => ∑ r, Chal.GAD.tensorKraus γ ν n r * P * Matrix.conjTranspose (Chal.GAD.tensorKraus γ ν n r)

def Chal.GAD.ensembleValue : ℝ → ℝ → {n : ℕ} → Chal.GAD.Ensemble n → ℝ :=
fun γ ν {n} E =>
  (Chal.GAD.entropy (∑ a, ↑(E.weight a) • Chal.GAD.channel γ ν n (E.signal a)) -
      ∑ a, E.weight a * Chal.GAD.entropy (Chal.GAD.channel γ ν n (E.signal a))) /
    Real.log 2
def Chal.GAD.holevo : ℝ → ℝ → ℕ → ℝ :=
fun γ ν n => sSup (Set.range (Chal.GAD.ensembleValue γ ν))

def Chal.GAD.averageError : ℝ → ℝ → {n : ℕ} → Chal.GAD.Code n → ℝ :=
fun γ ν {n} C => 1 - (∑ a, (C.decoding a * Chal.GAD.channel γ ν n (C.encoding a)).trace.re) / ↑C.messages
def Chal.GAD.Achievable : ℝ → ℝ → ℝ → Prop :=
fun γ ν R =>
  ∃ C,
    Filter.Tendsto (fun n => Chal.GAD.averageError γ ν (C n)) Filter.atTop (nhds 0) ∧
      ∀ (ε : ℝ), 0 < ε → ∀ᶠ (n : ℕ) in Filter.atTop, R - ε ≤ Real.log ↑(C n).messages / Real.log 2 / ↑n
def Chal.GAD.capacity : ℝ → ℝ → ℝ :=
fun γ ν => sSup {R | Chal.GAD.Achievable γ ν R}
def Chal.GAD.g : ℝ → ℝ :=
fun u => Real.binEntropy ((1 + √(1 - 4 * u)) / 2)
def Chal.GAD.v : ℝ → ℝ → ℝ → ℝ :=
fun γ ν p => γ * ν * (1 - ν) + γ * (1 - γ) * (p - ν) ^ 2
def Chal.GAD.objective : ℝ → ℝ → ℝ → ℝ :=
fun γ ν p => Real.binEntropy ((1 - γ) * p + γ * ν) - Chal.GAD.g (Chal.GAD.v γ ν p)
def Chal.GAD.Maximizes : ℝ → ℝ → ℝ → Prop :=
fun γ ν p => p ∈ Set.Icc 0 1 ∧ ∀ q ∈ Set.Icc 0 1, Chal.GAD.objective γ ν q ≤ Chal.GAD.objective γ ν p
def Chal.GAD.phaseSignal : ℝ → Fin 2 → Fin 2 → ℂ :=
fun p s b => if b = 0 then ↑√(1 - p) else if s = 0 then ↑√p else -↑√p
def Chal.GAD.phaseProduct : ℝ → {n : ℕ} → Chal.GAD.Basis n → Chal.GAD.Basis n → ℂ :=
fun p {n} s i => ∏ k, Chal.GAD.phaseSignal p (s k) (i k)
def Chal.GAD.phaseState : ℝ → {n : ℕ} → Chal.GAD.Basis n → Chal.GAD.QMatrix n :=
fun p {n} s i j => Chal.GAD.phaseProduct p s i * star (Chal.GAD.phaseProduct p s j)
def Chal.GAD.phaseValue : ℝ → ℝ → ℝ → ℕ → ℝ :=
fun γ ν p n =>
  (Chal.GAD.entropy (∑ s, (↑2 ^ n)⁻¹ • Chal.GAD.channel γ ν n (Chal.GAD.phaseState p s)) -
      ∑ s, (2 ^ n)⁻¹ * Chal.GAD.entropy (Chal.GAD.channel γ ν n (Chal.GAD.phaseState p s))) /
    Real.log 2
