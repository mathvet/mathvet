def OAI.IndependentProducts.CofiniteSubspaces.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Set (Submodule ℝ X) :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] => {F | IsClosed ↑F ∧ FiniteDimensional ℝ (X ⧸ F)}
def OAI.IndependentProducts.averagedMidpointModulus.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Seminorm ℝ X → ℝ → ℝ :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N t =>
  sInf
    {r |
      ∃ x,
        N x = 1 ∧
          r =
            sSup
              {a |
                ∃ F ∈ OAI.IndependentProducts.CofiniteSubspaces X,
                  a = sInf {b | ∃ y ∈ F, 1 ≤ N y ∧ b = (N (x + t • y) + N (x - t • y)) / 2 - 1}}}
def OAI.IndependentProducts.oneSidedModulus.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Seminorm ℝ X → ℝ → ℝ :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N t =>
  sInf
    {r |
      ∃ x,
        N x = 1 ∧
          r =
            sSup
              {a |
                ∃ F ∈ OAI.IndependentProducts.CofiniteSubspaces X,
                  a = sInf {b | ∃ y ∈ F, N y = 1 ∧ b = N (x + t • y) - 1}}}
def OAI.IndependentProducts.IsEquivalentNorm.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Seminorm ℝ X → Prop :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N => ∃ a b, 0 < a ∧ a ≤ b ∧ ∀ (x : X), a * ‖x‖ ≤ N x ∧ N x ≤ b * ‖x‖
def OAI.IndependentProducts.IsAUC.{uX} : (X : Type uX) →
  [inst : NormedAddCommGroup X] → [inst_1 : NormedSpace ℝ X] → Seminorm ℝ X → Prop :=
fun X [NormedAddCommGroup X] [NormedSpace ℝ X] N => ∀ (t : ℝ), 0 < t → 0 < OAI.IndependentProducts.oneSidedModulus X N t
def OAI.IndependentProducts.gaussianTail : ℝ → ℝ :=
fun r => ((ProbabilityTheory.gaussianReal 0 1) {x | r ≤ |x|}).toReal
def OAI.IndependentProducts.excessConstant : ℝ → ℝ :=
fun K => 1 / 16 * ∫ (x : ℝ), max (|x| - 2 * K) 0 ∂ProbabilityTheory.gaussianReal 0 1
def OAI.IndependentProducts.exponentialLaw : MeasureTheory.Measure ℝ :=
ProbabilityTheory.expMeasure 1
def OAI.IndependentProducts.gaussianSquareLaw : MeasureTheory.Measure ℝ :=
MeasureTheory.Measure.map (fun x => x ^ 2) (ProbabilityTheory.gaussianReal 0 1)
