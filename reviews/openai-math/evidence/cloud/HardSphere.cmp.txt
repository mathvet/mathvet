Chal.HardSphere.condensation_with_mixed : ∃ ε₀ c₀,
  0 < ε₀ ∧
    0 < c₀ ∧
      ∀ (a ρ : ℝ),
        0 < a →
          0 < ρ →
            ρ * a ^ 3 < ε₀ →
              ∀ (N : ℕ → ℕ) (L : ℕ → Chal.HardSphere.Length) (hN : ∀ (k : ℕ), 0 < N k),
                Tendsto N atTop atTop →
                  Tendsto (fun k => ↑(L k)) atTop atTop →
                    Tendsto (fun k => ↑(N k) / ↑(L k) ^ 3) atTop (𝓝 ρ) →
                      (∀ (Ψ : (k : ℕ) → Chal.HardSphere.Jet (N k) (L k)),
                          (∀ (k : ℕ), Chal.HardSphere.IsGroundVector (N k) (L k) a (Ψ k)) →
                            c₀ ≤ liminf (fun k => Chal.HardSphere.occupation ⋯ (Ψ k).1) atTop) ∧
                        ∀ (T : (k : ℕ) → Chal.HardSphere.Density.Operator (Chal.HardSphere.ScalarL2 (N k) (L k))),
                          (∀ (k : ℕ), Chal.HardSphere.GroundSupported a (T k)) →
                            c₀ ≤ liminf (fun k => Chal.HardSphere.mixedOccupation ⋯ (T k)) atTop

def Chal.HardSphere.torusDistance : (L : Chal.HardSphere.Length) →

    (Chal.HardSphere.Configuration N L → ℂ) → Chal.HardSphere.Direction N → Chal.HardSphere.Configuration N L → ℂ :=
