def OAI.DiscreteFree.blockBound : ℕ → ℝ :=
fun p => 300 * ↑p ^ 2 + 150 * ↑p + 5
def OAI.DiscreteFree.BlockStatement : Prop :=
∀ (p : ℕ),
  1 ≤ p →
    ∃ M d o,
      Countable M ∧
        (∀ (x y : M), x ≠ y → 1 ≤ dist x y) ∧
          (∀ (x y : M), dist x y ≤ OAI.DiscreteFree.blockBound p) ∧
            ∃ A,
              o ∈ A ∧
                ∀ (T : OAI.DiscreteFree.FreeSpace o →L[ℝ] OAI.DiscreteFree.FreeSpace o),
                  Module.Finite ℝ ↥(↑T).range →
                    ‖T‖ ≤ ↑p → ∃ a ∈ A, 1 / 2 ≤ ‖T (OAI.DiscreteFree.delta o a) - OAI.DiscreteFree.delta o a‖
def OAI.DiscreteFree.MainStatement : Prop :=
∃ M d o,
  Countable M ∧
    (∀ (x y : M), x ≠ y → 1 ≤ dist x y) ∧
      ¬Bornology.IsBounded Set.univ ∧
        ¬ProperSpace M ∧
          OAI.DiscreteFree.HasAP (OAI.DiscreteFree.FreeSpace o) ∧
            ∀ (Λ : ℝ), 1 ≤ Λ → ¬OAI.DiscreteFree.HasBAP (OAI.DiscreteFree.FreeSpace o) Λ
