def Chal.WitnessedChoice.BGS.WitnessesHistory : {A : Type} →
  [Fintype A] →
    {n : ℕ} →
      Chal.WitnessedSeparation.Input A →
        Finset (Fin n) →
          (Fin n → Chal.WitnessedSeparation.Hereditary.HF A) →
            (ℕ → Chal.WitnessedSeparation.Hereditary.HF A) →
              ℕ → Chal.WitnessedSeparation.Hereditary.HF A → Chal.WitnessedSeparation.Hereditary.HF A → Prop :=
fun {A} [Fintype A] {n} S free env b i D M =>
  Chal.WitnessedSeparation.Hereditary.isSet M = true ∧
    (∀ w ∈ Chal.WitnessedSeparation.Hereditary.elements M,
        ∃ g,
          w = Chal.WitnessedChoice.BGS.permutationGraph g ∧
            Chal.WitnessedChoice.BGS.Automorphism S g ∧
              Chal.WitnessedChoice.BGS.FixesParameters free env g ∧ ∀ j ≤ i, g • b j = b j) ∧
      ∀ u ∈ Chal.WitnessedSeparation.Hereditary.elements D,
        ∀ v ∈ Chal.WitnessedSeparation.Hereditary.elements D,
          ∃ g, Chal.WitnessedChoice.BGS.permutationGraph g ∈ Chal.WitnessedSeparation.Hereditary.elements M ∧ g • u = v
def Chal.WitnessedChoice.BGS.GoodWSC : {A : Type} →
  [Fintype A] →
    {n : ℕ} →
      Chal.WitnessedSeparation.Input A →
        ℕ →
          Finset (Fin n) →
            (Fin n → Chal.WitnessedSeparation.Hereditary.HF A) →
              (Chal.WitnessedSeparation.Hereditary.HF A →
                  Chal.WitnessedSeparation.Hereditary.HF A →
                    Chal.WitnessedChoice.BGS.Result (Chal.WitnessedSeparation.Hereditary.HF A)) →
                (Chal.WitnessedSeparation.Hereditary.HF A →
                    Chal.WitnessedChoice.BGS.Result (Chal.WitnessedSeparation.Hereditary.HF A)) →
                  (Chal.WitnessedSeparation.Hereditary.HF A →
                      Chal.WitnessedSeparation.Hereditary.HF A →
                        Chal.WitnessedChoice.BGS.Result (Chal.WitnessedSeparation.Hereditary.HF A)) →
                    (Chal.WitnessedSeparation.Hereditary.HF A → Chal.WitnessedChoice.BGS.Result Bool) → Prop :=
fun {A} [Fintype A] {n} S p free env step choice witness output =>
  (∀ (b c : ℕ → Chal.WitnessedSeparation.Hereditary.HF A) (k : ℕ),
      Chal.WitnessedChoice.BGS.ChoicePrefix step choice b c k →
        Chal.WitnessedChoice.BGS.StrictPrefix b k →
          k ≤ p ∧
            (Chal.WitnessedChoice.BGS.TC (b k)).card ≤ p ∧
              ∃ D,
                choice (b k) = some D ∧
                  (Chal.WitnessedSeparation.Hereditary.elements D).Nonempty ∧
                    ∀ u ∈ Chal.WitnessedSeparation.Hereditary.elements D, Option.isSome (step (b k) u) = true) ∧
    ∀ (b c : ℕ → Chal.WitnessedSeparation.Hereditary.HF A) (t : ℕ),
      Chal.WitnessedChoice.BGS.FirstPath step choice b c t →
        t + 2 ≤ p ∧
          (∀ i ≤ t,
              ∃ D M,
                choice (b i) = some D ∧
                  witness (b t) (b i) = some M ∧ Chal.WitnessedChoice.BGS.WitnessesHistory S free env b i D M) ∧
            Option.isSome (output (b t)) = true
def Chal.WitnessedChoice.BGS.witnessedIteration : {A : Type} →
  [Fintype A] →
    {n : ℕ} →
      Chal.WitnessedSeparation.Input A →
        ℕ →
          Finset (Fin n) →
            (Fin n → Chal.WitnessedSeparation.Hereditary.HF A) →
              (Chal.WitnessedSeparation.Hereditary.HF A →
                  Chal.WitnessedSeparation.Hereditary.HF A →
                    Chal.WitnessedChoice.BGS.Result (Chal.WitnessedSeparation.Hereditary.HF A)) →
                (Chal.WitnessedSeparation.Hereditary.HF A →
                    Chal.WitnessedChoice.BGS.Result (Chal.WitnessedSeparation.Hereditary.HF A)) →
                  (Chal.WitnessedSeparation.Hereditary.HF A →
                      Chal.WitnessedSeparation.Hereditary.HF A →
                        Chal.WitnessedChoice.BGS.Result (Chal.WitnessedSeparation.Hereditary.HF A)) →
                    (Chal.WitnessedSeparation.Hereditary.HF A → Chal.WitnessedChoice.BGS.Result Bool) →
                      Chal.WitnessedChoice.BGS.Result Bool :=
fun {A} [Fintype A] {n} S p free env step choice witness output =>
  if Chal.WitnessedChoice.BGS.GoodWSC S p free env step choice witness output then
    some
      (decide
        (∀ (b c : ℕ → Chal.WitnessedSeparation.Hereditary.HF A) (t : ℕ),
          Chal.WitnessedChoice.BGS.FirstPath step choice b c t → output (b t) = some true))
  else none
