Chal.OneWayLiveness.main_theorem : ∀ (h : ℕ),
  2 ≤ h →
    (∃ N, N.NoLeft ∧ ∀ (positive : Bool), N.Recognizes positive (Chal.OneWayLiveness.OWL h)) ∧
      ∀ (positive : Bool) (s : ℕ) (D : Chal.OneWayLiveness.DMachine (Chal.OneWayLiveness.Alphabet h) s),
        D.Recognizes positive (Chal.OneWayLiveness.OWL h) →
          2 ^ ((h - 2) / 31) ≤ 4 * (s + if positive = true then 2 else 1) ^ 2
