Chal.LipschitzHilbert.uniform_truncation : ∃ C,
  0 < C ∧
    ∀ (K : NNReal),
      0 < K →
        ∀ (epsilon R : ℝ),
          0 < epsilon →
            epsilon ≤ R →
              R ≤ 1 / (1000000 * ↑K) →
                ∀ (v : Chal.LipschitzHilbert.Plane → Chal.LipschitzHilbert.Plane),
                  LipschitzWith K v →
                    (∀ (x : Chal.LipschitzHilbert.Plane), ‖v x‖ = 1) →
                      ∀ (f : SchwartzMap Chal.LipschitzHilbert.Plane ℂ),
                        MeasureTheory.eLpNorm (Chal.LipschitzHilbert.shortTransform v epsilon R ⇑f) 2
                            MeasureTheory.volume ≤
                          ENNReal.ofReal (C * ‖f.toLp 2 MeasureTheory.volume‖)
