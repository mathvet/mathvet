  Chal.CKSRound.Idx → Chal.CKSRound.Idx → Chal.CKSRound.Idx → Chal.CKSRound.Idx → ℝ :=

  (∑ l,
      (j.dInv s a l * (j.d i k l + j.d k i l - j.d l i k) + j.inv a l * (j.dd s i k l + j.dd s k i l - j.dd s l i k))) /
