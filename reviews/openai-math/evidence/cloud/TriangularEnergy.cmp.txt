Chal.AtomicTriangular.universal_energy_minimum : ∀ (g : ℝ → ℝ) (C : Set Chal.AtomicTriangular.Plane),
  Chal.AtomicTriangular.AdmissiblePotential g →
    Chal.AtomicTriangular.LocallyFinite C →
      Chal.AtomicTriangular.DensityOne C →
        Chal.AtomicTriangular.latticeEnergy g ≤ Chal.AtomicTriangular.energy g C ∧
          Chal.AtomicTriangular.latticeEnergy g = Chal.AtomicTriangular.energy g Chal.AtomicTriangular.A
@[reducible] def Chal.TriangularUniversal.Plane : Type :=
EuclideanSpace ℝ (Fin 2)
def Chal.TriangularUniversal.LocallyFinite : Set Chal.TriangularUniversal.Plane → Prop :=
fun C => ∀ (R : ℝ), (C ∩ Metric.closedBall 0 R).Finite
def Chal.TriangularUniversal.diskPoints : Set Chal.TriangularUniversal.Plane →
  ℝ → Finset Chal.TriangularUniversal.Plane :=
fun C R => if h : (C ∩ Metric.closedBall 0 R).Finite then h.toFinset else ∅
def Chal.TriangularUniversal.diskCount : Set Chal.TriangularUniversal.Plane → ℝ → ℕ :=
fun C R => (Chal.TriangularUniversal.diskPoints C R).card
def Chal.TriangularUniversal.DensityOne : Set Chal.TriangularUniversal.Plane → Prop :=
fun C => Filter.Tendsto (fun R => ↑(Chal.TriangularUniversal.diskCount C R) / (Real.pi * R ^ 2)) Filter.atTop (nhds 1)
def Chal.TriangularUniversal.AdmissiblePotential : (ℝ → ℝ) → Prop :=
fun g =>
  ContDiffOn ℝ (↑⊤) g (Set.Ioi 0) ∧
    (∀ (t : ℝ), 0 < t → 0 ≤ g t) ∧ ∀ (r : ℕ) (t : ℝ), 0 < t → 0 ≤ (-1) ^ r * iteratedDeriv r g t
def Chal.TriangularUniversal.diskEnergy : (ℝ → ℝ) → Set Chal.TriangularUniversal.Plane → ℝ → ENNReal :=
fun g C R =>
  (↑(Chal.TriangularUniversal.diskCount C R))⁻¹ *
    ∑ x ∈ Chal.TriangularUniversal.diskPoints C R,
      ∑ y ∈ (Chal.TriangularUniversal.diskPoints C R).erase x, ENNReal.ofReal (g (‖x - y‖ ^ 2))
def Chal.TriangularUniversal.energy : (ℝ → ℝ) → Set Chal.TriangularUniversal.Plane → ENNReal :=
fun g C => Filter.liminf (Chal.TriangularUniversal.diskEnergy g C) Filter.atTop
