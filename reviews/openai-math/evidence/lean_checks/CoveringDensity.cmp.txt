def Chal.TranslativeCovering.ConvexBody : {n : ℕ} → Set (Chal.TranslativeCovering.Space n) → Prop :=
fun {n} K => IsCompact K ∧ Convex ℝ K ∧ (interior K).Nonempty
def Chal.TranslativeCovering.CentrallySymmetric : {n : ℕ} → Set (Chal.TranslativeCovering.Space n) → Prop :=
fun {n} K => ∃ z, ∀ (x : Chal.TranslativeCovering.Space n), x ∈ K ↔ 2 • z - x ∈ K

fun {n} X =>

def Chal.CoveringOrder.OptimalOrder : Prop :=

  0 < c ∧
