def Chal.MatchingFPRAS.tick : (A : Chal.MatchingFPRAS.RandomMachine) →
  Chal.MatchingFPRAS.Configuration A → Bool → Chal.MatchingFPRAS.Configuration A :=
fun A c bit =>
  match c.state with
  | none => c
  | some s =>
    match A.transition s c.tape.head bit with
    | none => { state := none, tape := c.tape }
    | some (s', action) =>
      { state := some s',
        tape :=
          match action with
          | Turing.TM0.Stmt.move d => Turing.Tape.move d c.tape
          | Turing.TM0.Stmt.write a => Turing.Tape.write a c.tape }

def Chal.ContingencyTables.Algorithms.BoundedSamplingStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (m n : ℕ) (r : Fin m → ℕ) (c : Fin n → ℕ),
      ∑ i, r i = ∑ j, c j →
        ∀ (k : ℕ),
          1 ≤ k →
            have input := Chal.ContingencyTables.Algorithms.encodeSamplingInput r c k;
            have t := C * Chal.ContingencyTables.Algorithms.samplingSize n r k ^ d;
            (∀ (bits : Fin t → Bool), ∃ X, Chal.ContingencyTables.Algorithms.OutputsTable A input bits ↑X) ∧
              (∑ X,
                    |Chal.ContingencyTables.Algorithms.tableMass A input t ↑X -
                        (↑(Fintype.card (Chal.ContingencyTables.Table r c)))⁻¹|) /
                  2 ≤
                (2 ^ k)⁻¹
def Chal.ContingencyTables.Algorithms.ExactSamplingStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (m n : ℕ) (r : Fin m → ℕ) (c : Fin n → ℕ),
      ∑ i, r i = ∑ j, c j →
        have input := Chal.ContingencyTables.Algorithms.encodeMargins r c;
        have tail := fun t => 1 - Chal.ContingencyTables.Algorithms.haltMass A input t;
        (∀ (t : ℕ) (bits : Fin t → Bool),
            (Chal.MatchingFPRAS.run A input bits).state = none →
              ∃ X, Chal.ContingencyTables.Algorithms.OutputsTable A input bits ↑X) ∧
          (∀ (X : Chal.ContingencyTables.Table r c),
              Filter.Tendsto (fun t => Chal.ContingencyTables.Algorithms.tableMass A input t ↑X) Filter.atTop
                (nhds (↑(Fintype.card (Chal.ContingencyTables.Table r c)))⁻¹)) ∧
            Filter.Tendsto (Chal.ContingencyTables.Algorithms.haltMass A input) Filter.atTop (nhds 1) ∧
              Summable tail ∧ tsum tail ≤ ↑C * ↑(Chal.ContingencyTables.Algorithms.samplingSize n r 0) ^ d

def Chal.ContingencyTables.Algorithms.countingTime : ℕ →

fun C d {m n} r c b ε δ =>

def Chal.ContingencyTables.Algorithms.goodCountingTapes : Chal.ContingencyTables.Algorithms.Machine →
  {m n : ℕ} → (Fin m → ℕ) → (Fin n → ℕ) → (Fin m → Fin n → ℕ) → ℚ → ℚ → (t : ℕ) → Finset (Fin t → Bool) :=
fun A {m n} r c b ε δ t =>
  {bits |
    ∃ q,
      Chal.MatchingFPRAS.Outputs A (Chal.ContingencyTables.Algorithms.encodeCountingInput r c b ε δ) bits q ∧
        (1 - ε) * ↑(Chal.ContingencyTables.Algorithms.count r c b) ≤ q ∧
          q ≤ (1 + ε) * ↑(Chal.ContingencyTables.Algorithms.count r c b)}
def Chal.ContingencyTables.Algorithms.CountingStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (m n : ℕ),
      0 < m →
        0 < n →
          ∀ (r : Fin m → ℕ) (c : Fin n → ℕ) (b : Fin m → Fin n → ℕ),
            ∑ i, r i = ∑ j, c j →
              ∀ (ε δ : ℚ),
                0 < ε →
                  ε < 1 →
                    0 < δ →
                      δ < 1 →
                        have input := Chal.ContingencyTables.Algorithms.encodeCountingInput r c b ε δ;
                        have t := Chal.ContingencyTables.Algorithms.countingTime C d r c b ε δ;
                        (∀ (bits : Fin t → Bool), ∃ q, 0 ≤ q ∧ Chal.MatchingFPRAS.Outputs A input bits q) ∧
                          (Chal.ContingencyTables.Algorithms.count r c b = 0 →
                              ∀ (bits : Fin t → Bool), Chal.MatchingFPRAS.Outputs A input bits 0) ∧
                            ↑(Chal.ContingencyTables.Algorithms.goodCountingTapes A r c b ε δ t).card / 2 ^ t ≥ 1 - δ
