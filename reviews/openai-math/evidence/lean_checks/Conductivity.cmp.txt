def Chal.ScalarConductivity.smoothJets : Set ↥Chal.ScalarConductivity.JetSpace :=
{z |
  ∃ f,
    ∃ (_ : ContDiff ℝ (↑⊤) f) (hf :
      MeasureTheory.MemLp (Chal.ScalarConductivity.smoothJet f) 2 Chal.ScalarConductivity.ballMeasure),
      z = MeasureTheory.MemLp.toLp (Chal.ScalarConductivity.smoothJet f) hf}

{z |
  ∃ f,
    ∃ (_ : ContDiff ℝ (↑⊤) f) (_ : HasCompactSupport f) (_ : tsupport f ⊆ Chal.ScalarConductivity.ball) (hf :
      MeasureTheory.MemLp (Chal.ScalarConductivity.smoothJet f) 2 Chal.ScalarConductivity.ballMeasure),
      z = MeasureTheory.MemLp.toLp (Chal.ScalarConductivity.smoothJet f) hf}

  ∀ (f : Chal.ScalarConductivity.TraceSpace),
    ∃ u,
      Chal.ScalarConductivity.trace u = f ∧
        Chal.ScalarConductivity.Harmonic γ u ∧
          (∀ (u' : ↥Chal.ScalarConductivity.H1),
              Chal.ScalarConductivity.trace u' = f → Chal.ScalarConductivity.Harmonic γ u' → u' = u) ∧

              (Λ f) (Chal.ScalarConductivity.trace v) = Chal.ScalarConductivity.energy γ u v
def Chal.ScalarConductivity.EqualOneNearBoundary : (Chal.ScalarConductivity.R3 → ℝ) → Prop :=

  ∃ r, 0 < r ∧ r < 3 ∧ ∀ᵐ (x : Chal.ScalarConductivity.R3) ∂Chal.ScalarConductivity.ballMeasure, r < ‖x‖ → γ x = 1
def Chal.ScalarConductivity.MainStatement : Prop :=
∃ γ₀ γ₁ c C Λ,
  0 < c ∧
    c < C ∧
      Measurable γ₀ ∧
        Measurable γ₁ ∧
          MeasureTheory.MemLp γ₀ ⊤ Chal.ScalarConductivity.ballMeasure ∧
            MeasureTheory.MemLp γ₁ ⊤ Chal.ScalarConductivity.ballMeasure ∧
              (∀ᵐ (x : Chal.ScalarConductivity.R3) ∂Chal.ScalarConductivity.ballMeasure, c ≤ γ₀ x ∧ γ₀ x ≤ C) ∧
                (∀ᵐ (x : Chal.ScalarConductivity.R3) ∂Chal.ScalarConductivity.ballMeasure, c ≤ γ₁ x ∧ γ₁ x ≤ C) ∧
                  Chal.ScalarConductivity.EqualOneNearBoundary γ₀ ∧
                    Chal.ScalarConductivity.EqualOneNearBoundary γ₁ ∧
                      0 < Chal.ScalarConductivity.ballMeasure {x | γ₀ x ≠ γ₁ x} ∧
                        Chal.ScalarConductivity.IsDirichletToNeumann γ₀ Λ ∧
                          Chal.ScalarConductivity.IsDirichletToNeumann γ₁ Λ
