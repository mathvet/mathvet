def OAI.NoiselessRegression.SubquadraticMemory.{u} : Prop :=
∃ c,
  0 < c ∧
    ∀ (M : ℕ → ℕ),
      ((fun d => ↑(M d)) =o[Filter.atTop] fun d => ↑d ^ 2) →
        ∃ d₀,
          ∀ (d : ℕ),
            d₀ ≤ d →
              ∀ (T : ℕ) (ε : ℝ),
                0 < ε →
                  ε ≤ 1 / 10 →
                    ∀ (Ω : Type u) [inst : MeasurableSpace Ω] (ρ : MeasureTheory.Measure Ω),
                      MeasureTheory.IsProbabilityMeasure ρ →
                        ∀ (L : OAI.NoiselessRegression.Learner d (M d) T Ω),
                          L.Admissible ρ →
                            (↑2 / ↑3 ≤ L.uniformSuccess ρ ε ∨
                                ∀ (s : ↑(OAI.NoiselessRegression.UnitSphere d)), ↑2 / ↑3 ≤ L.success ρ ε s) →
                              c * ↑d * Real.log (1 / ε) ≤ ↑T
def OAI.NoiselessRegression.FixedQuadraticMemory.{u} : Prop :=
