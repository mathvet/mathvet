def Chal.ThreeMachine.Instance.acyclic : {n : ℕ} → Chal.ThreeMachine.Instance n → Prop :=

def Chal.ThreeMachine.Feasible : {n : ℕ} → Chal.ThreeMachine.Instance n → ℕ → (Fin n → ℕ) → Prop :=
fun {n} G T τ =>
  (∀ (v : Fin n), 1 ≤ τ v ∧ τ v ≤ T) ∧ (∀ (t : ℕ), {v | τ v = t}.card ≤ 3) ∧ ∀ (u v : Fin n), G.edge u v → τ u < τ v
def Chal.ThreeMachine.encodeNat : ℕ → List Bool :=

def Chal.ThreeMachine.encodeInput : {n : ℕ} → Chal.ThreeMachine.Instance n → Option ℕ → List Bool :=
fun {n} G deadline =>
  (match deadline with
      | none => [false]
      | some T => true :: Chal.ThreeMachine.encodeNat T) ++
      Chal.ThreeMachine.encodeList Chal.ThreeMachine.encodeNat (List.map (fun x => x + 1) (List.range n)) ++
    Chal.ThreeMachine.encodeList

def Chal.ThreeMachine.encodeAnswer : Option (List ℕ) → List Bool :=
fun x =>
  match x with
  | none => [false]
  | some slots => true :: Chal.ThreeMachine.encodeList Chal.ThreeMachine.encodeNat slots
def Chal.ThreeMachine.slots : {n : ℕ} → (Fin n → ℕ) → List ℕ :=
fun {n} τ => List.ofFn τ
def Chal.ThreeMachine.CorrectOutput : {n : ℕ} → Chal.ThreeMachine.Instance n → Option ℕ → List Bool → Prop :=
fun {n} G deadline out =>
  match deadline with
  | none =>
    ∃ T τ,
      Chal.ThreeMachine.Feasible G T τ ∧
        (∀ (T' : ℕ) (τ' : Fin n → ℕ), Chal.ThreeMachine.Feasible G T' τ' → T ≤ T') ∧
          out = Chal.ThreeMachine.encodeAnswer (some (Chal.ThreeMachine.slots τ))
  | some T =>
    (out = Chal.ThreeMachine.encodeAnswer none ∧ ¬∃ τ, Chal.ThreeMachine.Feasible G T τ) ∨
      ∃ τ, Chal.ThreeMachine.Feasible G T τ ∧ out = Chal.ThreeMachine.encodeAnswer (some (Chal.ThreeMachine.slots τ))

fun {g} b => if b = true then ⟨2, ⋯⟩ else ⟨1, ⋯⟩
def Chal.ThreeMachine.Machine.step : {k q g : ℕ} →
  Chal.ThreeMachine.Machine k q g →
    Chal.ThreeMachine.Configuration k q g → Option (Chal.ThreeMachine.Configuration k q g) :=
fun {k q g} M c => do
  let __x ← M.transition c.state fun i => (c.tapes i).head
  match __x with
    | (state, action) =>
      pure
        { state := state,
          tapes := fun i =>
            have tape := Turing.Tape.write (action i).1 (c.tapes i);
            match (action i).2 with
            | none => tape
            | some d => Turing.Tape.move d tape }
def Chal.ThreeMachine.Machine.initial : {k q g : ℕ} →
  Chal.ThreeMachine.Machine k q g → List Bool → Chal.ThreeMachine.Configuration k q g :=
fun {k q g} _M input =>
  { state := 0, tapes := fun i => Turing.Tape.mk₁ (if i = 0 then List.map Chal.ThreeMachine.bitSymbol input else []) }
def Chal.ThreeMachine.Machine.run : {k q g : ℕ} →
  Chal.ThreeMachine.Machine k q g → List Bool → ℕ → Chal.ThreeMachine.Configuration k q g :=
fun {k q g} M input t => (fun c => (M.step c).getD c)^[t] (M.initial input)
def Chal.ThreeMachine.Machine.Produces : {k q g : ℕ} →
  Chal.ThreeMachine.Machine k q g → List Bool → List Bool → ℕ → Prop :=
fun {k q g} M input out timeBound =>
  ∃ t ≤ timeBound,
    M.step (M.run input t) = none ∧

def Chal.ThreeMachine.ReleaseTheorem : Prop :=
∃ k q g M C,
  0 < C ∧
    ∀ (n : ℕ) (G : Chal.ThreeMachine.Instance n),
      1 ≤ n →
        G.acyclic →
          ∀ (deadline : Option ℕ),
            (∀ T ∈ deadline, 1 ≤ T ∧ T ≤ n) →
              ∃ out,
                Chal.ThreeMachine.CorrectOutput G deadline out ∧
                  M.Produces (Chal.ThreeMachine.encodeInput G deadline) out
                    (C * ((Chal.ThreeMachine.encodeInput G deadline).length + 2) ^ 150020)
