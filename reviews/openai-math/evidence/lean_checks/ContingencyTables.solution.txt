def OAI.MatchingFPRAS.tick : (A : OAI.MatchingFPRAS.RandomMachine) →
  OAI.MatchingFPRAS.Configuration A → Bool → OAI.MatchingFPRAS.Configuration A :=
fun A c bit =>
  match c.state with
  | none => c
  | some s =>
    match A.transition s c.tape.head bit with
    | none => { state := none, tape := c.tape }
    | some (s', action) =>
      { state := some s',
        tape :=
          match action with
          | Turing.TM0.Stmt.move d => Turing.Tape.move d c.tape
          | Turing.TM0.Stmt.write a => Turing.Tape.write a c.tape }

def OAI.ContingencyTables.Algorithms.BoundedSamplingStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (m n : ℕ) (r : Fin m → ℕ) (c : Fin n → ℕ),
      ∑ i, r i = ∑ j, c j →
        ∀ (k : ℕ),
          1 ≤ k →
            have input := OAI.ContingencyTables.Algorithms.encodeSamplingInput r c k;
            have t := C * OAI.ContingencyTables.Algorithms.samplingSize n r k ^ d;
            (∀ (bits : Fin t → Bool), ∃ X, OAI.ContingencyTables.Algorithms.OutputsTable A input bits ↑X) ∧
              (∑ X,
                    |OAI.ContingencyTables.Algorithms.tableMass A input t ↑X -
                        (↑(Fintype.card (OAI.ContingencyTables.Table r c)))⁻¹|) /
                  2 ≤
                (2 ^ k)⁻¹
def OAI.ContingencyTables.Algorithms.ExactSamplingStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (m n : ℕ) (r : Fin m → ℕ) (c : Fin n → ℕ),
      ∑ i, r i = ∑ j, c j →
        have input := OAI.ContingencyTables.Algorithms.encodeMargins r c;
        have tail := fun t => 1 - OAI.ContingencyTables.Algorithms.haltMass A input t;
        (∀ (t : ℕ) (bits : Fin t → Bool),
            (OAI.MatchingFPRAS.run A input bits).state = none →
              ∃ X, OAI.ContingencyTables.Algorithms.OutputsTable A input bits ↑X) ∧
          (∀ (X : OAI.ContingencyTables.Table r c),
              Filter.Tendsto (fun t => OAI.ContingencyTables.Algorithms.tableMass A input t ↑X) Filter.atTop
                (nhds (↑(Fintype.card (OAI.ContingencyTables.Table r c)))⁻¹)) ∧
            Filter.Tendsto (OAI.ContingencyTables.Algorithms.haltMass A input) Filter.atTop (nhds 1) ∧
              Summable tail ∧ tsum tail ≤ ↑C * ↑(OAI.ContingencyTables.Algorithms.samplingSize n r 0) ^ d

def OAI.ContingencyTables.Algorithms.countingTime : ℕ →
  ℕ → {m n : ℕ} → (Fin m → ℕ) → (Fin n → ℕ) → (Fin m → Fin n → ℕ) → ℚ → ℚ → ℕ :=
fun C d {m n} r c b ε δ =>
  C * ((OAI.ContingencyTables.Algorithms.encodeCountingInput r c b ε δ).length + ⌈ε⁻¹⌉₊ + Nat.clog 2 ⌈δ⁻¹⌉₊ + 1) ^ d
def OAI.ContingencyTables.Algorithms.goodCountingTapes : OAI.ContingencyTables.Algorithms.Machine →
  {m n : ℕ} → (Fin m → ℕ) → (Fin n → ℕ) → (Fin m → Fin n → ℕ) → ℚ → ℚ → (t : ℕ) → Finset (Fin t → Bool) :=
fun A {m n} r c b ε δ t =>
  {bits |
    ∃ q,
      OAI.MatchingFPRAS.Outputs A (OAI.ContingencyTables.Algorithms.encodeCountingInput r c b ε δ) bits q ∧
        (1 - ε) * ↑(OAI.ContingencyTables.Algorithms.count r c b) ≤ q ∧
          q ≤ (1 + ε) * ↑(OAI.ContingencyTables.Algorithms.count r c b)}
def OAI.ContingencyTables.Algorithms.CountingStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (m n : ℕ),
      0 < m →
        0 < n →
          ∀ (r : Fin m → ℕ) (c : Fin n → ℕ) (b : Fin m → Fin n → ℕ),
            ∑ i, r i = ∑ j, c j →
              ∀ (ε δ : ℚ),
                0 < ε →
                  ε < 1 →
                    0 < δ →
                      δ < 1 →
                        have input := OAI.ContingencyTables.Algorithms.encodeCountingInput r c b ε δ;
                        have t := OAI.ContingencyTables.Algorithms.countingTime C d r c b ε δ;
                        (∀ (bits : Fin t → Bool), ∃ q, 0 ≤ q ∧ OAI.MatchingFPRAS.Outputs A input bits q) ∧
                          (OAI.ContingencyTables.Algorithms.count r c b = 0 →
                              ∀ (bits : Fin t → Bool), OAI.MatchingFPRAS.Outputs A input bits 0) ∧
                            ↑(OAI.ContingencyTables.Algorithms.goodCountingTapes A r c b ε δ t).card / 2 ^ t ≥ 1 - δ
