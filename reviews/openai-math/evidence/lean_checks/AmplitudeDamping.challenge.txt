def OAI.GAD.kraus : ℝ → ℝ → Fin 4 → Matrix (Fin 2) (Fin 2) ℂ :=
fun γ ν r =>
  if r = 0 then !![↑√(1 - ν), 0; 0, ↑√(1 - ν) * ↑√(1 - γ)]
  else
    if r = 1 then !![0, ↑√(γ * (1 - ν)); 0, 0]
    else if r = 2 then !![↑√ν * ↑√(1 - γ), 0; 0, ↑√ν] else !![0, 0; ↑√(γ * ν), 0]

def OAI.GAD.averageError : ℝ → ℝ → {n : ℕ} → OAI.GAD.Code n → ℝ :=
fun γ ν {n} C => 1 - (∑ a, (C.decoding a * OAI.GAD.channel γ ν n (C.encoding a)).trace.re) / ↑C.messages
def OAI.GAD.Achievable : ℝ → ℝ → ℝ → Prop :=
fun γ ν R =>
  ∃ C,
    Filter.Tendsto (fun n => OAI.GAD.averageError γ ν (C n)) Filter.atTop (nhds 0) ∧
      ∀ (ε : ℝ), 0 < ε → ∀ᶠ (n : ℕ) in Filter.atTop, R - ε ≤ Real.log ↑(C n).messages / Real.log 2 / ↑n
def OAI.GAD.capacity : ℝ → ℝ → ℝ :=
fun γ ν => sSup {R | OAI.GAD.Achievable γ ν R}
def OAI.GAD.g : ℝ → ℝ :=
fun u => Real.binEntropy ((1 + √(1 - 4 * u)) / 2)
def OAI.GAD.v : ℝ → ℝ → ℝ → ℝ :=
fun γ ν p => γ * ν * (1 - ν) + γ * (1 - γ) * (p - ν) ^ 2
def OAI.GAD.objective : ℝ → ℝ → ℝ → ℝ :=
fun γ ν p => Real.binEntropy ((1 - γ) * p + γ * ν) - OAI.GAD.g (OAI.GAD.v γ ν p)
def OAI.GAD.Maximizes : ℝ → ℝ → ℝ → Prop :=
fun γ ν p => p ∈ Set.Icc 0 1 ∧ ∀ q ∈ Set.Icc 0 1, OAI.GAD.objective γ ν q ≤ OAI.GAD.objective γ ν p
def OAI.GAD.phaseSignal : ℝ → Fin 2 → Fin 2 → ℂ :=
fun p s b => if b = 0 then ↑√(1 - p) else if s = 0 then ↑√p else -↑√p

fun p {n} s i => ∏ k, OAI.GAD.phaseSignal p (s k) (i k)

fun p {n} s i j => OAI.GAD.phaseProduct p s i * star (OAI.GAD.phaseProduct p s j)
