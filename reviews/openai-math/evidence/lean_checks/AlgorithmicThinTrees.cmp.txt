def Chal.AlgorithmicThinTrees.AlgorithmicStrongThinTrees : Prop :=
∃ C,
  0 < C ∧
    ∃ M a degree,
      0 < a ∧
        ∀ (x : Chal.AlgorithmicThinTrees.Input),
          Chal.AlgorithmicThinTrees.Valid x →
            have result := M.run x.encode (a * (x.length + 1) ^ degree);
            result.state = none ∧ Chal.AlgorithmicThinTrees.GoodOutput C x (result.stacks M.outputStack)
