def Chal.SubpolynomialLp.coordinateDistance : ℝ → {d : ℕ} → (Fin d → ℝ) → (Fin d → ℝ) → ℝ :=
fun p {d} x y => (∑ a, |x a - y a| ^ p) ^ (1 / p)
