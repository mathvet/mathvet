def OAI.CoveringOrder.OptimalOrder : Prop :=
∃ c C,
  0 < c ∧
    0 < C ∧
      ∃ n₀,
        ∀ (n : ℕ),
          n₀ ≤ n →
            OAI.CoveringOrder.orderBounds c C n (OAI.CoveringOrder.translativeSup n) ∧
              OAI.CoveringOrder.orderBounds c C n (OAI.CoveringOrder.symmetricTranslativeSup n) ∧
                OAI.CoveringOrder.orderBounds c C n (OAI.CoveringOrder.latticeSup n) ∧
                  OAI.CoveringOrder.orderBounds c C n (OAI.CoveringOrder.symmetricLatticeSup n)
