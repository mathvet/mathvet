  match w with

  | b :: bs => { left := [], head := some b, right := List.map some bs }
def OAI.UniformKServer.BitTape.shift : OAI.UniformKServer.BitTape →
  OAI.UniformKServer.HeadMove → OAI.UniformKServer.BitTape :=
fun T x =>
  match x with
  | OAI.UniformKServer.HeadMove.stay => T
  | OAI.UniformKServer.HeadMove.left =>
    match T.left with
    | [] => { left := [], head := none, right := T.head :: T.right }
    | a :: as => { left := as, head := a, right := T.head :: T.right }
  | OAI.UniformKServer.HeadMove.right =>
    match T.right with
    | [] => { left := T.head :: T.left, head := none, right := [] }
    | a :: as => { left := T.head :: T.left, head := a, right := as }

def OAI.UniformKServer.BitMachine.initial : (M : OAI.UniformKServer.BitMachine) →
  List Bool → OAI.UniformKServer.MachineState M :=
fun M input =>
  { control := M.start, input := OAI.UniformKServer.BitTape.ofWord input,
    work := fun x => OAI.UniformKServer.BitTape.blank, outputRev := [], yielded := false }
def OAI.UniformKServer.BitMachine.step : (M : OAI.UniformKServer.BitMachine) →
  OAI.UniformKServer.MachineState M → Bool → OAI.UniformKServer.MachineState M :=
fun M s coin =>
  if s.yielded = true then s
  else
    have a := M.transition s.control s.input.head (fun i => (s.work i).head) coin;
    { control := a.control, input := s.input.shift a.inputMove,
      work := fun i =>
        (have __src := s.work i;
            { left := __src.left, head := a.write i, right := __src.right }).shift
          (a.workMove i),
      outputRev :=
        match a.emit with
        | none => s.outputRev
        | some b => b :: s.outputRev,
      yielded := a.yield }

fun M s coins => List.foldl M.step s coins

def OAI.UniformKServer.UniformAlgorithmClaim : Prop :=
∃ M c e C,
  0 < c ∧
    0 < C ∧
      ∀ (n k : ℕ),
        3 ≤ n →
          ∀ (hk : 2 ≤ k),
            k < n →
              ∀ (d : OAI.UniformKServer.RationalMetric n) (s : OAI.UniformKServer.Configuration n k),
                have L := (OAI.UniformKServer.instanceCode d s).length;
                (OAI.UniformKServer.boot M c e d s).yielded = true ∧
                  (∀ (t : ℕ) (z : OAI.UniformKServer.MachineState M),
                      OAI.UniformKServer.Reachable M c e d s t z →
                        ∀ (r : Fin n) (bits : Fin (OAI.UniformKServer.requestBudget c e L (t + 1)) → Bool),
                          have z' := OAI.UniformKServer.requestStep M c e L (t + 1) z r bits;
                          z'.yielded = true ∧ OAI.UniformKServer.outputValue z' < k) ∧
                    ∃ B,
                      0 ≤ B ∧
                        ∀ (w : List (Fin n)),
                          OAI.UniformKServer.machineExpectedCost M c e L d ⋯ 0 (OAI.UniformKServer.boot M c e d s) s w ≤
                            C * Real.log (↑k + 1) ^ 2 * OAI.UniformKServer.offlineCost d s w + B
