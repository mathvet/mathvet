def OAI.RVM.Equations : OAI.RVM.Solution → Prop :=
fun s =>
  ∀ (t : ℝ),
    0 ≤ t →
      (∀ (x v : OAI.RVM.Vec),
          derivWithin (fun t' => s.f (t', x, v)) (Set.Ici 0) t +
                ∑ i, (OAI.RVM.velocity v).ofLp i * OAI.RVM.spatialPartial (fun y => s.f (t, y, v)) x i +
              ∑ i,
                (s.E (t, x) + OAI.RVM.cross (OAI.RVM.velocity v) (s.B (t, x))).ofLp i *
                  OAI.RVM.spatialPartial (fun v' => s.f (t, x, v')) v i =
            0) ∧
        (∀ (x : OAI.RVM.Vec),
            derivWithin (fun t' => s.E (t', x)) (Set.Ici 0) t - OAI.RVM.curl (fun y => s.B (t, y)) x =
              -OAI.RVM.current (fun z => s.f (t, z)) x) ∧
          (∀ (x : OAI.RVM.Vec),
              derivWithin (fun t' => s.B (t', x)) (Set.Ici 0) t + OAI.RVM.curl (fun y => s.E (t, y)) x = 0) ∧
            (∀ (x : OAI.RVM.Vec), OAI.RVM.divergence (fun y => s.E (t, y)) x = OAI.RVM.rho (fun z => s.f (t, z)) x) ∧
              ∀ (x : OAI.RVM.Vec), OAI.RVM.divergence (fun y => s.B (t, y)) x = 0

def OAI.RVM.Classical : OAI.RVM.Datum → OAI.RVM.Solution → Prop :=
fun d s =>
  ContDiffOn ℝ 1 s.f OAI.RVM.particleHalfSpace ∧
    ContDiffOn ℝ 1 s.E OAI.RVM.fieldHalfSpace ∧
      ContDiffOn ℝ 1 s.B OAI.RVM.fieldHalfSpace ∧
        (∀ (t : ℝ), 0 ≤ t → ∀ (z : OAI.RVM.Phase), 0 ≤ s.f (t, z)) ∧
          OAI.RVM.TimeContinuousL2 s.E ∧
            OAI.RVM.TimeContinuousL2 s.B ∧
              OAI.RVM.CompactOnFiniteHorizons s ∧
                OAI.RVM.Equations s ∧
                  (∀ (z : OAI.RVM.Phase), s.f (0, z) = d.f₀ z) ∧
                    (∀ (x : OAI.RVM.Vec), s.E (0, x) = d.E₀ x) ∧ ∀ (x : OAI.RVM.Vec), s.B (0, x) = d.B₀ x
