def Chal.MatchingFPRAS.tick : (A : Chal.MatchingFPRAS.RandomMachine) →
  Chal.MatchingFPRAS.Configuration A → Bool → Chal.MatchingFPRAS.Configuration A :=
fun A c bit =>
  match c.state with
  | none => c
  | some s =>
    match A.transition s c.tape.head bit with
    | none => { state := none, tape := c.tape }
    | some (s', action) =>
      { state := some s',
        tape :=
          match action with
          | Turing.TM0.Stmt.move d => Turing.Tape.move d c.tape
          | Turing.TM0.Stmt.write a => Turing.Tape.write a c.tape }

fun A input => { state := some 0, tape := Turing.Tape.mk₁ input }

  List Chal.MatchingFPRAS.Symbol → {t : ℕ} → (Fin t → Bool) → Chal.MatchingFPRAS.Configuration A :=
fun A input {t} bits => List.foldl (Chal.MatchingFPRAS.tick A) (Chal.MatchingFPRAS.initial A input) (List.ofFn bits)

def Chal.MatchingFPRAS.goodTapes : Chal.MatchingFPRAS.RandomMachine →
  Chal.MatchingFPRAS.GraphInput → ℚ → ℚ → (t : ℕ) → Finset (Fin t → Bool) :=
fun A G ε δ t =>
  {bits |
    ∃ q,
      Chal.MatchingFPRAS.Outputs A (Chal.MatchingFPRAS.encodeInput G ε δ) bits q ∧
        (1 - ε) * ↑(Chal.MatchingFPRAS.Z G) ≤ q ∧ q ≤ (1 + ε) * ↑(Chal.MatchingFPRAS.Z G)}
def Chal.MatchingFPRAS.MainStatement : Prop :=
∃ A C d,
  0 < C ∧
    ∀ (G : Chal.MatchingFPRAS.GraphInput) (ε δ : ℚ),
      0 < ε →
        ε < 1 →
          0 < δ →
            δ < 1 / 2 →
              have t := Chal.MatchingFPRAS.timeBound C d G ε δ;
              (∀ (bits : Fin t → Bool),
                  ∃ q, 0 ≤ q ∧ Chal.MatchingFPRAS.Outputs A (Chal.MatchingFPRAS.encodeInput G ε δ) bits q) ∧
                (Chal.MatchingFPRAS.Z G = 0 →
                    ∀ (bits : Fin t → Bool),
                      Chal.MatchingFPRAS.Outputs A (Chal.MatchingFPRAS.encodeInput G ε δ) bits 0) ∧
                  ↑(Chal.MatchingFPRAS.goodTapes A G ε δ t).card / 2 ^ t ≥ 1 - δ
