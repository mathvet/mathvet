    (∀ (x : OAI.HarmonicCounterexample.Space n), OAI.HarmonicCounterexample.laplaceBeltrami g u x = 0) ∧
      ∃ C,

def OAI.HarmonicCounterexample.euclideanDimension : ℕ → ℕ → ℕ :=
fun n k => (n + k - 1).choose k + (n + k - 2).choose (k - 1)
def OAI.HarmonicCounterexample.MainClaim : Prop :=
∃ n k g,
  Even n ∧
    8 ≤ n ∧
      2 ≤ k ∧
        OAI.HarmonicCounterexample.Complete g ∧
          OAI.HarmonicCounterexample.RicciNonnegative g ∧
            OAI.HarmonicCounterexample.EuclideanNearOrigin g ∧
              OAI.HarmonicCounterexample.PositiveSubunitAVR g ∧
                ∃ u,
                  LinearIndependent ℝ u ∧
                    ∀ (i : Fin (OAI.HarmonicCounterexample.euclideanDimension n k + 1)),
                      OAI.HarmonicCounterexample.HarmonicGrowth g k (u i)
