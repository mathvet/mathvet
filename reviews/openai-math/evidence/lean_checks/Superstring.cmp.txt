Chal.Superstring.main : ∃ f,
  Chal.Superstring.HasPolynomialImplementation f ∧
    ∀ (S : Chal.Superstring.Instance),
      Chal.Superstring.IsCommonSuperstring S (f S) ∧ List.length (f S) ≤ 2 * Chal.Superstring.opt S

def Chal.Superstring.IsCommonSuperstring : Chal.Superstring.Instance → Chal.Superstring.Word → Prop :=
