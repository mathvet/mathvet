def Chal.UniformKServer.BitMachine.step : (M : Chal.UniformKServer.BitMachine) →
  Chal.UniformKServer.MachineState M → Bool → Chal.UniformKServer.MachineState M :=
fun M s coin =>
  if s.yielded = true then s
  else
    have a := M.transition s.control s.input.head (fun i => (s.work i).head) coin;
    { control := a.control, input := s.input.shift a.inputMove,
      work := fun i =>
        (have __src := s.work i;
            { left := __src.left, head := a.write i, right := __src.right }).shift
          (a.workMove i),
      outputRev :=
        match a.emit with
        | none => s.outputRev
        | some b => b :: s.outputRev,
      yielded := a.yield }

def Chal.UniformKServer.UniformAlgorithmClaim : Prop :=

  0 < c ∧
    0 < C ∧
      ∀ (n k : ℕ),
        3 ≤ n →
          ∀ (hk : 2 ≤ k),
            k < n →

                have L := (Chal.UniformKServer.instanceCode d s).length;
                (Chal.UniformKServer.boot M c e d s).yielded = true ∧
                  (∀ (t : ℕ) (z : Chal.UniformKServer.MachineState M),
