      ContDiff ℝ (↑⊤) φ →
        HasCompactSupport φ →
          ∫ (x : Chal.Coulomb.Configuration n), self.value s x * ↑((fderiv ℝ φ x) (EuclideanSpace.single a 1)) =
            -∫ (x : Chal.Coulomb.Configuration n), self.gradient s a x * ↑(φ x)
constructor:

    (value_L2 : ∀ (s : Chal.Coulomb.Spins n), MeasureTheory.MemLp (value s) 2 MeasureTheory.volume)
    (partial_L2 :
      ∀ (s : Chal.Coulomb.Spins n) (a : Fin n × Fin 3), MeasureTheory.MemLp (gradient s a) 2 MeasureTheory.volume)

def Chal.Coulomb.Antisymmetric : {n : ℕ} → Chal.Coulomb.H1Vector n → Prop :=
fun {n} ψ =>
  ∀ (p : Equiv.Perm (Fin n)) (s : Chal.Coulomb.Spins n),
    ∀ᵐ (x : Chal.Coulomb.Configuration n),
      ψ.value (s ∘ ⇑p) (Chal.Coulomb.permute p x) = ↑↑(Equiv.Perm.sign p) * ψ.value s x
def Chal.Coulomb.mass : {n : ℕ} → Chal.Coulomb.H1Vector n → ℝ :=
fun {n} ψ => ∑ s, ∫ (x : Chal.Coulomb.Configuration n), ‖ψ.value s x‖ ^ 2
def Chal.Coulomb.kinetic : {n : ℕ} → Chal.Coulomb.H1Vector n → ℝ :=
fun {n} ψ => 1 / 2 * ∑ s, ∑ a, ∫ (x : Chal.Coulomb.Configuration n), ‖ψ.gradient s a x‖ ^ 2
def Chal.Coulomb.nuclearEnergy : {M n : ℕ} → Chal.Coulomb.Nuclei M → Chal.Coulomb.H1Vector n → ℝ :=
fun {M n} S ψ =>
  ∑ s,
    ∫ (x : Chal.Coulomb.Configuration n),
      (∑ i, Chal.Coulomb.attraction S (Chal.Coulomb.position x i)) * ‖ψ.value s x‖ ^ 2
def Chal.Coulomb.pairEnergy : {n : ℕ} → Chal.Coulomb.H1Vector n → ℝ :=
fun {n} ψ =>
  ∑ s,
    ∫ (x : Chal.Coulomb.Configuration n),
      (∑ i,
          ∑ j,
            if i < j then Chal.Coulomb.coulombKernel (Chal.Coulomb.position x i - Chal.Coulomb.position x j) else 0) *
        ‖ψ.value s x‖ ^ 2
def Chal.Coulomb.form : {M n : ℕ} → Chal.Coulomb.Nuclei M → Chal.Coulomb.H1Vector n → ℝ :=
fun {M n} S ψ => Chal.Coulomb.kinetic ψ - Chal.Coulomb.nuclearEnergy S ψ + Chal.Coulomb.pairEnergy ψ

def Chal.ContinuumCoulomb.QMAGate.WellFormed : ℕ → Chal.ContinuumCoulomb.QMAGate → Prop :=
fun n x =>
  match x with
  | Chal.ContinuumCoulomb.QMAGate.hadamard i => i < n
  | Chal.ContinuumCoulomb.QMAGate.phaseT i => i < n
  | Chal.ContinuumCoulomb.QMAGate.controlledNot i j => i < n ∧ j < n ∧ i ≠ j

def Chal.ContinuumCoulomb.qmaHadamard : Matrix (Fin 2) (Fin 2) ℂ :=
Chal.ContinuumCoulomb.bellScale • !![1, 1; 1, -1]
def Chal.ContinuumCoulomb.qmaPhaseT : Matrix (Fin 2) (Fin 2) ℂ :=
!![1, 0; 0, Chal.ContinuumCoulomb.bellScale * (1 + Complex.I)]
def Chal.ContinuumCoulomb.qmaQubit : (work : ℕ) → ℕ → Fin (work + 1) :=
fun work index => ⟨index % (work + 1), ⋯⟩
def Chal.ContinuumCoulomb.qmaControlledNot : (work : ℕ) →
  Fin (work + 1) →
    Fin (work + 1) →
      Chal.ContinuumCoulomb.SourceSpinBasis (work + 1) → Chal.ContinuumCoulomb.SourceSpinBasis (work + 1) :=
fun work control target s i => if i = target ∧ s control = 1 then (Equiv.swap 0 1) (s i) else s i
def Chal.ContinuumCoulomb.qmaGateMatrix : (work : ℕ) →
  Chal.ContinuumCoulomb.QMAGate →
    Matrix (Chal.ContinuumCoulomb.SourceSpinBasis (work + 1)) (Chal.ContinuumCoulomb.SourceSpinBasis (work + 1)) ℂ :=
fun work x =>
  match x with
  | Chal.ContinuumCoulomb.QMAGate.hadamard i =>
    Chal.ContinuumCoulomb.sourceTensor (work + 1) fun k =>
      if k = Chal.ContinuumCoulomb.qmaQubit work i then Chal.ContinuumCoulomb.qmaHadamard else 1
  | Chal.ContinuumCoulomb.QMAGate.phaseT i =>
    Chal.ContinuumCoulomb.sourceTensor (work + 1) fun k =>
      if k = Chal.ContinuumCoulomb.qmaQubit work i then Chal.ContinuumCoulomb.qmaPhaseT else 1
  | Chal.ContinuumCoulomb.QMAGate.controlledNot i j => fun s t =>
    if
        s =
          Chal.ContinuumCoulomb.qmaControlledNot work (Chal.ContinuumCoulomb.qmaQubit work i)
            (Chal.ContinuumCoulomb.qmaQubit work j) t then
      1
    else 0
def Chal.ContinuumCoulomb.qmaCircuitMatrix : (c : Chal.ContinuumCoulomb.QMACircuit) →
  Matrix (Chal.ContinuumCoulomb.SourceSpinBasis (c.work + 1)) (Chal.ContinuumCoulomb.SourceSpinBasis (c.work + 1)) ℂ :=
fun c => List.foldl (fun M g => Chal.ContinuumCoulomb.qmaGateMatrix c.work g * M) 1 c.gates
def Chal.ContinuumCoulomb.qmaInitialState : (c : Chal.ContinuumCoulomb.QMACircuit) →
  c.WellFormed →
    EuclideanSpace ℂ (Chal.ContinuumCoulomb.SourceSpinBasis c.witness) →
      Chal.ContinuumCoulomb.SourceSpinBasis (c.work + 1) → ℂ :=
fun c hc psi s => if ∀ (i : Fin (c.work + 1)), c.witness ≤ ↑i → s i = 0 then psi.ofLp fun i => s ⟨↑i, ⋯⟩ else 0
def Chal.ContinuumCoulomb.qmaAcceptance : (c : Chal.ContinuumCoulomb.QMACircuit) →
  c.WellFormed → EuclideanSpace ℂ (Chal.ContinuumCoulomb.SourceSpinBasis c.witness) → ℝ :=
fun c hc psi =>
  ∑ s,
    if s (Fin.last c.work) = 1 then
      Complex.normSq
        ((Chal.ContinuumCoulomb.qmaCircuitMatrix c).mulVec (Chal.ContinuumCoulomb.qmaInitialState c hc psi) s)
    else 0
